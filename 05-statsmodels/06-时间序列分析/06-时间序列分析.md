时间序列分析是统计学中最迷人也最实用的领域之一！从股票价格预测到天气预报，从销售预测到疫情追踪，时间序列无处不在。

## 从生活中理解时间序列

```python
# 1. 体重有没有整体下降的趋势？（趋势）
```

### 什么是时间序列？

```python
# 时间序列的核心特征
时间序列 = {
    "定义": "按时间顺序排列的数据点",
    "特点": [
        "有时间顺序（不能随意打乱）",
        "相邻观测往往相关（今天和昨天的温度相似）",
        "可能有趋势（长期上升或下降）",
        "可能有季节性（周期性重复模式）"
    ],
    "例子": [
        "股票价格：每秒/分/小时的收盘价",
        "温度：每天的最高温",
        "销售：每月的销售额",
        "GDP：每季度的国内生产总值"
    ]
}
```

## 时间序列的组成部分

时间序列 = 趋势 + 季节性 + 周期性 + 随机噪声

### 1. 趋势 (Trend)

```python
# 长期的上升或下降趋势
# 例：经济增长、人口增加、气候变暖

趋势类型 = {
    "线性趋势": "匀速增长/下降",
    "指数趋势": "加速增长/下降",
    "无趋势": "上下波动但没有明显方向"
}
```

### 2. 季节性 (Seasonality)

```python
# 固定周期的重复模式
季节性模式 = {
    "每天": "上午高峰、下午低谷（交通流量）",
    "每周": "周末高、工作日低（电影票房）",
    "每月": "月初高、月末低（工资发放后的消费）",
    "每年": "夏季高、冬季低（冰淇淋销量）"
}
```

### 3. 周期性 (Cycle)

```python
# 非固定周期的波动
# 例：经济周期（3-7年不等）、库存周期

# 与季节性的区别：
季节性 = "周期固定（如每年12月）"
周期性 = "周期不固定（如经济危机间隔不定）"
```

### 4. 随机噪声 (Random Noise)

```python
# 无法解释的随机波动
# 例：测量误差、突发事件
```

## 平稳性：时间序列的核心概念

### 什么是平稳性？

```python
# 平稳序列的特征
平稳序列 = {
    "均值不变": "长期平均值保持恒定",
    "方差不变": "波动幅度保持恒定",
    "无趋势": "不存在上升或下降趋势",
    "无季节性": "不存在周期性模式"
}

# 生活类比
非平稳 = "你的收入逐年增长（有趋势）"
平稳 = "你每天的心率波动（围绕均值上下波动）"
```

### 为什么平稳性重要？

```python
# 大多数时间序列模型（如 ARIMA）要求数据平稳

原因 = {
    "统计性质稳定": "平稳数据的均值、方差不随时间变化",
    "模型假设": "很多模型假设参数不随时间变化",
    "预测可靠": "平稳序列的未来行为更可预测"
}
```

### 检验平稳性：ADF检验

```python
from statsmodels.tsa.stattools import adfuller

# ADF (Augmented Dickey-Fuller) 检验
# H0: 序列非平稳（有单位根）
# H1: 序列平稳

def check_stationarity(timeseries, name=''):
    """检验时间序列的平稳性"""
    print(f'===== {name} 的平稳性检验 =====')

    # ADF检验
    result = adfuller(timeseries, autolag='AIC')

    print(f'ADF统计量: {result[0]:.4f}')
    print(f'p值: {result[1]:.4f}')
    print(f'临界值:')
    for key, value in result[4].items():
        print(f'  {key}: {value:.3f}')

    # 判断
    if result[1] <= 0.05:
        print('✅ 结论：序列平稳 (p < 0.05)')
    else:
        print('❌ 结论：序列非平稳 (p >= 0.05)')

    return result[1] <= 0.05
```

## 案例1：航空乘客数量预测（完整流程）

```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
from statsmodels.tsa.seasonal import seasonal_decompose
from statsmodels.tsa.stattools import adfuller, acf, pacf
from statsmodels.tsa.arima.model import ARIMA
from statsmodels.graphics.tsaplots import plot_acf, plot_pacf
import warnings

# 设置中文显示

# ===== 1. 生成模拟数据 =====
print("=" * 60)
print("案例1：航空乘客数量预测")
print("=" * 60)

np.random.seed(42)
n_months = 144  # 12年的月度数据

# 创建日期索引
dates = pd.date_range(start='2012-01', periods=n_months, freq='ME')

# 生成数据：趋势 + 季节性 + 噪声
trend = np.linspace(100, 500, n_months)  # 线性增长趋势
seasonal = 50 * np.sin(np.arange(n_months) * 2 * np.pi / 12)  # 年度季节性
noise = np.random.randn(n_months) * 10  # 随机噪声

passengers = trend + seasonal + noise

# 创建DataFrame
df = pd.DataFrame({
    '乘客数': passengers
}, index=dates)

print(f"\n数据概览：")
print(f"时间范围: {df.index[0]} 到 {df.index[-1]}")
print(f"观测数量: {len(df)}")
print(f"\n描述统计:")
print(df.describe())

# ===== 2. 数据可视化 =====
fig, axes = plt.subplots(3, 1, figsize=(14, 10))

# 原始数据
axes[0].plot(df.index, df['乘客数'], linewidth=2)
axes[0].set_title('原始数据：月度航空乘客数', fontsize=14, fontweight='bold')
axes[0].set_ylabel('乘客数', fontsize=12)
axes[0].grid(True, alpha=0.3)

# 滚动统计
rolling_mean = df['乘客数'].rolling(window=12).mean()
rolling_std = df['乘客数'].rolling(window=12).std()

axes[1].plot(df.index, df['乘客数'], label='原始数据', alpha=0.5)
axes[1].plot(df.index, rolling_mean, label='12个月滚动均值', color='red', linewidth=2)
axes[1].plot(df.index, rolling_std, label='12个月滚动标准差', color='green', linewidth=2)
axes[1].set_title('滚动统计（检查平稳性）', fontsize=14, fontweight='bold')
axes[1].set_ylabel('值', fontsize=12)
axes[1].legend()
axes[1].grid(True, alpha=0.3)

# 对数变换（如果数据有明显的方差增长）
# 这里只是演示，我们的模拟数据方差相对稳定
axes[2].plot(df.index, df['乘客数'], label='原始数据', linewidth=2)
axes[2].set_title('原始数据（无需变换）', fontsize=14, fontweight='bold')
axes[2].set_xlabel('日期', fontsize=12)
axes[2].set_ylabel('乘客数', fontsize=12)
axes[2].legend()
axes[2].grid(True, alpha=0.3)

plt.tight_layout()

# 加法分解：Y = T + S + R

def check_stationarity(timeseries, name=''):
    """检验时间序列的平稳性"""
    print(f'\n===== {name} 的平稳性检验 =====')
    result = adfuller(timeseries.dropna(), autolag='AIC')

    print(f'ADF统计量: {result[0]:.4f}')

        return True
    else:
        print('❌ 结论：序列非平稳 (p >= 0.05)，需要差分')

    axes[2].axhline(y=0, color='r', linestyle='--', alpha=0.5)
    axes[2].grid(True, alpha=0.3)

    df['混合差分'].plot(ax=axes[3], title='一阶+季节差分', color='purple', linewidth=2)

    axes[3].set_xlabel('日期')

print("""
ACF (Autocorrelation Function):
- 显示数据与其滞后值的相关性
- 拖尾：缓慢衰减 → 使用MA模型
- 截尾：突然截断 → 确定MA阶数q

PACF (Partial Autocorrelation Function):
- 显示消除中间影响后的相关性
- 拖尾：缓慢衰减 → 使用AR模型
- 截尾：突然截断 → 确定AR阶数p

常见模式：
- AR(p): PACF截尾在p阶，ACF拖尾
- MA(q): ACF截尾在q阶，PACF拖尾
- ARMA(p,q): ACF和PACF都拖尾
""")
```

## ARIMA模型详解

### 什么是ARIMA？

```python
# ARIMA = AutoRegressive Integrated Moving Average
# ARIMA(p, d, q)

ARIMA_components = {
    "AR (AutoRegressive)": {
        "含义": "自回归",
        "公式": "y_t = c + φ1*y_{t-1} + φ2*y_{t-2} + ... + φp*y_{t-p} + ε_t",
        "p": "AR阶数（使用多少个过去值）",
        "例子": "今天的温度受昨天和前天的温度影响"
    },

    "I (Integrated)": {
        "含义": "差分",
        "公式": "差分d次后变平稳",
        "d": "差分次数",
        "例子": "股票价格非平稳，但收益率（一阶差分）平稳"
    },

    "MA (Moving Average)": {
        "含义": "移动平均",
        "公式": "y_t = μ + ε_t + θ1*ε_{t-1} + θ2*ε_{t-2} + ... + θq*ε_{t-q}",
        "q": "MA阶数（使用多少个过去误差）",
        "例子": "今天的销售额受之前预测误差的影响"
    }
}
```

### 如何选择p, d, q？

```python
# 步骤1：确定d（差分次数）
如何确定d = """
1. 画时间序列图，看是否有趋势
2. 进行ADF检验
3. 如果非平稳，进行一阶差分
4. 重复2-3，直到平稳
5. d = 差分次数（通常d=0,1,2）
"""

# 步骤2：确定p和q
如何确定p和q = """
1. 画ACF图和PACF图
2. 根据模式判断：

模式识别：
- PACF在lag=p后截尾，ACF拖尾 → AR(p)
- ACF在lag=q后截尾，PACF拖尾 → MA(q)
- 都拖尾 → ARMA(p,q)

实践技巧：
- p和q通常不超过5
- 从简单模型开始：ARIMA(1,1,1)
- 用AIC/BIC选择最优模型
- 交叉验证
"""

# 实用建议
实用建议 = """
对于初学者：
1. 先尝试ARIMA(1,1,1) 或 ARIMA(0,1,1)
2. 比较多个模型的AIC
3. 选择AIC最小的模型
4. 检查残差是否为白噪声

常用组合：
- ARIMA(0,1,0): 随机游走
- ARIMA(0,1,1): 随机游走+噪声修正
- ARIMA(1,1,0): 差分+自回归
- ARIMA(1,1,1): 全能型
"""
```

### 残差诊断

```python
print("\n" + "=" * 60)
print("残差诊断")
print("=" * 60)

def diagnose_residuals(model, model_name=''):
    """
    诊断模型残差
    """
    print(f"\n{'='*50}")
    print(f"{model_name} 残差诊断")
    print(f"{'='*50}")

    residuals = model.resid

    # 1. 均值检验
    mean_resid = residuals.mean()
    print(f"\n1. 残差均值: {mean_resid:.6f}")
    if abs(mean_resid) < 0.01:
        print("   ✅ 残差均值接近0，无偏")
    else:
        print("   ⚠️ 残差均值偏离0，可能有偏")

    # 2. 正态性检验（Jarque-Bera）
    from scipy.stats import jarque_bera
    jb_stat, jb_pvalue = jarque_bera(residuals)
    print(f"\n2. Jarque-Bera检验:")
    print(f"   统计量: {jb_stat:.4f}")
    print(f"   p值: {jb_pvalue:.4f}")
    if jb_pvalue > 0.05:
        print("   ✅ 残差服从正态分布")
    else:
        print("   ⚠️ 残差不服从正态分布")

    # 3. 自相关检验（Ljung-Box）
    from statsmodels.stats.diagnostic import acorr_ljungbox
    lb_test = acorr_ljungbox(residuals, lags=[10], return_df=True)
    lb_pvalue = lb_test['lb_pvalue'].values[0]
    print(f"\n3. Ljung-Box检验 (lag=10):")
    print(f"   p值: {lb_pvalue:.4f}")
    if lb_pvalue > 0.05:
        print("   ✅ 残差无自相关（白噪声）")
    else:
        print("   ⚠️ 残差存在自相关")

    # 4. 异方差检验（ARCH效应）
    from statsmodels.stats.diagnostic import het_arch
    arch_test = het_arch(residuals, nlags=10)
    arch_pvalue = arch_test[1]
    print(f"\n4. ARCH检验 (异方差):")
    print(f"   p值: {arch_pvalue:.4f}")
    if arch_pvalue > 0.05:
        print("   ✅ 残差方差齐性")
    else:
        print("   ⚠️ 残差存在异方差（可能需要GARCH模型）")

    return {
        'mean': mean_resid,
        'jb_pvalue': jb_pvalue,
        'lb_pvalue': lb_pvalue,
        'arch_pvalue': arch_pvalue
    }
```

```python
# 简单的异常检测方法
def detect_anomalies(timeseries, window=20, sigma=3):
    """
    使用滚动均值和标准差检测异常
    """
    rolling_mean = timeseries.rolling(window=window).mean()
    rolling_std = timeseries.rolling(window=window).std()

    # 异常：超出均值±3倍标准差
    upper_bound = rolling_mean + sigma * rolling_std
    lower_bound = rolling_mean - sigma * rolling_std

    anomalies = (timeseries > upper_bound) | (timeseries < lower_bound)

    return anomalies, upper_bound, lower_bound

# 示例
print("\n示例：检测异常值")

# 生成带异常的数据
np.random.seed(42)
normal_data = np.random.randn(200) * 10 + 100
anomaly_indices = [50, 120, 180]
normal_data[anomaly_indices] = [150, 50, 160]  # 人为添加异常

ts_anomaly = pd.Series(normal_data)

# 检测异常
anomalies, upper, lower = detect_anomalies(ts_anomaly)

print(f"检测到 {anomalies.sum()} 个异常点")
print(f"异常点位置: {ts_anomaly[anomalies].index.tolist()}")

# 可视化
plt.figure(figsize=(14, 6))
plt.plot(ts_anomaly.index, ts_anomaly, label='数据', linewidth=2)
plt.plot(upper.index, upper, label='上界', color='red', linestyle='--', alpha=0.7)
plt.plot(lower.index, lower, label='下界', color='red', linestyle='--', alpha=0.7)
plt.scatter(ts_anomaly[anomalies].index, ts_anomaly[anomalies],
            color='red', s=100, label='异常点', zorder=5)
plt.fill_between(upper.index, lower, upper, alpha=0.2, color='gray')
plt.title('时间序列异常检测', fontsize=16, fontweight='bold')
```
