# 第08章：统计检验

统计检验是数据分析的"验证工具"！当你说"A比B好"、"这个药有效"、"两组有差异"时，统计检验帮你回答：这是真的差异，还是随机波动？

## 从生活中理解统计检验

### 场景1：新药是否有效？

```python
# 临床试验
对照组 = [血压下降: 5, 3, 7, 4, 6]  # mmHg
试验组 = [血压下降: 12, 15, 10, 13, 14]  # mmHg

# 问题：试验组效果更好，是真的吗？还是碰巧？
# 答案：用t检验！
# H0: 两组没有差异（新药无效）
# H1: 试验组效果更好（新药有效）
```

### 场景2：硬币是否公平？

```python
# 抛硬币100次
结果 = {"正面": 60, "反面": 40}

# 问题：60:40是正常波动，还是硬币有问题？
# 答案：用二项检验！
# H0: 硬币公平（p=0.5）
# H1: 硬币不公平（p≠0.5）
```

### 场景3：考试成绩与学习方法有关吗？

```python
# 学习方法 vs 成绩
数据 = {
    "传统方法": [70, 75, 68, 72, 69],
    "新方法A": [85, 88, 82, 87, 86],
    "新方法B": [78, 80, 76, 79, 77]
}

# 问题：三种方法的效果有差异吗？
# 答案：用方差分析（ANOVA）！
# H0: 三组均值相等
# H1: 至少有一组不同
```

### 什么是假设检验？

```python
假设检验框架 = {
    "定义": "用样本数据推断总体特征的方法",

    "基本逻辑": """
    1. 提出假设
       - H0（零假设/原假设）：保守的说法（如"没有差异"）
       - H1（备择假设）：你想证明的说法（如"有差异"）

    2. 收集数据
       - 样本数据

    3. 计算检验统计量
       - 如：t统计量、F统计量、卡方统计量

    4. 计算p值
       - p值 = 在H0为真的情况下，观察到当前结果（或更极端）的概率

    5. 做出决策
       - p < 0.05 → 拒绝H0，接受H1（有显著差异）
       - p ≥ 0.05 → 不拒绝H0（没有足够证据）
    """,

    "关键概念": {
        "显著性水平α": "犯第一类错误的概率（错误拒绝H0），通常取0.05",
        "p值": "在H0为真的情况下，得到观测结果的概率",
        "第一类错误": "H0为真但拒绝（假阳性）",
        "第二类错误": "H0为假但未拒绝（假阴性）",
        "检验效能": "正确拒绝错误H0的概率 = 1-β"
    }
}
```

## 案例1：单样本t检验

```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
from scipy import stats
import statsmodels.api as sm
from statsmodels.stats.api import DescrStatsW
from statsmodels.stats.weightstats import ztest, ttest_ind
from statsmodels.stats.power import TTestPower, FTestAnovaPower
import warnings
warnings.filterwarnings('ignore')

# 设置中文显示
plt.rcParams['font.sans-serif'] = ['SimHei']
plt.rcParams['axes.unicode_minus'] = False
plt.style.use('seaborn-v0_8-darkgrid')

print("=" * 60)
print("案例1：单样本t检验 - 咖啡杯容量检查")
print("=" * 60)

print("""
场景：
某咖啡店声称其大杯咖啡容量为500ml。
质检人员随机抽取30杯，测量实际容量。
问题：实际容量是否显著低于500ml？
""")

# ===== 1. 生成数据 =====
np.random.seed(42)
n_samples = 30

# 实际容量（假设真实均值为480ml，标准差为20ml）
actual_volume = np.random.normal(480, 20, n_samples)

# 数据整理
df_coffee = pd.DataFrame({
    '杯号': range(1, n_samples + 1),
    '容量ml': actual_volume
})

print("\n数据概览：")
print(df_coffee.head(10))

print(f"\n描述统计：")
print(f"样本均值: {actual_volume.mean():.2f} ml")
print(f"样本标准差: {actual_volume.std(ddof=1):.2f} ml")
print(f"样本中位数: {np.median(actual_volume):.2f} ml")
print(f"最小值: {actual_volume.min():.2f} ml")
print(f"最大值: {actual_volume.max():.2f} ml")

# ===== 2. 可视化 =====
fig, axes = plt.subplots(1, 3, figsize=(16, 5))

# 直方图
axes[0].hist(actual_volume, bins=15, color='steelblue', edgecolor='black', alpha=0.7)
axes[0].axvline(x=500, color='red', linestyle='--', linewidth=2, label='声称容量(500ml)')
axes[0].axvline(x=actual_volume.mean(), color='green', linestyle='--', linewidth=2,
                label=f'样本均值({actual_volume.mean():.1f}ml)')
axes[0].set_title('容量分布', fontsize=14, fontweight='bold')
axes[0].set_xlabel('容量 (ml)', fontsize=12)
axes[0].set_ylabel('频数', fontsize=12)
axes[0].legend()
axes[0].grid(True, alpha=0.3)

# 箱线图
axes[1].boxplot(actual_volume, vert=True)
axes[1].axhline(y=500, color='red', linestyle='--', linewidth=2, label='声称容量')
axes[1].set_ylabel('容量 (ml)', fontsize=12)
axes[1].set_title('箱线图', fontsize=14, fontweight='bold')
axes[1].legend()
axes[1].grid(True, alpha=0.3, axis='y')

# Q-Q图（检验正态性）
stats.probplot(actual_volume, dist="norm", plot=axes[2])
axes[2].set_title('Q-Q图（正态性检验）', fontsize=14, fontweight='bold')
axes[2].grid(True, alpha=0.3)

plt.tight_layout()
plt.savefig('08-统计检验-01-单样本数据.png', dpi=300, bbox_inches='tight')
plt.show()

# ===== 3. 正态性检验 =====
print("\n" + "=" * 60)
print("步骤1：正态性检验")
print("=" * 60)

print("""
t检验假设数据服从正态分布。
我们先检验这个假设。
""")

# Shapiro-Wilk检验
shapiro_stat, shapiro_p = stats.shapiro(actual_volume)
print(f"\nShapiro-Wilk检验：")
print(f"  统计量: {shapiro_stat:.4f}")
print(f"  p值: {shapiro_p:.4f}")
if shapiro_p > 0.05:
    print(f"  ✅ p > 0.05，数据符合正态分布")
else:
    print(f"  ⚠️ p < 0.05，数据可能不符合正态分布")

# Kolmogorov-Smirnov检验
ks_stat, ks_p = stats.kstest(actual_volume, 'norm',
                              args=(actual_volume.mean(), actual_volume.std(ddof=1)))
print(f"\nKolmogorov-Smirnov检验：")
print(f"  统计量: {ks_stat:.4f}")
print(f"  p值: {ks_p:.4f}")

# ===== 4. 单样本t检验 =====
print("\n" + "=" * 60)
print("步骤2：单样本t检验")
print("=" * 60)

# 假设
mu0 = 500  # 声称的容量

print(f"""
假设：
  H0: μ = {mu0} ml（容量符合声称）
  H1: μ < {mu0} ml（容量低于声称）
  显著性水平: α = 0.05
  检验类型: 单侧检验（左侧）
""")

# 方法1：使用scipy
t_stat, p_value_two = stats.ttest_1samp(actual_volume, mu0)
# 单侧p值
p_value_one = p_value_two / 2 if t_stat < 0 else 1 - p_value_two / 2

print(f"\n方法1：scipy.stats.ttest_1samp")
print(f"  t统计量: {t_stat:.4f}")
print(f"  p值（双侧）: {p_value_two:.4f}")
print(f"  p值（单侧，左侧）: {p_value_one:.4f}")

# 方法2：使用statsmodels
desc_stats = DescrStatsW(actual_volume)
t_stat_sm, p_value_sm, df = desc_stats.ttest_mean(mu0, alternative='smaller')

print(f"\n方法2：statsmodels.DescrStatsW")
print(f"  t统计量: {t_stat_sm:.4f}")
print(f"  p值（单侧）: {p_value_sm:.4f}")
print(f"  自由度: {df}")

# 手动计算（教学目的）
n = len(actual_volume)
sample_mean = actual_volume.mean()
sample_std = actual_volume.std(ddof=1)
se = sample_std / np.sqrt(n)
t_manual = (sample_mean - mu0) / se

print(f"\n方法3：手动计算")
print(f"  样本均值: {sample_mean:.4f}")
print(f"  样本标准差: {sample_std:.4f}")
print(f"  标准误: {se:.4f}")
print(f"  t = (x̄ - μ₀) / SE = ({sample_mean:.2f} - {mu0}) / {se:.4f} = {t_manual:.4f}")

# 临界值
alpha = 0.05
critical_t = stats.t.ppf(alpha, df=n-1)
print(f"  临界值（α={alpha}）: {critical_t:.4f}")

# ===== 5. 结论 =====
print("\n" + "=" * 60)
print("步骤3：结论")
print("=" * 60)

if p_value_one < alpha:
    print(f"✅ 结论：拒绝H0（p={p_value_one:.4f} < {alpha}）")
    print(f"   实际容量显著低于声称的{mu0}ml")
    print(f"   平均低了 {mu0 - sample_mean:.2f} ml")
else:
    print(f"⚠️ 结论：不拒绝H0（p={p_value_one:.4f} ≥ {alpha}）")
    print(f"   没有足够证据表明容量低于{mu0}ml")

# 置信区间
ci_lower, ci_upper = desc_stats.tconfint_mean(alpha=0.05)
print(f"\n95%置信区间：[{ci_lower:.2f}, {ci_upper:.2f}] ml")
print(f"解释：我们有95%的信心认为真实平均容量在此区间内")

# ===== 6. 效应量 =====
print("\n" + "=" * 60)
print("步骤4：效应量（Cohen's d）")
print("=" * 60)

cohens_d = (sample_mean - mu0) / sample_std
print(f"Cohen's d = {cohens_d:.4f}")

if abs(cohens_d) < 0.2:
    effect_size = "很小"
elif abs(cohens_d) < 0.5:
    effect_size = "小"
elif abs(cohens_d) < 0.8:
    effect_size = "中等"
else:
    effect_size = "大"

print(f"效应量: {effect_size}")
print("""
Cohen's d 解释：
  |d| < 0.2: 很小
  0.2 ≤ |d| < 0.5: 小
  0.5 ≤ |d| < 0.8: 中等
  |d| ≥ 0.8: 大
""")

# 可视化t分布和检验结果
plt.figure(figsize=(12, 6))

# 生成t分布
x = np.linspace(-4, 4, 1000)
y = stats.t.pdf(x, df=n-1)

plt.plot(x, y, 'b-', linewidth=2, label=f't分布 (df={n-1})')
plt.axvline(x=t_stat, color='red', linestyle='--', linewidth=2,
            label=f't统计量 = {t_stat:.2f}')
plt.axvline(x=critical_t, color='green', linestyle='--', linewidth=2,
            label=f'临界值 = {critical_t:.2f}')

# 填充拒绝域
x_reject = x[x < critical_t]
y_reject = stats.t.pdf(x_reject, df=n-1)
plt.fill_between(x_reject, y_reject, alpha=0.3, color='red', label='拒绝域')

plt.title('t检验可视化', fontsize=16, fontweight='bold')
plt.xlabel('t值', fontsize=12)
plt.ylabel('概率密度', fontsize=12)
plt.legend(fontsize=11)
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('08-统计检验-02-t检验可视化.png', dpi=300, bbox_inches='tight')
plt.show()

print("\n" + "=" * 60)
print("案例1 完成！")
print("=" * 60)
```

## 案例2：两独立样本t检验

```python
print("\n" + "=" * 60)
print("案例2：两独立样本t检验 - 教学方法比较")
print("=" * 60)

print("""
场景：
学校想知道新教学法是否比传统教学法更有效。
随机分配50名学生到两组，期末考试比较成绩。
""")

# ===== 1. 生成数据 =====
np.random.seed(123)

# 传统教学法：均值70，标准差10
group_traditional = np.random.normal(70, 10, 25)

# 新教学法：均值75，标准差12
group_new = np.random.normal(75, 12, 25)

df_teaching = pd.DataFrame({
    '组别': ['传统']*25 + ['新方法']*25,
    '成绩': np.concatenate([group_traditional, group_new])
})

print("\n数据概览：")
print(df_teaching.groupby('组别')['成绩'].describe())

# ===== 2. 可视化 =====
fig, axes = plt.subplots(1, 3, figsize=(16, 5))

# 箱线图
df_teaching.boxplot(column='成绩', by='组别', ax=axes[0])
axes[0].set_title('成绩分布（箱线图）', fontsize=14, fontweight='bold')
axes[0].set_xlabel('组别', fontsize=12)
axes[0].set_ylabel('成绩', fontsize=12)
plt.sca(axes[0])
plt.xticks([1, 2], ['传统', '新方法'])

# 小提琴图
parts = axes[1].violinplot([group_traditional, group_new],
                             positions=[1, 2], showmeans=True, showmedians=True)
axes[1].set_xticks([1, 2])
axes[1].set_xticklabels(['传统', '新方法'])
axes[1].set_ylabel('成绩', fontsize=12)
axes[1].set_title('成绩分布（小提琴图）', fontsize=14, fontweight='bold')
axes[1].grid(True, alpha=0.3, axis='y')

# 直方图叠加
axes[2].hist(group_traditional, bins=15, alpha=0.5, label='传统', color='blue')
axes[2].hist(group_new, bins=15, alpha=0.5, label='新方法', color='red')
axes[2].axvline(x=group_traditional.mean(), color='blue', linestyle='--',
                linewidth=2, label=f'传统均值: {group_traditional.mean():.1f}')
axes[2].axvline(x=group_new.mean(), color='red', linestyle='--',
                linewidth=2, label=f'新方法均值: {group_new.mean():.1f}')
axes[2].set_xlabel('成绩', fontsize=12)
axes[2].set_ylabel('频数', fontsize=12)
axes[2].set_title('成绩分布（直方图）', fontsize=14, fontweight='bold')
axes[2].legend()
axes[2].grid(True, alpha=0.3)

plt.tight_layout()
plt.savefig('08-统计检验-03-两样本数据.png', dpi=300, bbox_inches='tight')
plt.show()

# ===== 3. 方差齐性检验 =====
print("\n" + "=" * 60)
print("步骤1：方差齐性检验（Levene检验）")
print("=" * 60)

print("""
两样本t检验假设两组方差相等。
先检验这个假设。
""")

# Levene检验
levene_stat, levene_p = stats.levene(group_traditional, group_new)
print(f"\nLevene检验：")
print(f"  统计量: {levene_stat:.4f}")
print(f"  p值: {levene_p:.4f}")

if levene_p > 0.05:
    print(f"  ✅ p > 0.05，方差齐性，可以使用标准t检验")
    equal_var = True
else:
    print(f"  ⚠️ p < 0.05，方差不齐，应使用Welch's t检验")
    equal_var = False

# ===== 4. 两独立样本t检验 =====
print("\n" + "=" * 60)
print("步骤2：两独立样本t检验")
print("=" * 60)

print(f"""
假设：
  H0: μ₁ = μ₂（两组成绩无差异）
  H1: μ₁ ≠ μ₂（两组成绩有差异）
  显著性水平: α = 0.05
  检验类型: 双侧检验
""")

# scipy
t_stat_2samp, p_value_2samp = stats.ttest_ind(group_traditional, group_new,
                                                equal_var=equal_var)

print(f"\n方法1：scipy.stats.ttest_ind")
print(f"  t统计量: {t_stat_2samp:.4f}")
print(f"  p值: {p_value_2samp:.4f}")
print(f"  方差齐性假设: {equal_var}")

# statsmodels
cm = sm.stats.CompareMeans(DescrStatsW(group_traditional),
                            DescrStatsW(group_new))
print(f"\n方法2：statsmodels.CompareMeans")
print(cm.summary(use_t=True, usevar='pooled' if equal_var else 'unequal'))

# ===== 5. 结论 =====
print("\n" + "=" * 60)
print("步骤3：结论")
print("=" * 60)

mean_diff = group_new.mean() - group_traditional.mean()

if p_value_2samp < 0.05:
    print(f"✅ 结论：拒绝H0（p={p_value_2samp:.4f} < 0.05）")
    print(f"   两组成绩有显著差异")
    print(f"   新方法平均高出 {mean_diff:.2f} 分")
else:
    print(f"⚠️ 结论：不拒绝H0（p={p_value_2samp:.4f} ≥ 0.05）")
    print(f"   没有足够证据表明两组有差异")

# 置信区间
ci = cm.tconfint_diff(alpha=0.05, usevar='pooled' if equal_var else 'unequal')
print(f"\n均值差的95%置信区间：[{ci[0]:.2f}, {ci[1]:.2f}]")

# 效应量
pooled_std = np.sqrt(((len(group_traditional)-1)*group_traditional.std(ddof=1)**2 +
                      (len(group_new)-1)*group_new.std(ddof=1)**2) /
                     (len(group_traditional) + len(group_new) - 2))
cohens_d_2samp = mean_diff / pooled_std

print(f"\nCohen's d = {cohens_d_2samp:.4f}")

# ===== 6. 功效分析 =====
print("\n" + "=" * 60)
print("步骤4：功效分析（Power Analysis）")
print("=" * 60)

power_analysis = TTestPower()
power = power_analysis.power(effect_size=cohens_d_2samp,
                              nobs1=len(group_traditional),
                              alpha=0.05,
                              alternative='two-sided')

print(f"检验功效: {power:.4f}")
print(f"解释：有{power*100:.1f}%的概率能检测到真实差异（如果存在）")

# 样本量计算：需要多少样本才能达到80%功效？
required_n = power_analysis.solve_power(effect_size=cohens_d_2samp,
                                         alpha=0.05,
                                         power=0.8,
                                         alternative='two-sided')
print(f"\n如果想达到80%功效，每组需要 {int(np.ceil(required_n))} 个样本")

print("\n" + "=" * 60)
print("案例2 完成！")
print("=" * 60)
```

## 案例3：配对样本t检验

```python
print("\n" + "=" * 60)
print("案例3：配对样本t检验 - 减肥方案效果")
print("=" * 60)

print("""
场景：
20个人参加减肥计划，记录前后体重。
问题：减肥方案是否有效？
""")

# ===== 1. 生成数据 =====
np.random.seed(456)
n_persons = 20

# 减肥前体重
weight_before = np.random.normal(80, 15, n_persons)

# 减肥后体重（假设平均减重5kg，个体差异大）
weight_loss = np.random.normal(5, 3, n_persons)
weight_after = weight_before - weight_loss

df_weight = pd.DataFrame({
    '编号': range(1, n_persons + 1),
    '减肥前': weight_before,
    '减肥后': weight_after,
    '减重': weight_before - weight_after
})

print("\n数据概览：")
print(df_weight.head(10))

print(f"\n描述统计：")
print(f"减肥前平均体重: {weight_before.mean():.2f} kg")
print(f"减肥后平均体重: {weight_after.mean():.2f} kg")
print(f"平均减重: {df_weight['减重'].mean():.2f} kg")
print(f"减重标准差: {df_weight['减重'].std(ddof=1):.2f} kg")

# ===== 2. 可视化 =====
fig, axes = plt.subplots(1, 3, figsize=(16, 5))

# 前后对比
x_pos = np.arange(len(df_weight))
axes[0].plot(x_pos, df_weight['减肥前'], 'o-', label='减肥前', markersize=8)
axes[0].plot(x_pos, df_weight['减肥后'], 's-', label='减肥后', markersize=8)
axes[0].set_xlabel('个体编号', fontsize=12)
axes[0].set_ylabel('体重 (kg)', fontsize=12)
axes[0].set_title('个体体重变化', fontsize=14, fontweight='bold')
axes[0].legend()
axes[0].grid(True, alpha=0.3)

# 减重量分布
axes[1].hist(df_weight['减重'], bins=15, color='coral', edgecolor='black', alpha=0.7)
axes[1].axvline(x=0, color='red', linestyle='--', linewidth=2, label='无变化')
axes[1].axvline(x=df_weight['减重'].mean(), color='green', linestyle='--', linewidth=2,
                label=f'平均减重: {df_weight["减重"].mean():.1f}kg')
axes[1].set_xlabel('减重量 (kg)', fontsize=12)

print("\n" + "=" * 60)
print("步骤1：配对样本t检验")
print("=" * 60)

假设：
  H0: μd = 0（减肥无效，前后无差异）
  H1: μd > 0（减肥有效，体重下降）
  显著性水平: α = 0.05
  检验类型: 单侧检验（右侧）

t_stat_diff, p_value_diff = stats.ttest_1samp(df_weight['减重'], 0)

print("步骤2：结论")
print("=" * 60)

    print(f"   平均减重 {df_weight['减重'].mean():.2f} kg")

ci_lower_diff = df_weight['减重'].mean() - stats.t.ppf(0.975, n_persons-1) * df_weight['减重'].sem()
```

## 案例4：方差分析（ANOVA）

```python
print("\n" + "=" * 60)
print("案例4：单因素方差分析（One-Way ANOVA）")
print("=" * 60)

print("""
场景：
比较3种不同肥料对作物产量的影响。
每种肥料用于10块田地。
问题：3种肥料的效果有差异吗？
""")

# ===== 1. 生成数据 =====
np.random.seed(789)
n_per_group = 10

# 三种肥料的产量（吨/亩）
fertilizer_A = np.random.normal(5.0, 0.8, n_per_group)
fertilizer_B = np.random.normal(5.8, 0.9, n_per_group)
fertilizer_C = np.random.normal(5.3, 0.7, n_per_group)

df_fertilizer = pd.DataFrame({
    '肥料类型': ['A']*n_per_group + ['B']*n_per_group + ['C']*n_per_group,
    '产量': np.concatenate([fertilizer_A, fertilizer_B, fertilizer_C])
})

print("\n数据概览：")
print(df_fertilizer.groupby('肥料类型')['产量'].describe())

# ===== 2. 可视化 =====
fig, axes = plt.subplots(1, 3, figsize=(16, 5))

df_fertilizer.boxplot(column='产量', by='肥料类型', ax=axes[0])

axes[0].set_xlabel('肥料类型', fontsize=12)
axes[0].set_ylabel('产量 (吨/亩)', fontsize=12)

# 小提琴图
parts = axes[1].violinplot([fertilizer_A, fertilizer_B, fertilizer_C],

axes[1].set_xticks([1, 2, 3])
axes[1].set_xticklabels(['A', 'B', 'C'])
axes[1].set_xlabel('肥料类型', fontsize=12)

axes[1].set_title('产量分布（小提琴图）', fontsize=14, fontweight='bold')

# 均值条形图
means = df_fertilizer.groupby('肥料类型')['产量'].mean()

axes[2].bar(means.index, means.values, yerr=stds.values, capsize=10,
            color=['steelblue', 'coral', 'lightgreen'], alpha=0.7)
axes[2].set_xlabel('肥料类型', fontsize=12)
axes[2].set_ylabel('平均产量 (吨/亩)', fontsize=12)
axes[2].set_title('平均产量对比', fontsize=14, fontweight='bold')
axes[2].grid(True, alpha=0.3, axis='y')

plt.tight_layout()
plt.savefig('08-统计检验-05-ANOVA数据.png', dpi=300, bbox_inches='tight')
plt.show()

# ===== 3. 方差齐性检验 =====
print("\n" + "=" * 60)
print("步骤1：方差齐性检验（Levene检验）")
print("=" * 60)

levene_stat_anova, levene_p_anova = stats.levene(fertilizer_A, fertilizer_B, fertilizer_C)
print(f"\nLevene检验：")
print(f"  统计量: {levene_stat_anova:.4f}")
print(f"  p值: {levene_p_anova:.4f}")

if levene_p_anova > 0.05:
    print(f"  ✅ p > 0.05，方差齐性，可以使用ANOVA")
else:
    print(f"  ⚠️ p < 0.05，方差不齐，考虑使用Welch's ANOVA")

# ===== 4. 单因素方差分析 =====
print("\n" + "=" * 60)
print("步骤2：单因素方差分析（One-Way ANOVA）")
print("=" * 60)

print(f"""
假设：
  H0: μA = μB = μC（三种肥料效果无差异）
  H1: 至少有一对均值不相等
  显著性水平: α = 0.05
""")

# 方法1：scipy
f_stat, p_value_anova = stats.f_oneway(fertilizer_A, fertilizer_B, fertilizer_C)

print(f"\n方法1：scipy.stats.f_oneway")
print(f"  F统计量: {f_stat:.4f}")
print(f"  p值: {p_value_anova:.4f}")

# 方法2：statsmodels（使用OLS）
import statsmodels.formula.api as smf

anova_model = smf.ols('产量 ~ C(肥料类型)', data=df_fertilizer).fit()
anova_table = sm.stats.anova_lm(anova_model, typ=2)

print(f"\n方法2：statsmodels (Type II ANOVA)")
print(anova_table)

# ===== 5. 结论 =====
print("\n" + "=" * 60)
print("步骤3：结论")
print("=" * 60)

if p_value_anova < 0.05:
    print(f"✅ 结论：拒绝H0（p={p_value_anova:.4f} < 0.05）")
    print(f"   至少有一种肥料的效果显著不同")
    print(f"   需要进行事后检验（Post-hoc test）确定具体哪些组不同")
else:
    print(f"⚠️ 结论：不拒绝H0（p={p_value_anova:.4f} ≥ 0.05）")
    print(f"   没有足够证据表明三种肥料效果有差异")

# ===== 6. 事后检验（Post-hoc）=====
if p_value_anova < 0.05:
    print("\n" + "=" * 60)
    print("步骤4：事后检验（Tukey HSD）")
    print("=" * 60)

    from statsmodels.stats.multicomp import pairwise_tukeyhsd

    tukey_result = pairwise_tukeyhsd(df_fertilizer['产量'],
                                      df_fertilizer['肥料类型'],
                                      alpha=0.05)

    print(tukey_result)

    print("""
    解释：
    - reject=True：该对组间有显著差异
    - reject=False：该对组间无显著差异
    """)

# ===== 7. 效应量 =====
print("\n" + "=" * 60)
print("步骤5：效应量（Eta-squared）")
print("=" * 60)

# 计算eta-squared
ss_between = anova_table.loc['C(肥料类型)', 'sum_sq']
ss_total = anova_table['sum_sq'].sum()
eta_squared = ss_between / ss_total

print(f"Eta² = {eta_squared:.4f}")
print(f"解释：肥料类型解释了产量变异的 {eta_squared*100:.1f}%")

print("""
Eta² 解释：
  0.01: 小效应
  0.06: 中等效应
  0.14: 大效应
""")

print("\n" + "=" * 60)
print("案例4 完成！")
print("=" * 60)
```

```python
print("\n" + "=" * 60)
print("案例5：卡方检验 - 性别与产品偏好")

print("""

print("""
```

## 检验方法选择指南

```python
print("\n" + "=" * 60)
print("统计检验方法选择指南")

选择指南 = """
【第1步：确定研究问题】

1. 比较均值/中位数？
   ├─ 一个样本 vs 固定值
   │  ├─ 正态性 + 连续数据 → 单样本t检验

   │  └─ 非正态 / 等级数据 → Mann-Whitney U检验
   │
   ├─ 两个配对样本
   │  ├─ 正态性 → 配对t检验
   │  └─ 非正态 / 等级数据 → Wilcoxon符号秩检验
   │
   └─ 三个或更多组
      ├─ 正态性 + 方差齐性 → 单因素ANOVA
      ├─ 正态性 + 方差不齐 → Welch's ANOVA
      └─ 非正态 / 等级数据 → Kruskal-Wallis检验

2. 检验关联性？
   ├─ 两个分类变量 → 卡方独立性检验
   ├─ 两个连续变量
   │  ├─ 线性关系 + 正态性 → Pearson相关
   │  └─ 单调关系 / 非正态 → Spearman等级相关
   └─ 一个分类 + 一个连续 → t检验 或 ANOVA

3. 检验分布？
   ├─ 正态性检验 → Shapiro-Wilk, K-S检验
   ├─ 拟合优度检验 → 卡方拟合优度检验
   └─ 两分布是否相同 → Kolmogorov-Smirnov检验

4. 检验比例？
   ├─ 一个比例 vs 固定值 → 二项检验
   ├─ 两个比例 → 两比例z检验
   └─ 多个比例 → 卡方检验

【第2步：检查假设】

参数检验假设：
✅ 正态性（Shapiro-Wilk, Q-Q图）
✅ 方差齐性（Levene检验）
✅ 独立性
✅ 足够样本量（通常n≥30，t检验可以更小）

如果假设不满足：
⚠️ 数据变换（log, sqrt, Box-Cox）
⚠️ 使用非参数检验
⚠️ Bootstrap方法
⚠️ 增加样本量

【第3步：多重比较校正】

当进行多次检验时（如事后检验）：
- Bonferroni校正：α_new = α / k
- Holm-Bonferroni：逐步校正
- FDR（False Discovery Rate）：Benjamini-Hochberg
- Tukey HSD：专门用于ANOVA事后检验
"""

print(选择指南)
```

**本章重点**
- ✅ 假设检验：提出假设 → 收集数据 → 计算p值 → 做出决策
- ✅ p值是在H0为真下观察到数据的概率，不是H0为真的概率
- ✅ 统计显著性（p<0.05）≠ 实际重要性
- ✅ 总是报告效应量和置信区间
- ✅ 检验假设：正态性、方差齐性、独立性
- ✅ 假设不满足时使用非参数检验
- ✅ 避免p-hacking和HARKing

**记住**
- 永远先画图，再检验
- p值不是万能的，要结合效应量和置信区间
- 不显著 ≠ 没有差异（可能是样本量不够）
- 多重比较要校正
- 预先注册研究方案，避免事后诸葛

**Statsmodels统计检验核心API**：
```python
# 导入
from scipy import stats
import statsmodels.api as sm
from statsmodels.stats.api import DescrStatsW

# t检验
stats.ttest_1samp(data, popmean)  # 单样本
stats.ttest_ind(data1, data2)     # 两独立样本
stats.ttest_rel(data1, data2)     # 配对样本

# ANOVA
stats.f_oneway(group1, group2, group3)
sm.stats.anova_lm(model, typ=2)

# 非参数检验
stats.mannwhitneyu(data1, data2)  # Mann-Whitney
stats.wilcoxon(before, after)     # Wilcoxon
stats.kruskal(g1, g2, g3)         # Kruskal-Wallis

# 卡方检验
stats.chi2_contingency(table)     # 独立性检验

# 正态性检验
stats.shapiro(data)               # Shapiro-Wilk
stats.normaltest(data)            # D'Agostino-Pearson

# 方差齐性检验
stats.levene(g1, g2, g3)          # Levene
```
