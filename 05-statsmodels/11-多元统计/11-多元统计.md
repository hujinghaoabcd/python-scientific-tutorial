# 第11章：多元统计

多元统计是处理"多个变量"的艺术！当数据有很多维度时，多元统计方法帮我们降维、分组、发现模式。从市场细分到图像压缩，从因子分析到聚类，多元统计无处不在。

## 从生活中理解多元统计

### 场景1：学生综合评价

```python
# 学生有多门成绩
学生成绩 = {
    "张三": {"语文": 85, "数学": 90, "英语": 88, "物理": 92, "化学": 87},
    "李四": {"语文": 78, "数学": 95, "英语": 80, "物理": 93, "化学": 91},
    ...
}

# 问题：
# 1. 5门成绩能否归纳为"文科能力"和"理科能力"？ → 因子分析
# 2. 能否用2-3个综合指标代替5门成绩？ → 主成分分析
# 3. 如何将学生分成几类？ → 聚类分析
```

### 场景2：鸢尾花分类

```python
# 鸢尾花有4个测量值
鸢尾花特征 = {
    "花1": {"花萼长": 5.1, "花萼宽": 3.5, "花瓣长": 1.4, "花瓣宽": 0.2},
    ...
}

# 问题：
# 1. 哪些特征最能区分种类？ → 判别分析
# 2. 4个特征能否降到2维可视化？ → PCA
```

## 案例1：主成分分析（PCA）

```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.preprocessing import StandardScaler
from sklearn.decomposition import PCA, FactorAnalysis
from sklearn.cluster import KMeans, AgglomerativeClustering
from sklearn.discriminant_analysis import LinearDiscriminantAnalysis
from scipy.cluster.hierarchy import dendrogram, linkage
from scipy.spatial.distance import pdist, squareform
import statsmodels.api as sm
from statsmodels.multivariate.manova import MANOVA
import warnings
warnings.filterwarnings('ignore')

# 设置中文显示
plt.rcParams['font.sans-serif'] = ['SimHei']
plt.rcParams['axes.unicode_minus'] = False
plt.style.use('seaborn-v0_8-darkgrid')

print("=" * 60)
print("案例1：主成分分析（PCA）- 学生成绩分析")
print("=" * 60)

print("""
场景：
100名学生的6门课程成绩：语文、数学、英语、物理、化学、生物。
问题：
1. 能否用更少的"综合指标"代替这6门成绩？
2. 各门课之间有什么内在结构？
3. 如何在2维平面上展示学生分布？
""")

# ===== 1. 生成模拟数据 =====
np.random.seed(42)
n_students = 100

# 潜在因子：文科能力和理科能力
verbal_ability = np.random.normal(70, 15, n_students)
science_ability = np.random.normal(70, 15, n_students)

# 生成成绩（基于潜在因子 + 噪声）
df_scores = pd.DataFrame({
    '学号': [f'S{i+1:03d}' for i in range(n_students)],
    '语文': verbal_ability + np.random.normal(0, 5, n_students),
    '英语': verbal_ability * 0.9 + np.random.normal(0, 6, n_students),
    '数学': science_ability + np.random.normal(0, 5, n_students),
    '物理': science_ability * 0.95 + np.random.normal(0, 6, n_students),
    '化学': science_ability * 0.9 + np.random.normal(0, 7, n_students),
    '生物': (verbal_ability * 0.3 + science_ability * 0.7) + np.random.normal(0, 6, n_students)
})

print("\n数据概览：")
print(df_scores.head(10))

print("\n描述统计：")
print(df_scores.iloc[:, 1:].describe())

# ===== 2. 相关矩阵 =====
print("\n" + "=" * 60)
print("步骤1：探索变量间相关性")
print("=" * 60)

# 计算相关矩阵
corr_matrix = df_scores.iloc[:, 1:].corr()

print("\n相关矩阵：")
print(corr_matrix.round(3))

# 可视化相关矩阵
plt.figure(figsize=(10, 8))
sns.heatmap(corr_matrix, annot=True, fmt='.2f', cmap='coolwarm',
            center=0, square=True, linewidths=1, cbar_kws={"shrink": 0.8})
plt.title('课程成绩相关矩阵', fontsize=14, fontweight='bold')
plt.tight_layout()
plt.savefig('11-多元统计-01-相关矩阵.png', dpi=300, bbox_inches='tight')
plt.show()

print("""
观察：
语文和英语高度相关（文科）
数学、物理、化学高度相关（理科）
生物与两类都有关联
💡 这提示存在潜在的"文理"因子结构
""")

# ===== 3. 数据标准化 =====
print("\n" + "=" * 60)
print("步骤2：数据标准化")
print("=" * 60)

print("""
为什么要标准化？
- PCA对量纲敏感
- 标准化后各变量贡献相等
- z分数：(x - mean) / std
""")

# 提取特征
X = df_scores.iloc[:, 1:].values
feature_names = df_scores.columns[1:].tolist()

# 标准化
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

print(f"原始数据形状: {X.shape}")
print(f"标准化后均值: {X_scaled.mean(axis=0).round(4)}")
print(f"标准化后标准差: {X_scaled.std(axis=0).round(4)}")

# ===== 4. PCA拟合 =====
print("\n" + "=" * 60)
print("步骤3：主成分分析")
print("=" * 60)

# 拟合PCA
pca = PCA()
X_pca = pca.fit_transform(X_scaled)

# 解释方差比例
explained_variance_ratio = pca.explained_variance_ratio_
cumulative_variance = np.cumsum(explained_variance_ratio)

print(f"\n各主成分的解释方差比例：")
for i, (var, cum_var) in enumerate(zip(explained_variance_ratio, cumulative_variance)):
    print(f"  PC{i+1}: {var:.4f} ({var*100:.2f}%) - 累积: {cum_var:.4f} ({cum_var*100:.2f}%)")

# 碎石图
fig, axes = plt.subplots(1, 2, figsize=(14, 5))

# 解释方差
axes[0].bar(range(1, len(explained_variance_ratio)+1), explained_variance_ratio,
            alpha=0.7, color='steelblue', edgecolor='black')
axes[0].plot(range(1, len(explained_variance_ratio)+1), explained_variance_ratio,
             'ro-', linewidth=2, markersize=8)
axes[0].set_xlabel('主成分', fontsize=12)
axes[0].set_ylabel('解释方差比例', fontsize=12)
axes[0].set_title('碎石图', fontsize=14, fontweight='bold')
axes[0].set_xticks(range(1, len(explained_variance_ratio)+1))
axes[0].grid(True, alpha=0.3, axis='y')

# 累积解释方差
axes[1].plot(range(1, len(cumulative_variance)+1), cumulative_variance,
             'go-', linewidth=2, markersize=8)
axes[1].axhline(y=0.8, color='red', linestyle='--', linewidth=2, label='80%阈值')
axes[1].axhline(y=0.9, color='orange', linestyle='--', linewidth=2, label='90%阈值')
axes[1].set_xlabel('主成分数量', fontsize=12)
axes[1].set_ylabel('累积解释方差比例', fontsize=12)
axes[1].set_title('累积解释方差', fontsize=14, fontweight='bold')
axes[1].set_xticks(range(1, len(cumulative_variance)+1))
axes[1].set_ylim([0, 1.05])
axes[1].legend()
axes[1].grid(True, alpha=0.3)

plt.tight_layout()
plt.savefig('11-多元统计-02-PCA碎石图.png', dpi=300, bbox_inches='tight')
plt.show()

# ===== 5. 主成分载荷 =====
print("\n" + "=" * 60)
print("步骤4：主成分载荷（Component Loadings）")
print("=" * 60)

# 载荷矩阵 = 特征向量 × sqrt(特征值)
loadings = pca.components_.T * np.sqrt(pca.explained_variance_)

loadings_df = pd.DataFrame(
    loadings[:, :3],
    columns=['PC1', 'PC2', 'PC3'],
    index=feature_names
)

print("\n前3个主成分的载荷：")
print(loadings_df.round(3))

print("""
载荷解释：
- 载荷大小：该变量对主成分的贡献
- 载荷符号：正/负关系
""")

# 可视化载荷
fig, axes = plt.subplots(1, 2, figsize=(14, 6))

# PC1 vs PC2载荷图
for i, feature in enumerate(feature_names):
    axes[0].arrow(0, 0, loadings[i, 0], loadings[i, 1],
                  head_width=0.05, head_length=0.05, fc='blue', ec='blue')
    axes[0].text(loadings[i, 0]*1.15, loadings[i, 1]*1.15, feature,
                 fontsize=11, ha='center')

axes[0].set_xlabel(f'PC1 ({explained_variance_ratio[0]*100:.1f}%)', fontsize=12)
axes[0].set_ylabel(f'PC2 ({explained_variance_ratio[1]*100:.1f}%)', fontsize=12)
axes[0].set_title('主成分载荷图', fontsize=14, fontweight='bold')
axes[0].axhline(y=0, color='k', linewidth=0.5)
axes[0].axvline(x=0, color='k', linewidth=0.5)
axes[0].grid(True, alpha=0.3)
axes[0].set_xlim([-1.2, 1.2])
axes[0].set_ylim([-1.2, 1.2])

# 热图
sns.heatmap(loadings_df, annot=True, fmt='.2f', cmap='RdBu_r',
            center=0, ax=axes[1], cbar_kws={"shrink": 0.8})
axes[1].set_title('主成分载荷热图', fontsize=14, fontweight='bold')

plt.tight_layout()
plt.savefig('11-多元统计-03-PCA载荷.png', dpi=300, bbox_inches='tight')
plt.show()

print("""
载荷解释：
PC1（第1主成分）：
   - 理科课程（数学、物理、化学）载荷高
   - 代表"理科能力"

PC2（第2主成分）：
   - 文科课程（语文、英语）载荷高
   - 代表"文科能力"

生物：两个成分都有载荷（文理结合）
""")

# ===== 6. 主成分得分 =====
print("\n" + "=" * 60)
print("步骤5：主成分得分（Score Plot）")
print("=" * 60)

# 前两个主成分的得分
pc1_scores = X_pca[:, 0]
pc2_scores = X_pca[:, 1]

print(f"\n前10个学生的PC得分：")
score_df = pd.DataFrame({
    '学号': df_scores['学号'][:10],
    'PC1': pc1_scores[:10],
    'PC2': pc2_scores[:10]
})
print(score_df.round(2))

# 可视化得分
plt.figure(figsize=(10, 8))
plt.scatter(pc1_scores, pc2_scores, alpha=0.6, s=50, c='steelblue', edgecolor='black')
plt.xlabel(f'PC1 - 理科能力 ({explained_variance_ratio[0]*100:.1f}%)', fontsize=12)
plt.ylabel(f'PC2 - 文科能力 ({explained_variance_ratio[1]*100:.1f}%)', fontsize=12)
plt.title('学生在主成分空间的分布', fontsize=14, fontweight='bold')
plt.axhline(y=0, color='k', linewidth=0.5, linestyle='--', alpha=0.5)
plt.axvline(x=0, color='k', linewidth=0.5, linestyle='--', alpha=0.5)
plt.grid(True, alpha=0.3)

# 标注几个极端学生
for i in [pc1_scores.argmax(), pc1_scores.argmin(),
          pc2_scores.argmax(), pc2_scores.argmin()]:
    plt.annotate(df_scores.iloc[i, 0],
                 (pc1_scores[i], pc2_scores[i]),
                 fontsize=9, alpha=0.7)

plt.tight_layout()
plt.savefig('11-多元统计-04-PCA得分图.png', dpi=300, bbox_inches='tight')
plt.show()

print("\n" + "=" * 60)
print("案例1 完成！")
print("=" * 60)

print("""
PCA总结：
6个变量降到2个主成分，保留了约85%的信息
PC1代表理科能力，PC2代表文科能力
可以用PC1和PC2作为综合评价指标
降维后可以更容易可视化和理解
""")
```

## 案例2：聚类分析

```python
print("\n" + "=" * 60)
print("案例2：聚类分析 - 客户细分")
print("=" * 60)

print("""
场景：
超市有200个客户的消费数据：
- 年消费额
- 购物频次
- 平均客单价
- 会员年限

问题：如何将客户分成几类，进行差异化营销？
""")

# ===== 1. 生成数据 =====
np.random.seed(789)
n_customers = 200

# 生成3类客户
# 类别1：高价值客户（高消费、高频次）
cluster1 = np.random.multivariate_normal(
    [50000, 100, 500, 5],
    [[1000000, 500, 2000, 2],
     [500, 100, 50, 0.5],
     [2000, 50, 5000, 1],
     [2, 0.5, 1, 1]],
    60
)

# 类别2：中等客户
cluster2 = np.random.multivariate_normal(
    [20000, 50, 400, 3],
    [[500000, 200, 1000, 1],
     [200, 50, 20, 0.3],
     [1000, 20, 2000, 0.5],
     [1, 0.3, 0.5, 0.5]],
    80
)

# 类别3：低价值客户（低消费、低频次）
cluster3 = np.random.multivariate_normal(
    [5000, 20, 250, 1],
    [[100000, 50, 500, 0.5],
     [50, 20, 10, 0.2],
     [500, 10, 1000, 0.3],
     [0.5, 0.2, 0.3, 0.3]],
    60
)

# 合并数据
X_cluster = np.vstack([cluster1, cluster2, cluster3])
true_labels = np.array([1]*60 + [2]*80 + [3]*60)

df_customers = pd.DataFrame(
    X_cluster,
    columns=['年消费额', '购物频次', '平均客单价', '会员年限']
)
df_customers['客户ID'] = [f'C{i+1:03d}' for i in range(n_customers)]

print("\n数据概览：")
print(df_customers.head(10))

print("\n描述统计：")
print(df_customers.iloc[:, :-1].describe())

# ===== 2. K-means聚类 =====
print("\n" + "=" * 60)
print("步骤1：K-means聚类")
print("=" * 60)

# 标准化
X_cluster_scaled = StandardScaler().fit_transform(df_customers.iloc[:, :-1])

# 确定最佳聚类数（肘部法则）
inertias = []
silhouette_scores = []
K_range = range(2, 11)

from sklearn.metrics import silhouette_score

for k in K_range:
    kmeans = KMeans(n_clusters=k, random_state=42, n_init=10)
    kmeans.fit(X_cluster_scaled)
    inertias.append(kmeans.inertia_)
    silhouette_scores.append(silhouette_score(X_cluster_scaled, kmeans.labels_))

# 可视化
fig, axes = plt.subplots(1, 2, figsize=(14, 5))

# 肘部法则
axes[0].plot(K_range, inertias, 'bo-', linewidth=2, markersize=8)
axes[0].set_xlabel('聚类数 K', fontsize=12)
axes[0].set_ylabel('簇内平方和（Inertia）', fontsize=12)
axes[0].set_title('肘部法则', fontsize=14, fontweight='bold')
axes[0].grid(True, alpha=0.3)

# 轮廓系数
axes[1].plot(K_range, silhouette_scores, 'ro-', linewidth=2, markersize=8)
axes[1].set_xlabel('聚类数 K', fontsize=12)
axes[1].set_ylabel('轮廓系数', fontsize=12)
axes[1].set_title('轮廓系数', fontsize=14, fontweight='bold')
axes[1].grid(True, alpha=0.3)

plt.tight_layout()
plt.savefig('11-多元统计-05-聚类数选择.png', dpi=300, bbox_inches='tight')
plt.show()

print(f"\n推荐聚类数: K=3（肘部+轮廓系数）")

# 使用K=3聚类
kmeans = KMeans(n_clusters=3, random_state=42, n_init=10)
cluster_labels = kmeans.fit_predict(X_cluster_scaled)

df_customers['聚类'] = cluster_labels

# ===== 3. 聚类结果分析 =====
print("\n" + "=" * 60)
print("步骤2：聚类结果分析")
print("=" * 60)

print("\n各聚类的统计特征：")
cluster_summary = df_customers.groupby('聚类').agg({
    '年消费额': ['count', 'mean', 'std'],
    '购物频次': 'mean',
    '平均客单价': 'mean',
    '会员年限': 'mean'
})
print(cluster_summary.round(1))

# 聚类命名
cluster_names = {
    0: '未命名',
    1: '未命名',
    2: '未命名'
}

# 根据特征命名
for i in range(3):
    cluster_data = df_customers[df_customers['聚类'] == i]
    avg_spending = cluster_data['年消费额'].mean()

    if avg_spending > 40000:
        cluster_names[i] = 'VIP客户'
    elif avg_spending > 15000:
        cluster_names[i] = '普通客户'
    else:
        cluster_names[i] = '潜力客户'

print("\n聚类解释：")
for i, name in cluster_names.items():
    count = (cluster_labels == i).sum()
    print(f"  聚类{i} - {name}: {count}人 ({count/len(df_customers)*100:.1f}%)")

# ===== 4. 可视化聚类结果 =====
# 使用PCA降维到2D
pca_cluster = PCA(n_clusters=2)
X_pca_cluster = pca_cluster.fit_transform(X_cluster_scaled)

fig, axes = plt.subplots(1, 2, figsize=(14, 6))

# 真实标签（如果知道）
scatter1 = axes[0].scatter(X_pca_cluster[:, 0], X_pca_cluster[:, 1],
                            c=true_labels, cmap='viridis', alpha=0.6,
                            s=50, edgecolor='black')
axes[0].set_xlabel(f'PC1 ({pca_cluster.explained_variance_ratio_[0]*100:.1f}%)', fontsize=12)
axes[0].set_ylabel(f'PC2 ({pca_cluster.explained_variance_ratio_[1]*100:.1f}%)', fontsize=12)
axes[0].set_title('真实分类（已知）', fontsize=14, fontweight='bold')
axes[0].grid(True, alpha=0.3)
plt.colorbar(scatter1, ax=axes[0])

# K-means结果
scatter2 = axes[1].scatter(X_pca_cluster[:, 0], X_pca_cluster[:, 1],
                            c=cluster_labels, cmap='viridis', alpha=0.6,
                            s=50, edgecolor='black')

# 聚类中心
centers_scaled = kmeans.cluster_centers_
centers_pca = pca_cluster.transform(centers_scaled)
axes[1].scatter(centers_pca[:, 0], centers_pca[:, 1],
                c='red', marker='X', s=300, edgecolor='black',
                linewidth=2, label='聚类中心')

axes[1].set_xlabel(f'PC1 ({pca_cluster.explained_variance_ratio_[0]*100:.1f}%)', fontsize=12)
axes[1].set_ylabel(f'PC2 ({pca_cluster.explained_variance_ratio_[1]*100:.1f}%)', fontsize=12)
axes[1].set_title('K-means聚类结果', fontsize=14, fontweight='bold')
axes[1].legend()
axes[1].grid(True, alpha=0.3)
plt.colorbar(scatter2, ax=axes[1])

plt.tight_layout()
plt.savefig('11-多元统计-06-聚类结果.png', dpi=300, bbox_inches='tight')
plt.show()

# ===== 5. 层次聚类 =====
print("\n" + "=" * 60)
print("步骤3：层次聚类（Hierarchical Clustering）")
print("=" * 60)

# 使用前50个样本（层次聚类计算密集）
X_sample = X_cluster_scaled[:50]

# 层次聚类
linkage_matrix = linkage(X_sample, method='ward')

# 绘制树状图
plt.figure(figsize=(14, 6))
dendrogram(linkage_matrix,
           labels=[df_customers.iloc[i, -1] for i in range(50)],
           leaf_font_size=8)
plt.xlabel('客户ID', fontsize=12)
plt.ylabel('距离', fontsize=12)
plt.title('层次聚类树状图（前50个客户）', fontsize=14, fontweight='bold')
plt.axhline(y=10, color='red', linestyle='--', linewidth=2, label='切割线（3类）')
plt.legend()
plt.tight_layout()
plt.savefig('11-多元统计-07-层次聚类.png', dpi=300, bbox_inches='tight')
plt.show()

print("\n" + "=" * 60)
print("案例2 完成！")
print("=" * 60)

print("""
聚类分析总结：
将200个客户分成3类：VIP、普通、潜力
K-means快速有效
层次聚类提供树状结构
可用于差异化营销策略
""")
```

## 案例3：因子分析

```python
print("\n" + "=" * 60)
```

## 案例4：MANOVA（多元方差分析）

```python
print("\n" + "=" * 60)
print("案例4：MANOVA - 多元方差分析")
print("=" * 60)

print("""
场景：
比较3种教学方法对学生成绩的影响。
因变量：数学、英语两门成绩（多元）。
问题：3种方法对成绩整体是否有显著影响？
""")

# ===== 1. 生成数据 =====
np.random.seed(202)
n_per_method = 30

# 方法1：传统
method1_math = np.random.normal(70, 10, n_per_method)
method1_eng = np.random.normal(65, 12, n_per_method)

# 方法2：创新A（数学好）
method2_math = np.random.normal(80, 11, n_per_method)
method2_eng = np.random.normal(68, 13, n_per_method)

# 方法3：创新B（英语好）
method3_math = np.random.normal(73, 10, n_per_method)
method3_eng = np.random.normal(78, 11, n_per_method)

df_manova = pd.DataFrame({
    '数学': np.concatenate([method1_math, method2_math, method3_math]),
    '英语': np.concatenate([method1_eng, method2_eng, method3_eng]),
    '教学方法': ['传统']*n_per_method + ['创新A']*n_per_method + ['创新B']*n_per_method
})

print("\n数据概览：")
print(df_manova.head())

print("\n各组统计：")
print(df_manova.groupby('教学方法').mean())

# ===== 2. MANOVA =====
print("\n" + "=" * 60)
print("步骤1：多元方差分析（MANOVA）")
print("=" * 60)

# 使用statsmodels
manova = MANOVA.from_formula('数学 + 英语 ~ 教学方法', data=df_manova)
manova_results = manova.mv_test()

print("\nMANOVA结果：")
print(manova_results)

print("""
解释：
- Wilks' Lambda：检验统计量（越小越显著）
- F统计量和p值：判断显著性
- 如果p<0.05：教学方法对成绩整体有显著影响
""")

# ===== 3. 后续单变量ANOVA =====
print("\n" + "=" * 60)
print("步骤2：单变量ANOVA（后续分析）")
print("=" * 60)

from scipy import stats

# 数学成绩ANOVA
f_math, p_math = stats.f_oneway(method1_math, method2_math, method3_math)
print(f"\n数学成绩ANOVA：")
print(f"  F统计量: {f_math:.4f}")
print(f"  p值: {p_math:.4f}")

# 英语成绩ANOVA
f_eng, p_eng = stats.f_oneway(method1_eng, method2_eng, method3_eng)
print(f"\n英语成绩ANOVA：")
print(f"  F统计量: {f_eng:.4f}")
print(f"  p值: {p_eng:.4f}")

# ===== 4. 可视化 =====
fig, axes = plt.subplots(1, 2, figsize=(14, 6))

# 数学成绩
df_manova.boxplot(column='数学', by='教学方法', ax=axes[0])
axes[0].set_title('数学成绩分布', fontsize=12, fontweight='bold')
axes[0].set_xlabel('教学方法', fontsize=11)
axes[0].set_ylabel('数学成绩', fontsize=11)

# 英语成绩
df_manova.boxplot(column='英语', by='教学方法', ax=axes[1])
axes[1].set_title('英语成绩分布', fontsize=12, fontweight='bold')
axes[1].set_xlabel('教学方法', fontsize=11)
axes[1].set_ylabel('英语成绩', fontsize=11)

plt.suptitle('')  # 移除默认标题
plt.tight_layout()
plt.savefig('11-多元统计-10-MANOVA.png', dpi=300, bbox_inches='tight')
plt.show()

print("\n" + "=" * 60)
print("案例4 完成！")
print("=" * 60)
```

## PCA vs 因子分析

```python
print("\n" + "=" * 60)
print("PCA vs 因子分析：区别与选择")
print("=" * 60)

对比 = """
【相同点】
都用于降维
都识别变量间的关系
输出都包含载荷和得分

【关键区别】

1. 目的
   PCA：数据压缩，最大化方差
   FA：识别潜在因子，解释共同方差

2. 模型假设
   PCA：X = 主成分的线性组合
   FA：X = 因子的线性组合 + 特殊因子

3. 主成分/因子
   PCA：主成分是观测变量的线性组合
   FA：观测变量是因子的线性组合（因果方向相反！）

4. 方差分解
   PCA：解释总方差（包括特殊方差）
   FA：只解释共同方差

5. 旋转
   PCA：通常不旋转
   FA：经常旋转（如Varimax）以更好解释

6. 使用场景
   PCA：
   - 纯粹降维（如图像压缩）
   - 预处理（去相关）
   - 可视化

   FA：
   - 心理测量（识别人格维度）
   - 问卷设计（验证量表结构）
   - 理论验证（测试假设的因子结构）

【如何选择？】

用PCA如果：
主要目的是降维
不关心潜在变量
要保留尽可能多的信息

用FA如果：
假设存在潜在因子
要解释共同变异
理论驱动的研究

【实践建议】
💡 先用PCA探索，再用FA验证
💡 如果载荷模式清晰，两者结果相似
💡 FA需要更大样本量（通常 n ≥ 5×变量数）
"""

print(对比)
```

## 实战建议

```python
实战技巧 = """
【PCA】
数据必须标准化（如果变量量纲不同）
检查变量相关性（相关性低时PCA效果不佳）
保留的主成分：解释方差>80%或特征值>1
⚠️ 主成分可能难以解释
💡 画载荷图帮助解释

【聚类分析】
数据标准化很重要
K-means对初始值敏感，多运行几次
用肘部法则+轮廓系数确定K
⚠️ 对异常值敏感
⚠️ 假设簇是球形的
💡 尝试多种方法（K-means, 层次, DBSCAN）
💡 结合业务理解命名聚类

【因子分析】
样本量：至少5倍变量数，最好10倍
检查KMO和Bartlett检验（是否适合FA）
因子旋转（Varimax）提高可解释性
⚠️ 因子数量选择很重要
💡 结合理论确定因子数
💡 检查因子的内部一致性（Cronbach's α）

【MANOVA】
假设：多元正态性、协方差矩阵齐性
样本量：每组至少20
如果显著，进行单变量ANOVA
⚠️ 功效低于多个ANOVA（但控制I类错误）
💡 检查Box's M检验（协方差齐性）

【通用建议】
1. 永远先画图，理解数据结构
2. 检查假设（正态性、相关性等）
3. 标准化数据（大多数方法）
4. 交叉验证结果
5. 结合领域知识解释
6. 报告完整（方法、参数、诊断）

【样本量要求】
- PCA：变量数的3-5倍
- 因子分析：变量数的5-10倍
- 聚类：每簇至少20个样本
- MANOVA：每组至少20个样本

【软件选择】
- sklearn：PCA, 聚类
- statsmodels：MANOVA
- factor_analyzer：因子分析（更专业）
- scipy：距离矩阵、层次聚类
"""
```
