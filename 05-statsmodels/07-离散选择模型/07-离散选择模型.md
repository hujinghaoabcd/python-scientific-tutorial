# 第07章：离散选择模型

当我们要预测的不是连续数值，而是"选择"时，离散选择模型就派上用场了！从用户购买决策到投票选择，从交通方式到职业选择，离散选择模型无处不在。

## 从生活中理解离散选择

### 场景1：买不买iPhone？

```python
# 你在考虑是否买新iPhone
选择 = ["买", "不买"]  # 只有两个选项（二元选择）

# 影响因素
影响因素 = {
    "价格": 9999,      # 价格越高，越不想买
    "收入": 15000,     # 收入越高，越可能买

    "旧手机年龄": 3    # 旧手机越旧，越想换
}

# 这就是二元离散选择问题！
# 用Logit或Probit模型
```

```python
# 你有多种交通方式可选

```

### 场景3：对服务的满意度？


### 什么是离散选择模型？

```python
离散选择模型 = {
    "定义": "预测个体在有限选项中的选择",

    "vs 线性回归": {
        "线性回归": "预测连续值（如收入、价格）",
        "离散选择": "预测类别（如买/不买，选A/B/C）"
    },

    "核心思想": "计算每个选项的'效用'，选择效用最大的",

    "类型": [
        "二元选择：Logit, Probit（2个选项）",
        "多元选择：Multinomial Logit（3+个无序选项）",
        "有序选择：Ordered Logit/Probit（3+个有序选项）",
        "条件选择：Conditional Logit（选项有特征）"
    ]
}
```

## 二元选择模型：Logit vs Probit

### 理论基础

```python
# 效用理论
效用模型 = """
个体i选择选项j的效用：
U_ij = V_ij + ε_ij

其中：
- V_ij = β'X_i：可观测效用（由特征决定）
- ε_ij：随机误差项（不可观测）

个体选择效用最大的选项：
选择1 当 U_i1 > U_i0
即 V_i1 + ε_i1 > V_i0 + ε_i0
即 β'X_i > ε_i0 - ε_i1

P(选择1) = P(ε_i0 - ε_i1 < β'X_i)
"""

print(效用模型)

# Logit vs Probit的区别
差异 = {
    "Logit": {
        "假设": "误差项服从极值分布",
        "概率函数": "Logistic函数: 1 / (1 + e^(-x))",
        "优点": "计算简单，系数易解释（odds ratio）",
        "缺点": "假设IIA（无关选项独立性）"
    },

    "Probit": {
        "假设": "误差项服从正态分布",
        "概率函数": "标准正态CDF: Φ(x)",
        "优点": "更灵活，符合正态假设",
        "缺点": "计算复杂，系数不直观"
    },

    "实践建议": "两者结果通常很接近，Logit更常用"
}
```

## 案例1：信用卡违约预测（完整流程）

```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
from statsmodels.discrete.discrete_model import Logit, Probit
import statsmodels.api as sm
import statsmodels.formula.api as smf
from sklearn.model_selection import train_test_split
from sklearn.metrics import (confusion_matrix, classification_report,
                              roc_curve, roc_auc_score, accuracy_score)
import warnings
warnings.filterwarnings('ignore')

# 设置中文显示
plt.rcParams['font.sans-serif'] = ['SimHei']
plt.rcParams['axes.unicode_minus'] = False
plt.style.use('seaborn-v0_8-darkgrid')

print("=" * 60)
print("案例1：信用卡违约预测")
print("=" * 60)

# ===== 1. 生成模拟数据 =====
np.random.seed(42)
n_samples = 2000

# 生成特征
df = pd.DataFrame({
    '年龄': np.random.randint(22, 70, n_samples),
    '收入': np.random.exponential(8, n_samples) + 2,  # 万元/月
    '负债率': np.random.uniform(0, 1, n_samples),      # 0-1
    '信用卡数量': np.random.randint(1, 8, n_samples),
    '逾期次数': np.random.poisson(0.5, n_samples),     # 历史逾期
    '账户年限': np.random.randint(1, 20, n_samples),
    '婚姻状况': np.random.choice(['未婚', '已婚', '离异'], n_samples, p=[0.3, 0.6, 0.1])
})

# 生成违约标签（基于特征的逻辑关系）
# 违约概率 = f(低收入, 高负债率, 多次逾期...)
logit_score = (
    -3.0                                    # 截距
    - 0.02 * df['年龄']                     # 年龄大 → 违约少
    - 0.3 * df['收入']                      # 收入高 → 违约少
    + 3.0 * df['负债率']                    # 负债高 → 违约多
    + 0.1 * df['信用卡数量']                # 卡多 → 违约多
    + 0.5 * df['逾期次数']                  # 逾期多 → 违约多
    - 0.05 * df['账户年限']                 # 年限长 → 违约少
    + (df['婚姻状况'] == '离异').astype(int) * 0.5  # 离异 → 违约略多
)

# 转换为概率
prob_default = 1 / (1 + np.exp(-logit_score))

# 生成0/1标签
df['是否违约'] = (np.random.rand(n_samples) < prob_default).astype(int)

print(f"\n数据概览：")
print(f"样本数量: {len(df)}")
print(f"违约率: {df['是否违约'].mean():.2%}")
print(f"\n前5行数据：")
print(df.head())

print(f"\n描述统计：")
print(df.describe())

# ===== 2. 探索性数据分析 =====
print("\n" + "=" * 60)
print("步骤1：探索性数据分析")
print("=" * 60)

# 违约率分析
fig, axes = plt.subplots(2, 3, figsize=(16, 10))

# 按年龄分组的违约率
age_bins = [20, 30, 40, 50, 60, 70]
df['年龄段'] = pd.cut(df['年龄'], bins=age_bins)
age_default = df.groupby('年龄段')['是否违约'].mean()
age_default.plot(kind='bar', ax=axes[0, 0], color='steelblue')
axes[0, 0].set_title('不同年龄段的违约率', fontweight='bold')
axes[0, 0].set_ylabel('违约率')
axes[0, 0].set_xlabel('年龄段')
axes[0, 0].tick_params(axis='x', rotation=45)

# 按收入分组的违约率
income_bins = [0, 5, 10, 15, 100]
df['收入段'] = pd.cut(df['收入'], bins=income_bins, labels=['低', '中', '高', '很高'])
income_default = df.groupby('收入段')['是否违约'].mean()
income_default.plot(kind='bar', ax=axes[0, 1], color='coral')
axes[0, 1].set_title('不同收入段的违约率', fontweight='bold')
axes[0, 1].set_ylabel('违约率')
axes[0, 1].set_xlabel('收入段')
axes[0, 1].tick_params(axis='x', rotation=0)

# 按负债率分组的违约率
debt_bins = [0, 0.3, 0.6, 1.0]
df['负债段'] = pd.cut(df['负债率'], bins=debt_bins, labels=['低', '中', '高'])
debt_default = df.groupby('负债段')['是否违约'].mean()
debt_default.plot(kind='bar', ax=axes[0, 2], color='lightgreen')
axes[0, 2].set_title('不同负债率的违约率', fontweight='bold')
axes[0, 2].set_ylabel('违约率')
axes[0, 2].set_xlabel('负债率')
axes[0, 2].tick_params(axis='x', rotation=0)

# 按逾期次数的违约率
overdue_default = df.groupby('逾期次数')['是否违约'].mean()
overdue_default.plot(kind='bar', ax=axes[1, 0], color='salmon')
axes[1, 0].set_title('不同逾期次数的违约率', fontweight='bold')
axes[1, 0].set_ylabel('违约率')
axes[1, 0].set_xlabel('逾期次数')
axes[1, 0].tick_params(axis='x', rotation=0)

# 按婚姻状况的违约率
marriage_default = df.groupby('婚姻状况')['是否违约'].mean()
marriage_default.plot(kind='bar', ax=axes[1, 1], color='plum')
axes[1, 1].set_title('不同婚姻状况的违约率', fontweight='bold')
axes[1, 1].set_ylabel('违约率')
axes[1, 1].set_xlabel('婚姻状况')
axes[1, 1].tick_params(axis='x', rotation=0)

# 特征相关性

corr = df[numeric_cols].corr()['是否违约'].drop('是否违约').sort_values()
corr.plot(kind='barh', ax=axes[1, 2], color=['red' if x > 0 else 'green' for x in corr])
axes[1, 2].set_title('特征与违约的相关性', fontweight='bold')
axes[1, 2].set_xlabel('相关系数')
axes[1, 2].axvline(x=0, color='black', linestyle='--', alpha=0.5)

print("\n" + "=" * 60)
print("步骤2：数据准备")

df_model = df.copy()
df_model = pd.get_dummies(df_model, columns=['婚姻状况'], drop_first=True)

feature_cols = ['年龄', '收入', '负债率', '信用卡数量', '逾期次数',

print("步骤3：Logit模型拟合")

def evaluate_binary_model(y_true, y_pred_prob, threshold=0.5, model_name=''):
    """评估二元分类模型"""
    print(f"\n{'='*50}")
    print(f"{model_name} 评估结果（阈值={threshold}）")
    print(f"{'='*50}")

    # 转换为0/1预测
    y_pred = (y_pred_prob >= threshold).astype(int)

    # 1. 混淆矩阵
    cm = confusion_matrix(y_true, y_pred)
    print("\n混淆矩阵：")
    print(pd.DataFrame(cm,
                       columns=['预测：未违约', '预测：违约'],
                       index=['实际：未违约', '实际：违约']))

    # 2. 准确率、精确率、召回率
    acc = accuracy_score(y_true, y_pred)
    print(f"\n准确率 (Accuracy): {acc:.4f}")

    report = classification_report(y_true, y_pred,
                                    target_names=['未违约', '违约'],
                                    output_dict=True)
    print(f"精确率 (Precision) - 违约: {report['违约']['precision']:.4f}")
    print(f"召回率 (Recall) - 违约: {report['违约']['recall']:.4f}")
    print(f"F1分数 - 违约: {report['违约']['f1-score']:.4f}")

    # 3. ROC-AUC
    auc = roc_auc_score(y_true, y_pred_prob)
    print(f"\nROC-AUC: {auc:.4f}")

    return {
        'accuracy': acc,
        'precision': report['违约']['precision'],
        'recall': report['违约']['recall'],
        'f1': report['违约']['f1-score'],
        'auc': auc
    }

# 评估Logit模型

                                       threshold=0.5, model_name='Logit模型')

# 评估Probit模型

# 计算不同阈值下的指标

    if len(np.unique(y_pred)) > 1:  # 确保预测有两类

    threshold_metrics.append({
        '阈值': thresh,
        '准确率': acc,
        '精确率': precision,
        '召回率': recall,
        'F1': f1
```

```python
print("\n" + "=" * 60)
print("有序Logit建模")
print("=" * 60)

print("""
有序Logit模型的思想：

1. 存在一个潜在的连续变量 y*（如"真实满意度"）
   y* = β'X + ε

2. 我们观察到的是离散类别 y：
   y = 0 (非常不满意) if y* ≤ τ₁
   y = 1 (不满意)     if τ₁ < y* ≤ τ₂
   y = 2 (一般)       if τ₂ < y* ≤ τ₃
   y = 3 (满意)       if τ₃ < y* ≤ τ₄
   y = 4 (非常满意)   if y* > τ₄

3. τ₁, τ₂, τ₃, τ₄ 是"阈值参数"，需要估计

4. P(y=j|X) = P(τⱼ < y* ≤ τⱼ₊₁)
            = F(τⱼ₊₁ - β'X) - F(τⱼ - β'X)

   其中F是logistic函数（Logit）或标准正态CDF（Probit）

优点：
考虑了类别的顺序信息
比多元Logit更有效（参数更少）
系数解释直观：正系数表示倾向更高类别

限制：
⚠️ 平行线假设：每个特征对所有类别的影响方向相同
⚠️ 如果假设不成立，需要用Generalized Ordered Logit
""")

# 使用简化方法：将有序问题转换为多个二元问题（累积logit）

y_binary = (y_ord <= 2).astype(int)

print("\n满意度 ≤ 一般 vs > 一般：")
print(results_ord.summary())

print(f"质量评分系数: {results_ord.params['质量评分']:.4f}")
if results_ord.params['质量评分'] < 0:
    print("  → 质量越好，越不会停留在'一般'及以下，即越可能满意")

print(f"价格系数: {results_ord.params['价格']:.4f}")
if results_ord.params['价格'] > 0:
    print("  → 价格越高，越可能停留在'一般'及以下，即满意度下降")

scenarios_sat_X = sm.add_constant(scenarios_sat)

# 预测概率（满意度 ≤ 一般的概率）
```

```python
1. 二元变量（0/1，是/否，买/不买）

   ├─ 推荐Logit（更常用，系数易解释）
   └─ 如果强调正态假设 → Probit

2. 多元变量（3+个类别）
   ├─ 类别无顺序（如：交通方式、品牌选择）
   │  └─ 用Multinomial Logit
   │     ⚠️ 注意IIA假设（无关选项独立性）

   │
   └─ 类别有顺序（如：满意度、教育水平）

   └─ 用Poisson或Negative Binomial（见第03章）

4. 条件选择（选项本身有特征）
```

**本章重点**
- 离散选择模型预测类别，不是连续值
- Logit最常用：Odds Ratio易解释
- Multinomial Logit：多元无序选择
- Ordered Logit：多元有序选择
- 效用理论：个体选择效用最大的选项
- 重要假设：IIA、平行线
- 评估：混淆矩阵、ROC-AUC、准确率

**记住**
- 系数 ≠ 概率变化，需要计算边际效应
- Odds Ratio > 1：正向影响；< 1：负向影响
- 检验模型假设很重要（IIA、平行线）
- 从简单模型开始，逐步复杂化
- 可视化预测概率帮助理解
