# 函数最小化

学习如何使用 SciPy 寻找函数的最小值。

## 一维函数最小化

### 使用 minimize_scalar

```python
from scipy.optimize import minimize_scalar
import numpy as np
import matplotlib.pyplot as plt

# 定义函数
def f(x):
    return (x - 2)**2 + 1

# 寻找最小值
result = minimize_scalar(f)

print(f"最小值位置: x = {result.x:.4f}")
print(f"最小值: f(x) = {result.fun:.4f}")
print(f"成功: {result.success}")

# 可视化
x = np.linspace(-2, 6, 200)
plt.plot(x, f(x), 'b-', linewidth=2, label='f(x)')
plt.plot(result.x, result.fun, 'ro', markersize=12, label='最小值')
plt.xlabel('x')
plt.ylabel('f(x)')
plt.title('一维函数最小化')
plt.legend()
plt.grid(True, alpha=0.3)
plt.show()
```

### 指定搜索区间

```python
# 在特定范围内搜索
result = minimize_scalar(f, bounds=(0, 5), method='bounded')

print(f"在 [0, 5] 内的最小值: x = {result.x:.4f}")
```

### method 参数

```python
# 'brent' - Brent 方法（默认，无界）
result1 = minimize_scalar(f, method='brent')

# 'bounded' - 有界搜索
result2 = minimize_scalar(f, bounds=(0, 5), method='bounded')

# 'golden' - 黄金分割搜索
result3 = minimize_scalar(f, method='golden')

print(f"Brent:   {result1.x:.4f}")
print(f"Bounded: {result2.x:.4f}")
print(f"Golden:  {result3.x:.4f}")
```

## 多维函数最小化

### 使用 minimize

```python
from scipy.optimize import minimize

# 二维函数
def f_2d(x):
    """
    x 是一个数组 [x0, x1]
    """
    return x[0]**2 + x[1]**2 + 2*x[0] + 4*x[1]

# 初始猜测
x0 = np.array([0, 0])

# 优化
result = minimize(f_2d, x0)

print(f"最优解: x = {result.x}")
print(f"最小值: f(x) = {result.fun:.4f}")
```

### 可视化二维优化

```python
def plot_2d_optimization(func, result, x_range, y_range, title=''):
    """可视化二维函数的优化过程"""
    # 创建网格
    x = np.linspace(x_range[0], x_range[1], 100)
    y = np.linspace(y_range[0], y_range[1], 100)
    X, Y = np.meshgrid(x, y)

    # 计算函数值
    Z = np.zeros_like(X)
    for i in range(X.shape[0]):
        for j in range(X.shape[1]):
            Z[i, j] = func([X[i, j], Y[i, j]])

    # 绘制等高线图
    plt.figure(figsize=(10, 8))
    contour = plt.contour(X, Y, Z, levels=20, cmap='viridis')
    plt.colorbar(contour, label='f(x, y)')

    # 标记最优点
    plt.plot(result.x[0], result.x[1], 'r*', markersize=20, label='最优解')

    plt.xlabel('x')
    plt.ylabel('y')
    plt.title(title or '二维函数优化')
    plt.legend()
    plt.grid(True, alpha=0.3)
    plt.show()

# 使用
plot_2d_optimization(f_2d, result, (-5, 3), (-5, 3),
                     '二维二次函数最小化')
```

## 优化算法（method 参数）

### 常用算法对比

```python
# 测试函数：Rosenbrock 函数
def rosenbrock(x):
    return (1 - x[0])**2 + 100*(x[1] - x[0]**2)**2

x0 = np.array([0, 0])

# 不同算法
methods = ['BFGS', 'Nelder-Mead', 'CG', 'Powell']

print("算法性能对比（Rosenbrock 函数）:")
print("=" * 60)

for method in methods:
    result = minimize(rosenbrock, x0, method=method)
    print(f"{method:15s} | 迭代: {result.nfev:4d} | "
          f"最小值: {result.fun:.8f} | 成功: {result.success}")
```

### 各算法特点

#### 1. BFGS（推荐默认选择）

```python
# BFGS - 拟牛顿法
# 优点：快速、可靠、适用广泛
# 缺点：需要计算梯度（自动数值计算）

result_bfgs = minimize(rosenbrock, x0, method='BFGS')
```

**适用场景**：
- 光滑函数
- 中小规模问题（< 1000 变量）
- 无约束优化首选

#### 2. Nelder-Mead（不需要梯度）

```python
# Nelder-Mead - 单纯形法
# 优点：不需要梯度，稳健
# 缺点：较慢，高维性能差

result_nm = minimize(rosenbrock, x0, method='Nelder-Mead')
```

**适用场景**：
- 不可微函数
- 噪声函数
- 低维问题（< 10 变量）

#### 3. CG（共轭梯度）

```python
# CG - 共轭梯度法
# 优点：内存效率高
# 缺点：可能较慢

result_cg = minimize(rosenbrock, x0, method='CG')
```

**适用场景**：
- 大规模问题
- 内存受限
- 稀疏问题

#### 4. Powell

```python
# Powell - Powell 方法
# 优点：不需要梯度
# 缺点：高维可能不稳定

result_powell = minimize(rosenbrock, x0, method='Powell')
```

## 提供梯度加速优化

### 自动梯度 vs 手动梯度

```python
# 方法1：让 SciPy 自动计算梯度（默认）
result_auto = minimize(rosenbrock, x0, method='BFGS')

# 方法2：手动提供梯度函数
def rosenbrock_grad(x):
    """Rosenbrock 函数的梯度"""
    dfdx0 = -2*(1 - x[0]) - 400*x[0]*(x[1] - x[0]**2)
    dfdx1 = 200*(x[1] - x[0]**2)
    return np.array([dfdx0, dfdx1])

result_manual = minimize(rosenbrock, x0, method='BFGS',
                         jac=rosenbrock_grad)

print(f"自动梯度 - 函数调用次数: {result_auto.nfev}")
print(f"手动梯度 - 函数调用次数: {result_manual.nfev}")
```

**结论**：手动梯度更快，但需要额外工作。对于复杂函数，自动梯度已经足够好。

## 优化选项

### options 参数

```python
# 设置优化选项
options = {
    'maxiter': 1000,      # 最大迭代次数
    'disp': True,         # 显示优化过程
    'gtol': 1e-6,         # 梯度容差
}

result = minimize(rosenbrock, x0, method='BFGS', options=options)
```

### 容差参数

```python
# tol 参数：控制收敛精度
result_low = minimize(rosenbrock, x0, tol=1e-3)   # 低精度
result_high = minimize(rosenbrock, x0, tol=1e-10)  # 高精度

print(f"低精度  - 迭代: {result_low.nit}, 最小值: {result_low.fun:.10f}")
print(f"高精度 - 迭代: {result_high.nit}, 最小值: {result_high.fun:.10f}")
```

## 实际应用案例

### 案例1：投资组合优化

```python
"""
场景：分配资金到不同资产，最大化收益，同时控制风险
"""

# 三种资产的预期收益率和风险（标准差）
returns = np.array([0.10, 0.15, 0.20])  # 10%, 15%, 20%
risks = np.array([0.05, 0.10, 0.15])

def portfolio_risk(weights):
    """
    计算投资组合风险（要最小化）

    约束：
    - 权重和为 1
    - 预期收益不低于 15%
    """
    # 简化的风险模型：加权标准差
    risk = np.sum(weights * risks)
    return risk

def portfolio_return(weights):
    """计算投资组合预期收益"""
    return np.sum(weights * returns)

# 约束条件
constraints = [
    # 权重和为 1
    {'type': 'eq', 'fun': lambda w: np.sum(w) - 1},
    # 预期收益 >= 15%
    {'type': 'ineq', 'fun': lambda w: portfolio_return(w) - 0.15},
]

# 边界：权重在 [0, 1] 之间
bounds = [(0, 1)] * 3

# 初始猜测：平均分配
x0 = np.array([1/3, 1/3, 1/3])

# 优化
result = minimize(portfolio_risk, x0,
                  method='SLSQP',
                  constraints=constraints,
                  bounds=bounds)

print("=" * 60)
print("投资组合优化结果")
print("=" * 60)
print(f"资产配置比例: {result.x}")
print(f"总风险: {result.fun:.4f}")
print(f"预期收益: {portfolio_return(result.x):.4f} ({portfolio_return(result.x)*100:.1f}%)")
print(f"权重总和: {np.sum(result.x):.4f}")
```

### 案例2：参数拟合（最小二乘）

```python
"""
场景：根据实验数据拟合指数衰减模型
"""

# 真实模型：y = A * exp(-k * t) + C
A_true, k_true, C_true = 10, 0.5, 2

# 生成实验数据（带噪声）
t_data = np.linspace(0, 10, 50)
y_data = A_true * np.exp(-k_true * t_data) + C_true + np.random.randn(50) * 0.5

def model(t, params):
    """指数衰减模型"""
    A, k, C = params
    return A * np.exp(-k * t) + C

def residuals_squared(params):
    """残差平方和（要最小化）"""
    y_pred = model(t_data, params)
    return np.sum((y_data - y_pred)**2)

# 初始猜测
params0 = [5, 0.1, 0]

# 优化
result = minimize(residuals_squared, params0, method='Nelder-Mead')
A_fit, k_fit, C_fit = result.x

print("=" * 60)
print("指数衰减拟合结果")
print("=" * 60)
print(f"真实参数: A={A_true}, k={k_true}, C={C_true}")
print(f"拟合参数: A={A_fit:.2f}, k={k_fit:.2f}, C={C_fit:.2f}")
print(f"残差平方和: {result.fun:.2f}")

# 可视化
plt.figure(figsize=(10, 6))
plt.scatter(t_data, y_data, alpha=0.5, label='实验数据')
t_plot = np.linspace(0, 10, 200)
plt.plot(t_plot, model(t_plot, [A_true, k_true, C_true]),
         'g--', label='真实模型')
plt.plot(t_plot, model(t_plot, result.x),
         'r-', linewidth=2, label='拟合模型')
plt.xlabel('时间')
plt.ylabel('测量值')
plt.legend()
plt.grid(True, alpha=0.3)
plt.title('指数衰减拟合')
plt.show()
```

### 案例3：机器学习 - 逻辑回归

```python
"""
场景：使用优化实现简单的逻辑回归分类器
"""

# 生成二分类数据
np.random.seed(42)
n_samples = 100

# 类别 0
X0 = np.random.randn(n_samples//2, 2) + np.array([2, 2])
# 类别 1
X1 = np.random.randn(n_samples//2, 2) + np.array([-2, -2])

X = np.vstack([X0, X1])
y = np.hstack([np.zeros(n_samples//2), np.ones(n_samples//2)])

def sigmoid(z):
    """Sigmoid 函数"""
    return 1 / (1 + np.exp(-z))

def logistic_loss(params, X, y):
    """逻辑回归损失函数（交叉熵）"""
    w, b = params[:-1], params[-1]
    z = X @ w + b
    y_pred = sigmoid(z)

    # 交叉熵损失
    loss = -np.mean(y * np.log(y_pred + 1e-10) +
                    (1 - y) * np.log(1 - y_pred + 1e-10))
    return loss

# 初始参数：w1, w2, b
params0 = np.zeros(3)

# 优化
result = minimize(logistic_loss, params0, args=(X, y), method='BFGS')

w_fit = result.x[:-1]
b_fit = result.x[-1]

print("逻辑回归参数:")
print(f"权重 w: {w_fit}")
print(f"偏置 b: {b_fit:.4f}")
print(f"最终损失: {result.fun:.4f}")

# 可视化决策边界
plt.figure(figsize=(10, 8))
plt.scatter(X0[:, 0], X0[:, 1], c='blue', alpha=0.5, label='类别 0')
plt.scatter(X1[:, 0], X1[:, 1], c='red', alpha=0.5, label='类别 1')

# 绘制决策边界
x1_boundary = np.linspace(X[:, 0].min(), X[:, 0].max(), 100)
x2_boundary = -(w_fit[0] * x1_boundary + b_fit) / w_fit[1]
plt.plot(x1_boundary, x2_boundary, 'k-', linewidth=2, label='决策边界')

plt.xlabel('特征 1')
plt.ylabel('特征 2')
plt.legend()
plt.title('逻辑回归决策边界')
plt.grid(True, alpha=0.3)
plt.show()
```

## 调试和故障排除

### 检查优化结果

```python
def validate_result(result):
    """验证优化结果"""
    print("=" * 60)
    print("优化结果检查")
    print("=" * 60)

    if result.success:
        print("✓ 优化成功")
    else:
        print("✗ 优化失败")
        print(f"  原因: {result.message}")

    print(f"\n最优解: {result.x}")
    print(f"目标函数值: {result.fun:.6f}")
    print(f"函数调用次数: {result.nfev}")

    if hasattr(result, 'nit'):
        print(f"迭代次数: {result.nit}")

    # 检查梯度（如果可用）
    if hasattr(result, 'jac'):
        grad_norm = np.linalg.norm(result.jac)
        print(f"\n梯度范数: {grad_norm:.6e}")
        if grad_norm < 1e-5:
            print("✓ 梯度接近零（局部最优）")
        else:
            print("⚠ 梯度较大，可能未收敛")

# 使用
result = minimize(rosenbrock, [0, 0], method='BFGS')
validate_result(result)
```

### 常见问题

**问题1：收敛到局部最优**

```python
# 解决方案：多次尝试不同初始点
def multi_start_optimization(func, bounds, n_starts=10):
    """多起点优化"""
    best_result = None
    best_value = np.inf

    for i in range(n_starts):
        # 随机初始点
        x0 = np.random.uniform(bounds[:, 0], bounds[:, 1])
        result = minimize(func, x0)

        if result.fun < best_value:
            best_value = result.fun
            best_result = result

    return best_result
```

**问题2：优化不收敛**

```python
# 可能原因：
# 1. 函数不连续或有噪声 → 使用 Nelder-Mead
# 2. 初始点离最优点太远 → 尝试不同初始点
# 3. 函数梯度过大 → 归一化或缩放
# 4. 数值精度问题 → 调整容差参数
```

## 小结

- `minimize_scalar()` 用于一维优化
- `minimize()` 用于多维优化
- BFGS 是默认首选算法
- 初始猜测很重要
- 验证优化结果的成功和合理性
- 提供梯度可以加速（可选）
