# 一维插值实践

本章通过实际代码学习如何使用 SciPy 进行一维插值。

## 基础：使用 interp1d

### 最简单的例子

```python
import numpy as np
import matplotlib.pyplot as plt
from scipy.interpolate import interp1d

# 已知数据点
x = np.array([0, 1, 2, 3, 4, 5])
y = np.array([0, 1, 4, 9, 16, 25])  # y = x²

# 创建插值函数
f = interp1d(x, y)

# 计算插值
x_new = 2.5
y_new = f(x_new)

print(f"在 x = {x_new} 处的插值结果: y = {y_new}")
```

**输出**：
```
在 x = 2.5 处的插值结果: y = 6.5
```

### 在多个点上插值

```python
# 在多个点上插值
x_new_array = np.array([0.5, 1.5, 2.5, 3.5, 4.5])
y_new_array = f(x_new_array)

print("批量插值结果：")
for xi, yi in zip(x_new_array, y_new_array):
    print(f"  x = {xi} → y = {yi}")
```

### 生成平滑曲线

```python
# 生成密集的点来绘制平滑曲线
x_smooth = np.linspace(0, 5, 100)  # 100个均匀分布的点
y_smooth = f(x_smooth)

# 绘图
plt.figure(figsize=(10, 6))
plt.plot(x, y, 'o', markersize=10, label='原始数据点')
plt.plot(x_smooth, y_smooth, '-', linewidth=2, label='插值曲线')
plt.xlabel('x')
plt.ylabel('y')
plt.title('线性插值示例')
plt.legend()
plt.grid(True)
plt.show()
```

## 不同的插值方法

### 方法对比

```python
import numpy as np
import matplotlib.pyplot as plt
from scipy.interpolate import interp1d

# 准备数据
x = np.array([0, 1, 2, 3, 4, 5])
y = np.array([0, 2, 1, 3, 2, 1])

# 创建不同的插值函数
f_linear = interp1d(x, y, kind='linear')
f_quadratic = interp1d(x, y, kind='quadratic')
f_cubic = interp1d(x, y, kind='cubic')

# 密集采样点
x_dense = np.linspace(0, 5, 200)

# 计算插值
y_linear = f_linear(x_dense)
y_quadratic = f_quadratic(x_dense)
y_cubic = f_cubic(x_dense)

# 绘图对比
plt.figure(figsize=(12, 8))

methods = [
    ('linear', y_linear, '线性插值'),
    ('quadratic', y_quadratic, '二次插值'),
    ('cubic', y_cubic, '三次插值')
]

for i, (kind, y_interp, title) in enumerate(methods, 1):
    plt.subplot(2, 2, i)
    plt.plot(x, y, 'ro', markersize=10, label='原始数据', zorder=3)
    plt.plot(x_dense, y_interp, 'b-', linewidth=2, label=title)
    plt.xlabel('x')
    plt.ylabel('y')
    plt.title(title)
    plt.legend()
    plt.grid(True, alpha=0.3)

# 全部方法对比
plt.subplot(2, 2, 4)
plt.plot(x, y, 'ko', markersize=10, label='原始数据', zorder=3)
plt.plot(x_dense, y_linear, '-', label='线性', alpha=0.7)
plt.plot(x_dense, y_quadratic, '-', label='二次', alpha=0.7)
plt.plot(x_dense, y_cubic, '-', label='三次', alpha=0.7)
plt.xlabel('x')
plt.ylabel('y')
plt.title('所有方法对比')
plt.legend()
plt.grid(True, alpha=0.3)

plt.tight_layout()
plt.show()
```

## interp1d 的参数详解

### kind 参数（插值类型）

```python
# 可用的 kind 值：
# 'linear' - 线性插值（默认）
# 'nearest' - 最近邻插值（阶梯状）
# 'zero', 'slinear' - 零阶、一阶样条
# 'quadratic' - 二次插值
# 'cubic' - 三次插值
# 整数 - 指定阶数（如 5 表示 5 阶插值）

x = np.array([0, 1, 2, 3, 4])
y = np.array([0, 1, 0, 1, 0])

# 最近邻插值
f_nearest = interp1d(x, y, kind='nearest')

x_dense = np.linspace(0, 4, 100)
plt.plot(x, y, 'o', label='原始数据')
plt.plot(x_dense, f_nearest(x_dense), '-', label='最近邻（阶梯）')
plt.legend()
plt.title('最近邻插值 - 适合分类数据')
plt.show()
```

### fill_value 参数（边界外的值）

```python
x = np.array([1, 2, 3, 4])
y = np.array([10, 20, 30, 40])

# 默认：边界外报错
f_default = interp1d(x, y)

# 允许外推（不推荐）
f_extrapolate = interp1d(x, y, fill_value='extrapolate')

# 指定边界外的固定值
f_fixed = interp1d(x, y, fill_value=0, bounds_error=False)

# 测试
x_test = np.array([0.5, 2.5, 4.5])  # 包含边界外的点

try:
    print("默认行为（边界外报错）:")
    print(f_default(x_test))
except ValueError as e:
    print(f"  错误: {e}")

print("\n允许外推:")
print(f_extrapolate(x_test))

print("\n边界外使用固定值 0:")
print(f_fixed(x_test))
```

**输出**：
```
默认行为（边界外报错）:
  错误: A value in x_new is below the interpolation range.

允许外推:
[0. 25. 50.]

边界外使用固定值 0:
[ 0. 25.  0.]
```

### bounds_error 参数

```python
# bounds_error=True（默认）：边界外报错
# bounds_error=False：使用 fill_value

f1 = interp1d(x, y, bounds_error=True)  # 严格模式
f2 = interp1d(x, y, bounds_error=False, fill_value=np.nan)  # 宽松模式
```

## 样条插值：UnivariateSpline

`UnivariateSpline` 是更强大的插值工具，提供更多控制。

### 基本用法

```python
from scipy.interpolate import UnivariateSpline

x = np.array([0, 1, 2, 3, 4, 5])
y = np.array([0, 1, 4, 9, 16, 25])

# 创建样条插值
# s=0: 精确通过所有点（插值）
# s>0: 允许误差（拟合）
spl = UnivariateSpline(x, y, s=0)

x_dense = np.linspace(0, 5, 100)
y_spline = spl(x_dense)

plt.plot(x, y, 'o', label='原始数据')
plt.plot(x_dense, y_spline, '-', label='样条插值')
plt.legend()
plt.show()
```

### 平滑样条（s 参数）

```python
# 生成带噪声的数据
x = np.linspace(0, 10, 20)
y_true = np.sin(x)
y_noisy = y_true + np.random.normal(0, 0.1, x.shape)

# 不同的平滑度
spl_interpolate = UnivariateSpline(x, y_noisy, s=0)  # 插值：通过所有点
spl_smooth1 = UnivariateSpline(x, y_noisy, s=0.5)    # 轻微平滑
spl_smooth2 = UnivariateSpline(x, y_noisy, s=2)      # 较强平滑

x_dense = np.linspace(0, 10, 200)

plt.figure(figsize=(12, 4))

# 插值
plt.subplot(1, 3, 1)
plt.plot(x, y_noisy, 'o', label='带噪声数据')
plt.plot(x_dense, spl_interpolate(x_dense), '-', label='s=0 (插值)')
plt.plot(x_dense, np.sin(x_dense), '--', alpha=0.5, label='真实函数')
plt.title('s=0：精确插值\n（通过所有点，包括噪声）')
plt.legend()
plt.grid(True)

# 轻微平滑
plt.subplot(1, 3, 2)
plt.plot(x, y_noisy, 'o', label='带噪声数据')
plt.plot(x_dense, spl_smooth1(x_dense), '-', label='s=0.5 (轻微平滑)')
plt.plot(x_dense, np.sin(x_dense), '--', alpha=0.5, label='真实函数')
plt.title('s=0.5：轻微平滑\n（更接近真实函数）')
plt.legend()
plt.grid(True)

# 较强平滑
plt.subplot(1, 3, 3)
plt.plot(x, y_noisy, 'o', label='带噪声数据')
plt.plot(x_dense, spl_smooth2(x_dense), '-', label='s=2 (较强平滑)')
plt.plot(x_dense, np.sin(x_dense), '--', alpha=0.5, label='真实函数')
plt.title('s=2：较强平滑\n（可能过度平滑）')
plt.legend()
plt.grid(True)

plt.tight_layout()
plt.show()
```

### 样条求导

样条插值的一个优势是可以轻松求导。

```python
from scipy.interpolate import UnivariateSpline

# 原函数：y = x²
x = np.linspace(0, 5, 20)
y = x**2

# 创建样条
spl = UnivariateSpline(x, y, s=0)

# 求导（导数应该是 dy/dx = 2x）
spl_derivative = spl.derivative()

# 求积分
spl_integral = spl.antiderivative()

x_dense = np.linspace(0, 5, 100)

plt.figure(figsize=(12, 4))

# 原函数
plt.subplot(1, 3, 1)
plt.plot(x_dense, spl(x_dense), 'b-', label='y = x²')
plt.plot(x, y, 'ro')
plt.title('原函数')
plt.legend()
plt.grid(True)

# 一阶导数
plt.subplot(1, 3, 2)
plt.plot(x_dense, spl_derivative(x_dense), 'r-', label="y' = 2x")
plt.plot(x_dense, 2*x_dense, 'g--', alpha=0.5, label='理论值')
plt.title('一阶导数')
plt.legend()
plt.grid(True)

# 积分
plt.subplot(1, 3, 3)
plt.plot(x_dense, spl_integral(x_dense), 'g-', label='积分')
plt.title('不定积分')
plt.legend()
plt.grid(True)

plt.tight_layout()
plt.show()
```

## 实际应用案例

### 案例1：温度数据插值

```python
"""
场景：气象站每小时记录一次温度，
想要获得每分钟的温度估算
"""

# 每小时的温度记录（0-24小时）
hours = np.array([0, 3, 6, 9, 12, 15, 18, 21, 24])
temperatures = np.array([15, 14, 16, 20, 28, 30, 26, 20, 16])

# 创建插值函数
temp_interp = interp1d(hours, temperatures, kind='cubic')

# 每分钟的时间点
minutes = np.arange(0, 24*60 + 1) / 60  # 0 到 24 小时，以分钟为单位
temp_每分钟 = temp_interp(minutes)

# 可视化
plt.figure(figsize=(12, 6))
plt.plot(hours, temperatures, 'ro-', markersize=10,
         linewidth=2, label='实际测量（每3小时）')
plt.plot(minutes, temp_每分钟, 'b-', alpha=0.5,
         label='插值结果（每分钟）')
plt.xlabel('时间（小时）')
plt.ylabel('温度（°C）')
plt.title('一天中的温度变化')
plt.legend()
plt.grid(True, alpha=0.3)
plt.show()

# 查询特定时刻
query_times = [7.5, 10.25, 14.75, 19.5]
print("\n特定时刻的温度估算：")
for t in query_times:
    temp = temp_interp(t)
    print(f"  {t:5.2f} 点：{temp:.1f}°C")
```

### 案例2：缺失数据填补

```python
"""
场景：传感器数据有部分缺失，需要填补
"""

# 完整的时间序列
time_full = np.arange(0, 100)
signal_true = np.sin(time_full * 0.1) + 0.5 * np.sin(time_full * 0.05)

# 模拟数据缺失（随机删除30%的数据）
np.random.seed(42)
keep_indices = np.random.choice(len(time_full), int(len(time_full) * 0.7),
                                replace=False)
keep_indices = np.sort(keep_indices)

time_measured = time_full[keep_indices]
signal_measured = signal_true[keep_indices]

# 使用插值填补缺失数据
interpolator = interp1d(time_measured, signal_measured, kind='cubic')
signal_filled = interpolator(time_full)

# 可视化
plt.figure(figsize=(14, 5))

plt.subplot(1, 2, 1)
plt.plot(time_full, signal_true, 'g-', alpha=0.3,
         linewidth=3, label='真实信号')
plt.plot(time_measured, signal_measured, 'bo',
         markersize=4, label='实际测量（70%）')
plt.title('原始数据（有缺失）')
plt.xlabel('时间')
plt.ylabel('信号值')
plt.legend()
plt.grid(True, alpha=0.3)

plt.subplot(1, 2, 2)
plt.plot(time_full, signal_true, 'g-', alpha=0.3,
         linewidth=3, label='真实信号')
plt.plot(time_full, signal_filled, 'r-', alpha=0.7,
         label='插值填补')
plt.plot(time_measured, signal_measured, 'bo',
         markersize=4, label='实际测量')
plt.title('插值填补后')
plt.xlabel('时间')
plt.ylabel('信号值')
plt.legend()
plt.grid(True, alpha=0.3)

plt.tight_layout()
plt.show()

# 评估插值误差
error = np.abs(signal_filled - signal_true)
print(f"\n平均绝对误差: {np.mean(error):.4f}")
print(f"最大绝对误差: {np.max(error):.4f}")
```

## 常见错误和解决方案

### 错误1：数据点太少

```python
# 错误：只有两个点，无法进行三次插值
x = np.array([0, 1])
y = np.array([0, 1])

try:
    f = interp1d(x, y, kind='cubic')
except ValueError as e:
    print(f"错误: {e}")

# 解决方案：使用线性插值，或增加数据点
f = interp1d(x, y, kind='linear')  # OK
```

### 错误2：x 值不是递增的

```python
# 错误：x 值必须严格递增
x = np.array([0, 2, 1, 3])  # 不是递增的
y = np.array([0, 2, 1, 3])

try:
    f = interp1d(x, y)
except ValueError as e:
    print(f"错误: {e}")

# 解决方案：排序
sorted_indices = np.argsort(x)
x_sorted = x[sorted_indices]
y_sorted = y[sorted_indices]
f = interp1d(x_sorted, y_sorted)  # OK
```

### 错误3：查询点超出范围

```python
x = np.array([0, 1, 2])
y = np.array([0, 1, 4])
f = interp1d(x, y)

# 错误：查询点 3 超出了范围 [0, 2]
try:
    f(3)
except ValueError as e:
    print(f"错误: {e}")

# 解决方案1：使用 bounds_error=False
f = interp1d(x, y, bounds_error=False, fill_value='extrapolate')
print(f"外推结果: {f(3)}")

# 解决方案2：只查询范围内的点
x_query = np.clip(3, x.min(), x.max())
print(f"裁剪到范围内: {f(x_query)}")
```

## 性能提示

```python
import time

# 大数据集测试
x = np.linspace(0, 100, 10000)
y = np.sin(x) + np.random.randn(len(x)) * 0.1

# 方法1：interp1d（较慢，但功能丰富）
start = time.time()
f1 = interp1d(x, y, kind='cubic')
result1 = f1(np.linspace(0, 100, 50000))
time1 = time.time() - start

# 方法2：UnivariateSpline（较快）
start = time.time()
f2 = UnivariateSpline(x, y, s=0)
result2 = f2(np.linspace(0, 100, 50000))
time2 = time.time() - start

print(f"interp1d: {time1:.3f} 秒")
print(f"UnivariateSpline: {time2:.3f} 秒")
```

## 小结

- `interp1d`：简单易用，适合大多数情况
- `UnivariateSpline`：更强大，支持平滑和求导
- 选择合适的 `kind` 参数很重要
- 注意边界条件和数据范围
- 数据有噪声时考虑使用平滑样条
