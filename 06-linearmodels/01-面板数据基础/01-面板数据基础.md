# 第01章：面板数据基础

面板数据的结构和准备是一切的基础！这一章我们深入理解面板数据的组织形式，以及如何用 LinearModels 进行基础分析。

## 面板数据的结构

### 长格式 vs 宽格式

```python
import numpy as np
import pandas as pd

print("=" * 70)
print("面板数据格式：长格式 vs 宽格式")
print("=" * 70)

# ===== 宽格式（Wide Format）=====
print("\n宽格式（Wide Format）：")
print("每个个体一行，不同时间的数据放在不同列")

wide_data = pd.DataFrame({
    '企业': ['A', 'B', 'C'],
    '销售额_2020': [100, 80, 120],
    '销售额_2021': [110, 85, 130],
    '销售额_2022': [120, 90, 140]
})

print(wide_data)

print("\n优点：")
print("  ✅ 人类直观易读")
print("  ✅ Excel常用格式")
print("\n缺点：")
print("  ❌ 不利于统计分析")
print("  ❌ LinearModels不支持")

# ===== 长格式（Long Format）=====
print("\n" + "=" * 70)
print("长格式（Long Format）：")
print("每个(个体, 时间)组合一行")

long_data = pd.DataFrame({
    '企业': ['A', 'A', 'A', 'B', 'B', 'B', 'C', 'C', 'C'],
    '年份': [2020, 2021, 2022] * 3,
    '销售额': [100, 110, 120, 80, 85, 90, 120, 130, 140]
})

print(long_data)

print("\n优点：")
print("  ✅ 统计分析标准格式")
print("  ✅ LinearModels要求格式")
print("  ✅ 易于添加新变量")
print("\n缺点：")
print("  ❌ 行数更多")

# ===== 格式转换 =====
print("\n" + "=" * 70)
print("格式转换")
print("=" * 70)

# 宽格式 → 长格式
print("\n1. 宽格式 → 长格式（pd.melt）")
long_from_wide = wide_data.melt(
    id_vars='企业',
    var_name='变量',
    value_name='值'
)
long_from_wide['年份'] = long_from_wide['变量'].str.extract('(\d+)').astype(int)
long_from_wide = long_from_wide.drop('变量', axis=1)
print(long_from_wide)

# 长格式 → 宽格式
print("\n2. 长格式 → 宽格式（pd.pivot）")
wide_from_long = long_data.pivot(
    index='企业',
    columns='年份',
    values='销售额'
)
wide_from_long.columns = [f'销售额_{col}' for col in wide_from_long.columns]
print(wide_from_long)
```

## 案例1：企业生产率分析（平衡面板）

```python
print("\n" + "=" * 70)
print("案例1：企业生产率分析")
print("=" * 70)

print("""
场景：
分析100家制造企业的生产率，跟踪5年（2018-2022）。

变量：
- log_output: 对数产出（因变量）
- log_labor: 对数劳动投入
- log_capital: 对数资本投入
- rd_intensity: 研发强度（研发支出/销售额）
- export: 是否出口（0或1）

问题：
1. 研发投入如何影响生产率？
2. 出口企业更有效率吗？
""")

# ===== 1. 生成数据 =====
np.random.seed(123)
n_firms = 100
n_years = 5
years = range(2018, 2023)

# 生成面板数据
data_list = []
for firm_id in range(1, n_firms + 1):
    # 企业固定效应（生产率差异）
    firm_effect = np.random.normal(0, 0.5)

    # 企业特征（时间不变）
    rd_propensity = np.random.beta(2, 5)  # 研发倾向
    export_prob = np.random.uniform(0.2, 0.8)

    for t, year in enumerate(years):
        # 生成变量
        log_labor = np.random.normal(4, 0.5) + 0.02 * t
        log_capital = np.random.normal(5, 0.6) + 0.03 * t
        rd_intensity = rd_propensity * np.random.beta(3, 10)
        export = int(np.random.rand() < export_prob)

        # 生成产出（柯布-道格拉斯生产函数）
        log_output = (
            2.0 +                              # TFP基准
            0.6 * log_labor +                  # 劳动弹性
            0.3 * log_capital +                # 资本弹性
            0.5 * rd_intensity +               # 研发效应
            0.2 * export +                     # 出口溢价
            0.02 * t +                         # 时间趋势
            firm_effect +                      # 企业固定效应
            np.random.normal(0, 0.15)          # 随机冲击
        )

        data_list.append({
            'firm_id': firm_id,
            'year': year,
            'log_output': log_output,
            'log_labor': log_labor,
            'log_capital': log_capital,
            'rd_intensity': rd_intensity,
            'export': export
        })

df_prod = pd.DataFrame(data_list)

# 设置面板索引
df_prod = df_prod.set_index(['firm_id', 'year'])

print(f"\n数据概览：")
print(df_prod.head(10))

print(f"\n数据维度：")
print(f"  企业数量: {n_firms}")
print(f"  时间跨度: {n_years} 年")
print(f"  总观测值: {len(df_prod)}")

print(f"\n面板平衡性检查：")
obs_per_firm = df_prod.reset_index().groupby('firm_id').size()
print(f"  每个企业的观测数: {obs_per_firm.unique()}")
if len(obs_per_firm.unique()) == 1:
    print(f"  ✅ 平衡面板")
else:
    print(f"  ⚠️ 非平衡面板")

print(f"\n描述性统计：")
print(df_prod.describe())

# ===== 2. 混合OLS =====
from linearmodels import PooledOLS

print("\n" + "=" * 70)
print("模型1：混合OLS")
print("=" * 70)

pooled_model = PooledOLS.from_formula(
    'log_output ~ log_labor + log_capital + rd_intensity + export',
    data=df_prod
)
pooled_result = pooled_model.fit(cov_type='clustered', cluster_entity=True)

print(pooled_result)

print("\n解释：")
print(f"  劳动弹性: {pooled_result.params['log_labor']:.3f}")
print(f"  资本弹性: {pooled_result.params['log_capital']:.3f}")
print(f"  研发效应: {pooled_result.params['rd_intensity']:.3f}")
print(f"    - 研发强度每提高1个百分点，产出提高{pooled_result.params['rd_intensity']:.2%}")
print(f"  出口溢价: {pooled_result.params['export']:.3f}")
print(f"    - 出口企业产出比非出口企业高{pooled_result.params['export']:.2%}")

# ===== 3. 固定效应模型 =====
from linearmodels import PanelOLS

print("\n" + "=" * 70)
print("模型2：固定效应模型")
print("=" * 70)

fe_model = PanelOLS.from_formula(
    'log_output ~ log_labor + log_capital + rd_intensity + export + EntityEffects',
    data=df_prod
)
fe_result = fe_model.fit(cov_type='clustered', cluster_entity=True)

print(fe_result)

print("\n关键发现：")
print(f"  研发效应（固定效应）: {fe_result.params['rd_intensity']:.3f}")
print(f"  研发效应（混合OLS）: {pooled_result.params['rd_intensity']:.3f}")
print(f"  差异: {fe_result.params['rd_intensity'] - pooled_result.params['rd_intensity']:.3f}")

print("\n为什么不同？")
print("  - 混合OLS：高生产率企业可能本身研发更多（反向因果）")
print("  - 固定效应：控制了企业固定差异，更接近真实因果效应")

# ===== 4. 双向固定效应 =====
print("\n" + "=" * 70)
print("模型3：双向固定效应（个体 + 时间）")
print("=" * 70)

twoway_fe_model = PanelOLS.from_formula(
    'log_output ~ log_labor + log_capital + rd_intensity + export + EntityEffects + TimeEffects',
    data=df_prod
)
twoway_fe_result = twoway_fe_model.fit(cov_type='clustered', cluster_entity=True)

print(twoway_fe_result)

print("\n为什么加时间固定效应？")
print("  - 控制所有企业共同的时间趋势")
print("  - 例如：经济周期、技术进步、政策变化")
print("  - 确保识别的是企业内部的时间变化效应")

# ===== 5. 可视化 =====
import matplotlib.pyplot as plt
import seaborn as sns

plt.rcParams['font.sans-serif'] = ['SimHei']
plt.rcParams['axes.unicode_minus'] = False

fig, axes = plt.subplots(2, 2, figsize=(14, 10))

# 5.1 研发强度分布
df_reset = df_prod.reset_index()
axes[0, 0].hist(df_reset['rd_intensity'], bins=30, color='steelblue', alpha=0.7, edgecolor='black')
axes[0, 0].set_xlabel('研发强度', fontsize=11)
axes[0, 0].set_ylabel('频数', fontsize=11)
axes[0, 0].set_title('研发强度分布', fontsize=13, fontweight='bold')
axes[0, 0].grid(axis='y', alpha=0.3)

# 5.2 出口 vs 非出口企业的产出
sns.boxplot(data=df_reset, x='export', y='log_output', ax=axes[0, 1])
axes[0, 1].set_xlabel('出口状态', fontsize=11)
axes[0, 1].set_ylabel('对数产出', fontsize=11)
axes[0, 1].set_title('出口 vs 非出口企业的产出', fontsize=13, fontweight='bold')
axes[0, 1].set_xticklabels(['非出口', '出口'])

# 5.3 产出随时间变化
avg_output = df_reset.groupby('year')['log_output'].mean()
axes[1, 0].plot(avg_output.index, avg_output.values, marker='o', linewidth=2, markersize=8, color='coral')
axes[1, 0].set_xlabel('年份', fontsize=11)
axes[1, 0].set_ylabel('平均对数产出', fontsize=11)
axes[1, 0].set_title('产出随时间的变化', fontsize=13, fontweight='bold')
axes[1, 0].grid(True, alpha=0.3)

# 5.4 系数比较
models = ['混合OLS', '固定效应', '双向固定效应']
rd_coefs = [
    pooled_result.params['rd_intensity'],
    fe_result.params['rd_intensity'],
    twoway_fe_result.params['rd_intensity']
]
colors = ['orange', 'steelblue', 'green']

axes[1, 1].bar(models, rd_coefs, color=colors, alpha=0.7)
axes[1, 1].axhline(y=0.5, color='red', linestyle='--', label='真实值')
axes[1, 1].set_ylabel('研发效应系数', fontsize=11)
axes[1, 1].set_title('不同模型的研发效应估计', fontsize=13, fontweight='bold')
axes[1, 1].legend()
axes[1, 1].grid(axis='y', alpha=0.3)

plt.tight_layout()
plt.savefig('01-面板数据基础-01-企业生产率.png', dpi=300, bbox_inches='tight')
plt.show()

print("\n" + "=" * 70)
print("案例1 完成！")
print("=" * 70)
```

## 个体效应和时间效应

```python
print("\n" + "=" * 70)
print("理解个体效应和时间效应")
print("=" * 70)

个体和时间效应 = """
【面板数据回归模型】

基本形式：
y_it = α + β'X_it + ε_it

其中：
- i: 个体索引（如企业、个人、国家）
- t: 时间索引（如年份、季度）
- y_it: 因变量
- X_it: 自变量
- ε_it: 误差项

【分解误差项】

ε_it = μ_i + λ_t + ν_it

其中：
- μ_i: 个体效应（entity effect）
  → 个体特有的、时间不变的因素
  → 例如：企业管理能力、个人天赋、地理位置

- λ_t: 时间效应（time effect）
  → 所有个体共同的时间趋势
  → 例如：经济周期、技术进步、政策变化

- ν_it: 特质误差（idiosyncratic error）
  → 随个体和时间变化的随机冲击

【三种模型】

1. 混合OLS（Pooled OLS）
   y_it = α + β'X_it + ε_it

   假设：μ_i = 0, λ_t = 0
   问题：忽略个体和时间效应 → 有偏差

2. 固定效应（Fixed Effects）
   y_it = α_i + β'X_it + ε_it

   处理：为每个个体i添加虚拟变量 α_i
   效果：控制个体效应 μ_i
   代价：不能估计时间不变的变量

3. 双向固定效应（Two-Way Fixed Effects）
   y_it = α_i + γ_t + β'X_it + ε_it

   处理：同时添加个体和时间虚拟变量
   效果：控制 μ_i 和 λ_t
   最常用：经济学实证研究的标准做法
"""

print(个体和时间效应)
```

## 案例2：城市房价分析（非平衡面板）

```python
print("\n" + "=" * 70)
print("案例2：城市房价分析（非平衡面板）")
print("=" * 70)

print("""
场景：
分析50个城市的房价，2015-2022年（8年）。
部分城市数据缺失（非平衡面板）。

变量：
- price: 房价（万元/平米）
- gdp: GDP（亿元）
- population: 人口（万人）
- metro: 地铁线路数
- policy: 限购政策（0或1）

问题：
1. GDP增长如何影响房价？
2. 限购政策有效吗？
3. 如何处理非平衡面板？
""")

# ===== 1. 生成非平衡面板数据 =====
np.random.seed(456)
n_cities = 50
years = range(2015, 2023)

data_list = []
for city_id in range(1, n_cities + 1):
    # 城市固定效应
    city_effect = np.random.normal(0, 0.3)

    # 城市特征
    base_gdp = np.random.uniform(5000, 50000)
    base_pop = np.random.uniform(500, 3000)

    # 决定该城市有多少年的数据（模拟缺失）
    n_obs = np.random.choice([6, 7, 8], p=[0.2, 0.3, 0.5])
    available_years = np.random.choice(years, size=n_obs, replace=False)

    for t, year in enumerate(available_years):
        year_idx = year - 2015

        gdp = base_gdp * (1.06 ** year_idx) * np.random.uniform(0.95, 1.05)
        population = base_pop * (1.02 ** year_idx) * np.random.uniform(0.98, 1.02)
        metro = min(int(year_idx * 0.5), 10) + np.random.poisson(1)
        policy = 1 if (city_id > 25 and year >= 2017) else 0

        # 房价模型
        price = (
            2.0 +
            0.3 * np.log(gdp) +
            0.2 * np.log(population) +
            0.05 * metro +
            -0.15 * policy +
            0.03 * year_idx +
            city_effect +
            np.random.normal(0, 0.2)
        )

        data_list.append({
            'city_id': city_id,
            'year': year,
            'price': price,
            'log_gdp': np.log(gdp),
            'log_pop': np.log(population),
            'metro': metro,
            'policy': policy
        })

df_housing = pd.DataFrame(data_list)

# 设置索引
df_housing = df_housing.set_index(['city_id', 'year'])

print(f"\n数据概览：")
print(df_housing.head(10))

print(f"\n面板平衡性检查：")
obs_per_city = df_housing.reset_index().groupby('city_id').size()
print(f"  观测数分布:")
print(obs_per_city.value_counts().sort_index())
print(f"  ⚠️ 非平衡面板：有的城市缺少某些年份的数据")

print(f"\n总观测值: {len(df_housing)}")
print(f"如果是平衡面板应有: {n_cities * len(years)} 个观测值")
print(f"缺失率: {(1 - len(df_housing) / (n_cities * len(years))) * 100:.1f}%")

# ===== 2. 固定效应估计 =====
print("\n" + "=" * 70)
print("固定效应模型（处理非平衡面板）")
print("=" * 70)

fe_housing = PanelOLS.from_formula(
    'price ~ log_gdp + log_pop + metro + policy + EntityEffects + TimeEffects',
    data=df_housing
)
fe_housing_result = fe_housing.fit(cov_type='clustered', cluster_entity=True)

print(fe_housing_result)

print("\n关键发现：")
print(f"  GDP弹性: {fe_housing_result.params['log_gdp']:.3f}")
print(f"    - GDP每增长1%, 房价上涨{fe_housing_result.params['log_gdp']:.2%}")
print(f"  限购政策效应: {fe_housing_result.params['policy']:.3f}")
if fe_housing_result.params['policy'] < 0:
    print(f"    - 限购政策使房价下降{abs(fe_housing_result.params['policy']):.2f}万元/平米")
    if fe_housing_result.pvalues['policy'] < 0.05:
        print(f"    - ✅ 政策显著有效（p={fe_housing_result.pvalues['policy']:.4f}）")
    else:
        print(f"    - ⚠️ 政策效果不显著（p={fe_housing_result.pvalues['policy']:.4f}）")

print("\nLinearModels如何处理非平衡面板？")
print("  ✅ 自动使用所有可用数据")
print("  ✅ 固定效应对缺失数据稳健")
print("  ⚠️ 但要注意样本选择偏差")

# ===== 3. 可视化 =====
fig, axes = plt.subplots(2, 2, figsize=(14, 10))

df_housing_reset = df_housing.reset_index()

# 3.1 观测数分布
obs_counts = df_housing_reset.groupby('city_id').size()
axes[0, 0].hist(obs_counts, bins=range(6, 10), color='steelblue', alpha=0.7, edgecolor='black')
axes[0, 0].set_xlabel('每个城市的观测数', fontsize=11)
axes[0, 0].set_ylabel('城市数量', fontsize=11)
axes[0, 0].set_title('非平衡面板：观测数分布', fontsize=13, fontweight='bold')
axes[0, 0].grid(axis='y', alpha=0.3)

# 3.2 限购政策效果
policy_effect = df_housing_reset.groupby(['year', 'policy'])['price'].mean().unstack()
policy_effect.plot(ax=axes[0, 1], marker='o', linewidth=2)
axes[0, 1].set_xlabel('年份', fontsize=11)
axes[0, 1].set_ylabel('平均房价', fontsize=11)
axes[0, 1].set_title('限购 vs 非限购城市房价', fontsize=13, fontweight='bold')
axes[0, 1].legend(['非限购', '限购'], fontsize=10)
axes[0, 1].grid(True, alpha=0.3)

# 3.3 GDP vs 房价
sample_cities = df_housing_reset['city_id'].unique()[:10]
for city in sample_cities:
    city_data = df_housing_reset[df_housing_reset['city_id'] == city]
    axes[1, 0].plot(city_data['log_gdp'], city_data['price'], 'o-', alpha=0.5, markersize=4)

axes[1, 0].set_xlabel('对数GDP', fontsize=11)
axes[1, 0].set_ylabel('房价', fontsize=11)
axes[1, 0].set_title('GDP vs 房价（前10个城市）', fontsize=13, fontweight='bold')
axes[1, 0].grid(True, alpha=0.3)

# 3.4 数据缺失模式
missing_pattern = np.zeros((n_cities, len(years)))
for idx, row in df_housing_reset.iterrows():
    city_idx = row['city_id'] - 1
    year_idx = row['year'] - 2015
    missing_pattern[city_idx, year_idx] = 1

im = axes[1, 1].imshow(missing_pattern[:20], cmap='RdYlGn', aspect='auto')
axes[1, 1].set_xlabel('年份', fontsize=11)
axes[1, 1].set_ylabel('城市ID', fontsize=11)
axes[1, 1].set_title('数据缺失模式（前20城市）', fontsize=13, fontweight='bold')
axes[1, 1].set_xticks(range(len(years)))
axes[1, 1].set_xticklabels(years)
plt.colorbar(im, ax=axes[1, 1], label='1=有数据, 0=缺失')

plt.tight_layout()
plt.savefig('01-面板数据基础-02-城市房价.png', dpi=300, bbox_inches='tight')
```

```python
【1. 控制不可观测的个体异质性】

问题：
每个个体有不同的"天赋"，这些因素：
- 无法直接测量（如能力、管理水平）
- 但影响因变量
- 在横截面回归中导致遗漏变量偏差

✅ 固定效应模型给每个个体一个"专属截距"
✅ 控制所有时间不变的个体特征
✅ 专注分析时间变化的效应

例子：
研究教育对收入的影响
- 横截面：能力高的人教育多，收入也高 → 高估教育效应
- 面板数据 + 固定效应：控制个人能力 → 准确估计

【2. 更多信息，更高效率】

样本量：
- 横截面：N个个体
- 面板数据：N个个体 × T个时期 = N×T个观测

好处：
✅ 标准误更小
✅ 估计更精确
✅ 检验更有力

例子：
50个城市 × 10年 = 500个观测

- 投资回报需要时间
- 习惯形成需要观察多期

面板数据解决方案：
✅ 可以估计滞后效应
✅ 可以分析调整过程
✅ 可以研究因果关系的动态

例子：
研发投入对生产率的影响
- 横截面：只能看当期关系
- 面板数据：可以看3年后、5年后的效果

【4. 识别因果关系】

因果推断的黄金标准：
双重差分（Difference-in-Differences, DID）
- 需要面板数据
- 比较"处理组"和"对照组"在政策前后的差异
- 控制时间趋势和个体差异

例子：
评估最低工资法对就业的影响
- 处理组：提高最低工资的州
- 对照组：未提高的州
- 比较：(处理后-处理前) - (对照后-对照前)

【5. 减少多重共线性】

问题：
横截面数据中，变量之间可能高度相关

面板数据优势：
✅ 利用时间变化，增加变量的独立性
✅ 更容易分离各变量的独立效应

例子：
教育和经验高度相关（年龄大的人两者都高）

- 面板数据：利用个体内部的时间变化来识别
```

- 固定效应不能估计时间不变的变量
