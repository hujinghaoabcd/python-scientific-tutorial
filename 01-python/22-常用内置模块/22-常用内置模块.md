# 第22章：常用内置模块

掌握Python标准库中的常用模块,让编程事半功倍。

## 什么是内置模块?

想象你要开发一个程序,需要处理日期、读写文件、生成随机数...如果这些功能都要自己从零写,那太费时间了!好消息是,Python自带了一个超级丰富的"工具箱"——**标准库**。

**内置模块就是Python自带的、不需要额外安装的模块**,直接`import`就能用!

### 为什么要学习内置模块?

```python
# 不使用模块 - 自己实现很复杂
def get_current_date():
    # 要考虑闰年、月份天数、时区...
    # 写几百行代码还可能有bug!
    pass

# 使用模块 - 一行搞定
from datetime import datetime
now = datetime.now()
print(now)  # 2024-01-15 14:30:45
```

**内置模块的优势**:
- **省时间** - 不用重复造轮子
- **稳定可靠** - 经过大量测试
- **功能丰富** - 覆盖各种常见需求
- **性能优化** - 很多用C语言实现,速度快
- **免费开源** - 不花一分钱

## datetime - 日期时间模块

处理日期和时间是编程中最常见的需求之一。

### 获取当前日期时间

```python
from datetime import datetime, date, time

# 获取当前日期和时间
now = datetime.now()
print(now)  # 2024-01-15 14:30:45.123456
print(type(now))  # <class 'datetime.datetime'>

# 只获取日期
today = date.today()
print(today)  # 2024-01-15
print(type(today))  # <class 'datetime.date'>

# 只获取时间
current_time = datetime.now().time()
print(current_time)  # 14:30:45.123456
print(type(current_time))  # <class 'datetime.time'>

# 获取具体信息
print(f"年份: {now.year}")      # 2024
print(f"月份: {now.month}")     # 1
print(f"日期: {now.day}")       # 15
print(f"小时: {now.hour}")      # 14
print(f"分钟: {now.minute}")    # 30
print(f"秒数: {now.second}")    # 45
print(f"星期: {now.weekday()}")  # 0=周一, 6=周日
```

### 实战例子：生日倒计时

```python
from datetime import datetime, timedelta

def birthday_countdown(birth_month, birth_day):
    """计算距离下次生日还有多少天"""
    today = datetime.now()
    current_year = today.year

    # 今年的生日
    birthday_this_year = datetime(current_year, birth_month, birth_day)

    # 如果今年生日已过,计算明年的
    if birthday_this_year < today:
        birthday_this_year = datetime(current_year + 1, birth_month, birth_day)

    # 计算天数
    days_left = (birthday_this_year - today).days

    # 是否是今天
    if days_left == 0:
        return "今天就是你的生日!🎉"
    else:
        return f"距离下次生日还有 {days_left} 天"

# 使用
print(birthday_countdown(5, 20))  # 假设生日是5月20日
print(birthday_countdown(1, 15))  # 假设生日是1月15日
```

## collections - 专业容器

`collections`模块提供了比list、dict更强大的容器类型。

### Counter - 计数器

```python
from collections import Counter

# 统计元素出现次数
words = ["apple", "banana", "apple", "cherry", "banana", "apple", "date"]
counter = Counter(words)

print(counter)  # Counter({'apple': 3, 'banana': 2, 'cherry': 1, 'date': 1})

# 最常见的元素
print(counter.most_common(2))  # [('apple', 3), ('banana', 2)]
print(counter.most_common(1))  # [('apple', 3)]

# 访问计数
print(counter["apple"])   # 3
print(counter["grape"])   # 0 (不存在返回0,不会报错)

# 统计字符串
text = "hello world"
char_counter = Counter(text)
print(char_counter)  # Counter({'l': 3, 'o': 2, 'h': 1, 'e': 1, ...})

# 更新计数
counter.update(["apple", "banana"])
print(counter)  # apple变成4, banana变成3

# 减少计数
counter.subtract(["apple"])
print(counter["apple"])  # 3

# 获取总数
print(sum(counter.values()))  # 总共的水果数
```

### 实战例子：投票统计系统

```python
from collections import Counter

class VotingSystem:
    """简单的投票系统"""

    def __init__(self, candidates):
        """初始化候选人"""
        self.candidates = candidates
        self.votes = Counter()

    def vote(self, candidate):
        """投票"""
        if candidate in self.candidates:
            self.votes[candidate] += 1
            print(f"已为 {candidate} 投票")
            return True
        else:
            print(f"候选人 {candidate} 不存在")
            return False

    def get_results(self):
        """获取投票结果"""
        print("\n=== 投票结果 ===")
        total = sum(self.votes.values())

        for candidate, count in self.votes.most_common():
            percentage = (count / total * 100) if total > 0 else 0
            print(f"{candidate}: {count} 票 ({percentage:.1f}%)")

        print(f"\n总投票数: {total}")

    def get_winner(self):
        """获取获胜者"""
        if not self.votes:
            return None
        winner = self.votes.most_common(1)[0]
        return winner[0]

# 使用
voting = VotingSystem(["张三", "李四", "王五"])

# 模拟投票
voting.vote("张三")
voting.vote("李四")
voting.vote("张三")
voting.vote("王五")
voting.vote("张三")
voting.vote("李四")

# 查看结果
voting.get_results()
print(f"\n获胜者: {voting.get_winner()}")
```

### defaultdict - 带默认值的字典

```python
from collections import defaultdict

# 普通字典的问题
normal_dict = {}
# normal_dict["fruits"].append("apple")  # KeyError!

# defaultdict自动创建默认值
fruit_dict = defaultdict(list)

# 默认值为0(用于计数)
word_count = defaultdict(int)
words = ["hello", "world", "hello", "python"]
for word in words:
    word_count[word] += 1

print(dict(word_count))  # {'hello': 2, 'world': 1, 'python': 1}

# 默认值为set
tags = defaultdict(set)
tags["python"].add("programming")
tags["python"].add("scripting")
tags["java"].add("programming")

print(dict(tags))
# {'python': {'programming', 'scripting'}, 'java': {'programming'}}
```

### deque - 双端队列

```python
from collections import deque

# 创建双端队列
queue = deque([1, 2, 3])

# 右侧操作
queue.append(4)         # 右侧添加
print(queue)            # deque([1, 2, 3, 4])
print(queue.pop())      # 4 右侧弹出

# 左侧操作
queue.appendleft(0)     # 左侧添加
print(queue)            # deque([0, 1, 2, 3])
print(queue.popleft())  # 0 左侧弹出

# 旋转
queue.rotate(1)   # 向右旋转1位
print(queue)      # deque([3, 1, 2])

queue.rotate(-1)  # 向左旋转1位
print(queue)      # deque([1, 2, 3])

# 限制长度
limited_queue = deque(maxlen=3)
for i in range(5):
    limited_queue.append(i)
    print(limited_queue)
# deque([0], maxlen=3)
# deque([0, 1], maxlen=3)
# deque([0, 1, 2], maxlen=3)
# deque([1, 2, 3], maxlen=3)  # 0被挤出
# deque([2, 3, 4], maxlen=3)  # 1被挤出
```

## random - 随机数模块

### 生成随机数

```python
import random

# 随机整数
print(random.randint(1, 10))        # 1到10之间(包含1和10)
print(random.randrange(0, 10))      # 0到9之间
print(random.randrange(0, 10, 2))   # 0,2,4,6,8中随机一个

# 随机浮点数
print(random.random())              # 0.0到1.0之间
print(random.uniform(1.5, 5.5))     # 1.5到5.5之间

# 随机选择
colors = ["red", "blue", "green", "yellow"]
print(random.choice(colors))        # 随机选一个

# 随机抽样(不重复)
print(random.sample(colors, 2))     # 随机选2个,不重复
print(random.sample(range(1, 50), 6))  # 模拟彩票

# 随机抽样(可重复)
print(random.choices(colors, k=5))  # 随机选5个,可能重复

# 带权重的随机选择
fruits = ["apple", "banana", "cherry"]
weights = [5, 3, 2]  # apple被选中概率最高
print(random.choices(fruits, weights=weights, k=10))

# 打乱列表
numbers = [1, 2, 3, 4, 5]
random.shuffle(numbers)
print(numbers)  # [3, 1, 5, 2, 4] (顺序随机)
```

### 实战例子：抽奖系统

```python
import random

class LotterySystem:
    """抽奖系统"""

    def __init__(self, prizes):
        """
        初始化奖品
        prizes: {奖品名: (数量, 概率权重)}
        """
        self.prizes = prizes
        self.history = []

    def draw(self):
        """抽奖"""
        # 分离奖品名和权重
        prize_names = []
        weights = []

        for name, (count, weight) in self.prizes.items():
            if count > 0:  # 还有库存
                prize_names.append(name)
                weights.append(weight)

        if not prize_names:
            return "奖品已抽完"

        # 按权重随机选择
        prize = random.choices(prize_names, weights=weights)[0]

        # 减少库存
        count, weight = self.prizes[prize]
        self.prizes[prize] = (count - 1, weight)

        # 记录历史
        self.history.append(prize)

        return prize

    def get_statistics(self):
        """获取抽奖统计"""
        from collections import Counter
        counter = Counter(self.history)

        print("\n=== 抽奖统计 ===")
        for prize, count in counter.most_common():
            print(f"{prize}: {count} 次")
        print(f"总抽奖次数: {len(self.history)}")

# 使用
lottery = LotterySystem({
    "一等奖": (1, 1),      # 1个,权重1
    "二等奖": (3, 5),      # 3个,权重5
    "三等奖": (10, 20),    # 10个,权重20
    "谢谢参与": (100, 74)  # 100个,权重74
})

# 模拟100次抽奖
print("开始抽奖:")
for i in range(100):
    prize = lottery.draw()
    if "等奖" in prize:  # 只显示中奖的
        print(f"第{i+1}次: 恭喜获得{prize}!")

lottery.get_statistics()
```

## itertools - 迭代工具

### 无限迭代器

```python
import itertools

# count - 无限计数
counter = itertools.count(start=10, step=2)
print(next(counter))  # 10
print(next(counter))  # 12
print(next(counter))  # 14

# 常用于配合zip创建编号
names = ["张三", "李四", "王五"]
for i, name in zip(itertools.count(1), names):
    print(f"{i}. {name}")
# 1. 张三
# 2. 李四
# 3. 王五

# cycle - 无限循环
colors = itertools.cycle(["red", "green", "blue"])
for i in range(7):
    print(next(colors))
# red, green, blue, red, green, blue, red

# repeat - 重复元素
for num in itertools.repeat(5, 3):
    print(num)  # 5, 5, 5
```

### 组合迭代器

```python
import itertools

# 排列(考虑顺序)
items = [1, 2, 3]
print(list(itertools.permutations(items, 2)))
# [(1, 2), (1, 3), (2, 1), (2, 3), (3, 1), (3, 2)]

# 组合(不考虑顺序)
print(list(itertools.combinations(items, 2)))
# [(1, 2), (1, 3), (2, 3)]

# 可重复的组合
print(list(itertools.combinations_with_replacement(items, 2)))
# [(1, 1), (1, 2), (1, 3), (2, 2), (2, 3), (3, 3)]

# 笛卡尔积
colors = ["red", "blue"]
sizes = ["S", "M", "L"]
print(list(itertools.product(colors, sizes)))
# [('red', 'S'), ('red', 'M'), ('red', 'L'),
#  ('blue', 'S'), ('blue', 'M'), ('blue', 'L')]
```

### 其他实用工具

```python
import itertools

# chain - 连接多个迭代器
list1 = [1, 2, 3]
```

## functools - 函数工具

### lru_cache - 缓存装饰器

```python
from functools import lru_cache
import time

# 斐波那契数列(低效版本)
def fib_slow(n):
    if n < 2:
        return n
    return fib_slow(n-1) + fib_slow(n-2)

# 使用缓存优化
@lru_cache(maxsize=None)
def fib_fast(n):
    if n < 2:
        return n
    return fib_fast(n-1) + fib_fast(n-2)

# 对比性能
start = time.time()
print(fib_slow(30))
print(f"慢速版本耗时: {time.time() - start:.2f}秒")

start = time.time()
print(fib_fast(30))
print(f"快速版本耗时: {time.time() - start:.4f}秒")

# 查看缓存信息
print(fib_fast.cache_info())
```

### partial - 偏函数

```python
from functools import partial

# 原函数
def power(base, exponent):
    return base ** exponent

# 创建偏函数
square = partial(power, exponent=2)
cube = partial(power, exponent=3)

def log(message, level="INFO"):
    print(f"[{level}] {message}")

# 创建专用日志函数
error_log = partial(log, level="ERROR")
warning_log = partial(log, level="WARNING")

log("正常信息")          # [INFO] 正常信息
error_log("出错了")      # [ERROR] 出错了
warning_log("警告信息")  # [WARNING] 警告信息
```

### 例子1：文件整理工具

```python
from pathlib import Path
import shutil
from datetime import datetime

class FileOrganizer:
    """按文件类型整理文件"""

    def __init__(self, source_dir):
        self.source_dir = Path(source_dir)
        self.file_types = {
            "图片": [".jpg", ".jpeg", ".png", ".gif", ".bmp"],
            "文档": [".txt", ".doc", ".docx", ".pdf", ".xlsx"],
            "视频": [".mp4", ".avi", ".mkv", ".mov"],
            "音频": [".mp3", ".wav", ".flac"],
            "压缩包": [".zip", ".rar", ".7z", ".tar", ".gz"]
        }

    def organize(self):
        """整理文件"""
        if not self.source_dir.exists():
            print(f"目录不存在: {self.source_dir}")
            return

        # 创建分类文件夹
        for category in self.file_types.keys():
            category_dir = self.source_dir / category
            category_dir.mkdir(exist_ok=True)

        # 移动文件
        moved_count = 0
        for file in self.source_dir.iterdir():
            if file.is_file():
                # 查找文件类型
                category = self._get_category(file.suffix.lower())
                if category:
                    dest = self.source_dir / category / file.name

                    # 处理重名
                    if dest.exists():
                        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
                        dest = dest.parent / f"{dest.stem}_{timestamp}{dest.suffix}"

                    shutil.move(str(file), str(dest))
                    print(f"移动: {file.name} -> {category}/")
                    moved_count += 1

        print(f"\n整理完成! 共移动 {moved_count} 个文件")

    def _get_category(self, suffix):
        """根据后缀获取分类"""
        for category, extensions in self.file_types.items():
            if suffix in extensions:
                return category
        return None

# 使用
# organizer = FileOrganizer("Downloads")
# organizer.organize()
```

### 例子2：日程提醒系统

```python
from datetime import datetime, timedelta
from collections import defaultdict

class ScheduleReminder:
    """日程提醒系统"""

    def __init__(self):
        self.schedules = defaultdict(list)

    def add_schedule(self, date_str, time_str, event):
        """添加日程"""
        try:
            dt = datetime.strptime(f"{date_str} {time_str}", "%Y-%m-%d %H:%M")
            self.schedules[date_str].append({
                "time": time_str,
                "event": event,
                "datetime": dt
            })
            print(f"已添加日程: {date_str} {time_str} {event}")
            return True
        except ValueError as e:
            print(f"日期格式错误: {e}")
            return False

    def get_today_schedule(self):
        """获取今天的日程"""
        today = datetime.now().strftime("%Y-%m-%d")
        schedules = self.schedules.get(today, [])

        if not schedules:
            print("今天没有日程安排")
            return

        print(f"\n=== {today} 的日程 ===")
        for schedule in sorted(schedules, key=lambda x: x["time"]):
            print(f"{schedule['time']} - {schedule['event']}")

    def get_upcoming_events(self, days=7):
        """获取未来几天的日程"""
        now = datetime.now()
        upcoming = []

        for date_str, schedules in self.schedules.items():
            for schedule in schedules:
                dt = schedule["datetime"]
                if now < dt <= now + timedelta(days=days):
                    days_left = (dt - now).days
                    upcoming.append({
                        "date": date_str,
                        "time": schedule["time"],
                        "event": schedule["event"],
                        "days_left": days_left
                    })

        if not upcoming:
            print(f"未来{days}天没有日程安排")
            return

        print(f"\n=== 未来{days}天的日程 ===")
        for item in sorted(upcoming, key=lambda x: (x["date"], x["time"])):
            if item["days_left"] == 0:
                time_label = "今天"
            elif item["days_left"] == 1:
                time_label = "明天"
            else:
                time_label = f"{item['days_left']}天后"

            print(f"{item['date']} {item['time']} - {item['event']} ({time_label})")

# 使用
reminder = ScheduleReminder()
reminder.add_schedule("2024-01-15", "09:00", "团队会议")
reminder.add_schedule("2024-01-15", "14:30", "项目评审")
reminder.add_schedule("2024-01-16", "10:00", "客户拜访")
reminder.add_schedule("2024-01-20", "15:00", "培训课程")

reminder.get_today_schedule()
reminder.get_upcoming_events(7)
```

### 例子3：密码生成器

```python
import random
import string

class PasswordGenerator:
    """安全密码生成器"""

    def __init__(self):
        self.lowercase = string.ascii_lowercase
        self.uppercase = string.ascii_uppercase
        self.digits = string.digits
        self.special = "!@#$%^&*()_+-=[]{}|;:,.<>?"

    def generate(self, length=12, use_uppercase=True, use_digits=True,
                 use_special=True, exclude_ambiguous=True):
        """
        生成密码

        参数:
            length: 密码长度
            use_uppercase: 是否包含大写字母
            use_digits: 是否包含数字
            use_special: 是否包含特殊字符
            exclude_ambiguous: 是否排除易混淆字符(如 0,O,l,1)
        """
        # 构建字符池
        chars = self.lowercase

        if use_uppercase:
            chars += self.uppercase
        if use_digits:
            chars += self.digits
        if use_special:
            chars += self.special

        # 排除易混淆字符
        if exclude_ambiguous:
            ambiguous = "0Ol1"
            chars = "".join(c for c in chars if c not in ambiguous)

        # 确保至少包含各种字符类型
        password = []

        if use_uppercase:
            password.append(random.choice(self.uppercase))
        if use_digits:
            password.append(random.choice(self.digits))
        if use_special:
            password.append(random.choice(self.special))

        password.append(random.choice(self.lowercase))

        # 填充剩余长度
        remaining = length - len(password)
        password.extend(random.choices(chars, k=remaining))

        # 打乱顺序
        random.shuffle(password)

        return "".join(password)

    def generate_multiple(self, count=5, **kwargs):
        """生成多个密码"""
        passwords = []
        for i in range(count):
            pwd = self.generate(**kwargs)
            strength = self._check_strength(pwd)
            passwords.append((pwd, strength))
        return passwords

    def _check_strength(self, password):
        """检查密码强度"""
        score = 0

        if len(password) >= 8:
            score += 1
        if len(password) >= 12:
            score += 1
        if any(c.islower() for c in password):
            score += 1
        if any(c.isupper() for c in password):
            score += 1
        if any(c.isdigit() for c in password):
            score += 1
        if any(c in self.special for c in password):
            score += 1

        if score >= 5:
            return "强"
        elif score >= 3:
            return "中"
        else:
            return "弱"

# 使用
generator = PasswordGenerator()

print("=== 生成5个强密码 ===")
passwords = generator.generate_multiple(
    count=5,
    length=16,
    use_uppercase=True,
    use_digits=True,
    use_special=True
)

for pwd, strength in passwords:
    print(f"{pwd} (强度: {strength})")
```


### 1. 使用pathlib代替os.path

```python

import os

    with open(path) as f:
```

### 2. 使用缓存优化递归

```python
# 低效
def fibonacci(n):
    if n < 2:
        return n
    return fibonacci(n-1) + fibonacci(n-2)

# 高效
from functools import lru_cache

@lru_cache(maxsize=None)
def fibonacci(n):
    if n < 2:
        return n
    return fibonacci(n-1) + fibonacci(n-2)
```

### 4. 使用defaultdict避免KeyError
