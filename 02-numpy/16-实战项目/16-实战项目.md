## 项目1：图像处理工具

一个完整的图像处理系统，支持灰度化、滤波、边缘检测等常用操作。

### 需求分析

**项目背景**：

图像处理是计算机视觉的基础。一张彩色图片其实就是一个三维数组（高度×宽度×颜色通道）。使用NumPy可以高效地处理图像数据。

**核心功能**：
- 图像读取和保存
- 灰度化处理
- 图像滤波（均值、高斯、中值）
- 边缘检测（Sobel、Prewitt）
- 图像增强（对比度、亮度）
- 图像旋转和缩放

**技术要点**：
- 使用PIL读取图像转为NumPy数组
- 卷积操作实现滤波
- 梯度计算实现边缘检测
- 矩阵变换实现几何变换

```python
import numpy as np
from PIL import Image
import matplotlib.pyplot as plt

class ImageProcessor:
    """图像处理工具类"""

    def __init__(self, image_path=None):
        """
        初始化图像处理器

        参数：
            image_path: 图像文件路径
        """
        self.image = None
        self.original = None

        if image_path:
            self.load_image(image_path)

    def load_image(self, image_path):
        """
        加载图像

        参数：
            image_path: 图像文件路径
        """
        try:
            # 使用PIL读取图像
            img = Image.open(image_path)
            # 转换为NumPy数组
            self.image = np.array(img)
            # 保存原始图像
            self.original = self.image.copy()
            print(f"图像加载成功: {self.image.shape}")
            return True
        except Exception as e:
            print(f"图像加载失败: {e}")
            return False

    def save_image(self, save_path, image=None):
        """
        保存图像

        参数：
            save_path: 保存路径
            image: 要保存的图像数组，默认为当前图像
        """
        if image is None:
            image = self.image

        try:
            # 确保数据类型正确
            if image.dtype != np.uint8:
                image = np.clip(image, 0, 255).astype(np.uint8)

            # 转换为PIL图像并保存
            img = Image.fromarray(image)
            img.save(save_path)
            print(f"图像已保存: {save_path}")
            return True
        except Exception as e:
            print(f"图像保存失败: {e}")
            return False

    def reset(self):
        """重置为原始图像"""
        if self.original is not None:
            self.image = self.original.copy()
            print("已重置为原始图像")

    def to_grayscale(self, method='average'):
        """
        转换为灰度图

        参数：
            method: 转换方法
                - 'average': 平均值法
                - 'weighted': 加权法（更符合人眼感知）
                - 'max': 最大值法
                - 'min': 最小值法

        返回：
            灰度图像数组
        """
        if len(self.image.shape) == 2:
            print("图像已经是灰度图")
            return self.image

        if method == 'average':
            # 简单平均
            gray = np.mean(self.image, axis=2)

        elif method == 'weighted':
            # 加权平均（符合人眼对RGB的敏感度）
            # 人眼对绿色最敏感，红色次之，蓝色最不敏感
            weights = np.array([0.299, 0.587, 0.114])
            gray = np.dot(self.image, weights)

        elif method == 'max':
            # 最大值法
            gray = np.max(self.image, axis=2)

        elif method == 'min':
            # 最小值法
            gray = np.min(self.image, axis=2)

        else:
            raise ValueError(f"未知的方法: {method}")

        # 转换为uint8类型
        self.image = gray.astype(np.uint8)
        print(f"灰度化完成 (方法: {method})")
        return self.image

    def apply_filter(self, filter_type='mean', kernel_size=3):
        """
        应用滤波器

        参数：
            filter_type: 滤波器类型
                - 'mean': 均值滤波（去噪、模糊）
                - 'gaussian': 高斯滤波（去噪、保边缘）
                - 'median': 中值滤波（去椒盐噪声）
            kernel_size: 滤波器大小（奇数）

        返回：
            滤波后的图像
        """
        if kernel_size % 2 == 0:
            raise ValueError("kernel_size必须是奇数")

        # 如果是彩色图像，分别处理每个通道
        if len(self.image.shape) == 3:
            result = np.zeros_like(self.image)
            for c in range(self.image.shape[2]):
                result[:, :, c] = self._apply_filter_2d(
                    self.image[:, :, c], filter_type, kernel_size
                )
        else:
            result = self._apply_filter_2d(self.image, filter_type, kernel_size)

        self.image = result
        print(f"{filter_type}滤波完成 (kernel_size={kernel_size})")
        return self.image

    def _apply_filter_2d(self, image, filter_type, kernel_size):
        """对2D图像应用滤波器"""
        pad = kernel_size // 2
        # 填充边界
        padded = np.pad(image, pad, mode='edge')
        result = np.zeros_like(image)

        h, w = image.shape

        if filter_type == 'mean':
            # 均值滤波：用邻域平均值替换
            for i in range(h):
                for j in range(w):
                    window = padded[i:i+kernel_size, j:j+kernel_size]
                    result[i, j] = np.mean(window)

        elif filter_type == 'gaussian':
            # 高斯滤波：用高斯加权平均
            kernel = self._gaussian_kernel(kernel_size)
            for i in range(h):
                for j in range(w):
                    window = padded[i:i+kernel_size, j:j+kernel_size]
                    result[i, j] = np.sum(window * kernel)

        elif filter_type == 'median':
            # 中值滤波：用邻域中值替换
            for i in range(h):
                for j in range(w):
                    window = padded[i:i+kernel_size, j:j+kernel_size]
                    result[i, j] = np.median(window)

        else:
            raise ValueError(f"未知的滤波器类型: {filter_type}")

        return result.astype(np.uint8)

    def _gaussian_kernel(self, size, sigma=None):
        """
        生成高斯核

        参数：
            size: 核大小
            sigma: 标准差，默认为size/6
        """
        if sigma is None:
            sigma = size / 6

        # 创建坐标网格
        ax = np.arange(-size // 2 + 1, size // 2 + 1)
        xx, yy = np.meshgrid(ax, ax)

        # 计算高斯函数
        kernel = np.exp(-(xx**2 + yy**2) / (2 * sigma**2))

        # 归一化
        return kernel / np.sum(kernel)

    def detect_edges(self, method='sobel', threshold=None):
        """
        边缘检测

        参数：
            method: 检测方法
                - 'sobel': Sobel算子
                - 'prewitt': Prewitt算子
                - 'roberts': Roberts算子
            threshold: 阈值，低于此值的边缘会被过滤

        返回：
            边缘图像
        """
        # 如果是彩色图，先转灰度
        if len(self.image.shape) == 3:
            image = np.mean(self.image, axis=2)
        else:
            image = self.image.astype(float)

        if method == 'sobel':
            # Sobel算子
            kernel_x = np.array([[-1, 0, 1],
                                [-2, 0, 2],
                                [-1, 0, 1]])
            kernel_y = np.array([[-1, -2, -1],
                                [0, 0, 0],
                                [1, 2, 1]])

        elif method == 'prewitt':
            # Prewitt算子
            kernel_x = np.array([[-1, 0, 1],
                                [-1, 0, 1],
                                [-1, 0, 1]])
            kernel_y = np.array([[-1, -1, -1],
                                [0, 0, 0],
                                [1, 1, 1]])

        elif method == 'roberts':
            # Roberts算子
            kernel_x = np.array([[1, 0],
                                [0, -1]])
            kernel_y = np.array([[0, 1],
                                [-1, 0]])

        else:
            raise ValueError(f"未知的边缘检测方法: {method}")

        # 应用卷积
        grad_x = self._convolve2d(image, kernel_x)
        grad_y = self._convolve2d(image, kernel_y)

        # 计算梯度幅值
        gradient = np.sqrt(grad_x**2 + grad_y**2)

        # 归一化到0-255
        gradient = (gradient / gradient.max() * 255).astype(np.uint8)

        # 应用阈值
        if threshold is not None:
            gradient = np.where(gradient > threshold, 255, 0).astype(np.uint8)

        self.image = gradient
        print(f"{method}边缘检测完成")
        return self.image

    def _convolve2d(self, image, kernel):
        """
        2D卷积

        参数：
            image: 输入图像
            kernel: 卷积核
        """
        k_h, k_w = kernel.shape
        pad_h, pad_w = k_h // 2, k_w // 2

        # 填充
        padded = np.pad(image, ((pad_h, pad_h), (pad_w, pad_w)), mode='edge')

        h, w = image.shape
        result = np.zeros_like(image)

        # 卷积操作
        for i in range(h):
            for j in range(w):
                window = padded[i:i+k_h, j:j+k_w]
                result[i, j] = np.sum(window * kernel)

        return result

    def adjust_brightness(self, factor):
        """
        调整亮度

        参数：
            factor: 亮度因子
                > 1: 变亮
                < 1: 变暗
                = 1: 不变
        """
        adjusted = self.image * factor
        self.image = np.clip(adjusted, 0, 255).astype(np.uint8)
        print(f"亮度调整完成 (factor={factor})")
        return self.image

    def adjust_contrast(self, factor):
        """
        调整对比度

        参数：
            factor: 对比度因子
                > 1: 增强对比度
                < 1: 降低对比度
                = 1: 不变
        """
        # 计算图像均值
        mean = np.mean(self.image)

        # 调整对比度：(像素值 - 均值) * 因子 + 均值
        adjusted = (self.image - mean) * factor + mean
        self.image = np.clip(adjusted, 0, 255).astype(np.uint8)
        print(f"对比度调整完成 (factor={factor})")
        return self.image

    def add_noise(self, noise_type='gaussian', amount=0.1):
        """
        添加噪声

        参数：
            noise_type: 噪声类型
                - 'gaussian': 高斯噪声
                - 'salt_pepper': 椒盐噪声
            amount: 噪声强度
        """
        if noise_type == 'gaussian':
            # 高斯噪声
            noise = np.random.normal(0, amount * 255, self.image.shape)
            noisy = self.image + noise

        elif noise_type == 'salt_pepper':
            # 椒盐噪声
            noisy = self.image.copy()
            # 盐噪声（白点）
            salt_mask = np.random.random(self.image.shape) < amount / 2
            noisy[salt_mask] = 255
            # 椒噪声（黑点）
            pepper_mask = np.random.random(self.image.shape) < amount / 2
            noisy[pepper_mask] = 0

        else:
            raise ValueError(f"未知的噪声类型: {noise_type}")

        self.image = np.clip(noisy, 0, 255).astype(np.uint8)
        print(f"添加{noise_type}噪声完成 (amount={amount})")
        return self.image

    def rotate(self, angle):
        """
        旋转图像

        参数：
            angle: 旋转角度（度）
        """
        # 转换为弧度
        theta = np.radians(angle)

        h, w = self.image.shape[:2]

        # 旋转矩阵
        cos_theta = np.cos(theta)
        sin_theta = np.sin(theta)

        # 计算新图像大小
        new_h = int(abs(h * cos_theta) + abs(w * sin_theta))
        new_w = int(abs(w * cos_theta) + abs(h * sin_theta))

        # 中心点
        center_old = np.array([h / 2, w / 2])
        center_new = np.array([new_h / 2, new_w / 2])

        # 创建新图像
        if len(self.image.shape) == 3:
            rotated = np.zeros((new_h, new_w, self.image.shape[2]), dtype=np.uint8)
        else:
            rotated = np.zeros((new_h, new_w), dtype=np.uint8)

        # 反向映射（避免空洞）
        for i in range(new_h):
            for j in range(new_w):
                # 新坐标相对于中心
                y = i - center_new[0]
                x = j - center_new[1]

                # 反向旋转到原图坐标
                old_y = y * cos_theta + x * sin_theta + center_old[0]
                old_x = -y * sin_theta + x * cos_theta + center_old[1]

                # 检查是否在原图范围内
                if 0 <= old_y < h and 0 <= old_x < w:
                    # 最近邻插值
                    rotated[i, j] = self.image[int(old_y), int(old_x)]

        self.image = rotated
        print(f"旋转完成 (angle={angle}°)")
        return self.image

    def resize(self, new_height, new_width):
        """
        调整图像大小

        参数：
            new_height: 新高度
            new_width: 新宽度
        """
        h, w = self.image.shape[:2]

        # 计算缩放比例
        scale_h = h / new_height
        scale_w = w / new_width

        # 创建新图像
        if len(self.image.shape) == 3:
            resized = np.zeros((new_height, new_width, self.image.shape[2]), dtype=np.uint8)
        else:
            resized = np.zeros((new_height, new_width), dtype=np.uint8)

        # 最近邻插值
        for i in range(new_height):
            for j in range(new_width):
                # 映射到原图坐标
                old_i = int(i * scale_h)
                old_j = int(j * scale_w)

                # 边界检查
                old_i = min(old_i, h - 1)
                old_j = min(old_j, w - 1)

                resized[i, j] = self.image[old_i, old_j]

        self.image = resized
        print(f"大小调整完成 ({h}x{w} -> {new_height}x{new_width})")
        return self.image

    def histogram_equalization(self):
        """
        直方图均衡化（增强对比度）
        """
        # 如果是彩色图像，分别处理每个通道
        if len(self.image.shape) == 3:
            result = np.zeros_like(self.image)
            for c in range(self.image.shape[2]):
                result[:, :, c] = self._equalize_channel(self.image[:, :, c])
        else:
            result = self._equalize_channel(self.image)

        self.image = result
        print("直方图均衡化完成")
        return self.image

    def _equalize_channel(self, channel):
        """对单个通道进行直方图均衡化"""
        # 计算直方图
        hist, bins = np.histogram(channel.flatten(), 256, [0, 256])

        # 计算累积分布函数（CDF）
        cdf = hist.cumsum()

        # 归一化CDF
        cdf_normalized = cdf * 255 / cdf[-1]

        # 使用CDF进行映射
        equalized = np.interp(channel.flatten(), bins[:-1], cdf_normalized)

        return equalized.reshape(channel.shape).astype(np.uint8)

    def show(self, title='Image'):
        """
        显示图像

        参数：
            title: 窗口标题
        """
        plt.figure(figsize=(10, 8))

        if len(self.image.shape) == 2:
            # 灰度图
            plt.imshow(self.image, cmap='gray')
        else:
            # 彩色图
            plt.imshow(self.image)

        plt.title(title)
        plt.axis('off')
        plt.tight_layout()
        plt.show()

    def compare(self, other_image, titles=('Original', 'Processed')):
        """
        对比显示两张图像

        参数：
            other_image: 另一张图像
            titles: 图像标题
        """
        fig, axes = plt.subplots(1, 2, figsize=(15, 6))

        # 显示第一张图
        if len(self.original.shape) == 2:
            axes[0].imshow(self.original, cmap='gray')
        else:
            axes[0].imshow(self.original)
        axes[0].set_title(titles[0])
        axes[0].axis('off')

        # 显示第二张图
        if len(other_image.shape) == 2:
            axes[1].imshow(other_image, cmap='gray')
        else:
            axes[1].imshow(other_image)
        axes[1].set_title(titles[1])
        axes[1].axis('off')

        plt.tight_layout()
        plt.show()

    def show_histogram(self):
        """显示图像直方图"""
        plt.figure(figsize=(12, 4))

        if len(self.image.shape) == 2:
            # 灰度图直方图
            plt.hist(self.image.ravel(), bins=256, range=[0, 256], color='gray', alpha=0.7)
            plt.title('Grayscale Histogram')
            plt.xlabel('Pixel Value')
            plt.ylabel('Frequency')
        else:
            # 彩色图RGB直方图
            colors = ('red', 'green', 'blue')
            for i, color in enumerate(colors):
                plt.hist(self.image[:, :, i].ravel(), bins=256, range=[0, 256],
                        color=color, alpha=0.5, label=color.upper())
            plt.title('RGB Histogram')
            plt.xlabel('Pixel Value')
            plt.ylabel('Frequency')
            plt.legend()

        plt.tight_layout()
        plt.show()
```

```python
    def find_outliers(self, method='iqr', threshold=1.5):
        """
        检测异常值

        参数：
            method: 检测方法
                - 'iqr': 四分位距法
                - 'zscore': Z分数法
            threshold: 阈值
                - iqr方法: IQR倍数（默认1.5）
                - zscore方法: 标准差倍数（默认3）
        """
        print("\n" + "=" * 80)
        print(f"异常值检测 (方法: {method})".center(80))
        print("=" * 80)

        outliers = []

        for subject_idx, subject in enumerate(self.subjects):
            subject_grades = self.grades[:, subject_idx]

            if method == 'iqr':
                # 四分位距法
                q1 = np.percentile(subject_grades, 25)
                q3 = np.percentile(subject_grades, 75)
                iqr = q3 - q1

                lower_bound = q1 - threshold * iqr
                upper_bound = q3 + threshold * iqr

                # 找出异常值
                outlier_mask = (subject_grades < lower_bound) | (subject_grades > upper_bound)

            elif method == 'zscore':
                # Z分数法
                mean = np.mean(subject_grades)
                std = np.std(subject_grades)
                z_scores = np.abs((subject_grades - mean) / std)

                outlier_mask = z_scores > threshold
                lower_bound = mean - threshold * std
                upper_bound = mean + threshold * std

            else:
                raise ValueError(f"未知的方法: {method}")

            # 收集异常值
            outlier_indices = np.where(outlier_mask)[0]

            if len(outlier_indices) > 0:
                print(f"\n{subject}:")
                print(f"  正常范围: [{lower_bound:.2f}, {upper_bound:.2f}]")
                print(f"  异常值数量: {len(outlier_indices)}")
                print(f"  异常学生:")

                for idx in outlier_indices:
                    print(f"    - {self.students[idx]}: {subject_grades[idx]:.2f}分")
                    outliers.append((self.students[idx], subject, subject_grades[idx]))

        if not outliers:
            print("\n未发现异常值")

        print("=" * 80)

        return outliers
```

## 项目3：简单神经网络

从零开始实现一个简单的神经网络，理解深度学习的基本原理。

### 需求分析

**项目背景**：

神经网络是机器学习的核心算法。虽然现在有TensorFlow、PyTorch等强大的库，但从零实现一个简单的神经网络能帮助我们深入理解其工作原理。NumPy的矩阵运算能力使这个过程变得简单。

**核心功能**：
- 前向传播（计算输出）
- 反向传播（计算梯度）
- 梯度下降（更新权重）
- 激活函数（Sigmoid、ReLU、Softmax）
- 损失函数（交叉熵、均方误差）
- 训练和预测

**技术要点**：
- 矩阵乘法实现层间传播
- 链式法则计算梯度
- NumPy广播机制
- 批量处理数据

```python
import numpy as np
import matplotlib.pyplot as plt

class NeuralNetwork:
    """简单的神经网络实现"""

    def __init__(self, layer_sizes, activation='relu', loss='cross_entropy'):
        """
        初始化神经网络

        参数：
            layer_sizes: 各层神经元数量列表，例如 [784, 128, 64, 10]
            activation: 激活函数 ('relu', 'sigmoid', 'tanh')
            loss: 损失函数 ('cross_entropy', 'mse')
        """
        self.layer_sizes = layer_sizes
        self.num_layers = len(layer_sizes)
        self.activation_name = activation
        self.loss_name = loss

        # 初始化权重和偏置
        self.weights = []
        self.biases = []

        # He初始化（对ReLU友好）
        for i in range(self.num_layers - 1):
            w = np.random.randn(layer_sizes[i], layer_sizes[i+1]) * np.sqrt(2.0 / layer_sizes[i])
            b = np.zeros((1, layer_sizes[i+1]))
            self.weights.append(w)
            self.biases.append(b)

        # 训练历史
        self.train_loss = []
        self.train_acc = []
        self.val_loss = []
        self.val_acc = []

        print(f"神经网络初始化完成")
        print(f"结构: {' -> '.join(map(str, layer_sizes))}")
        print(f"激活函数: {activation}")
        print(f"损失函数: {loss}")

    def _sigmoid(self, z):
        """Sigmoid激活函数"""
        return 1 / (1 + np.exp(-np.clip(z, -500, 500)))

    def _sigmoid_derivative(self, a):
        """Sigmoid导数"""
        return a * (1 - a)

    def _relu(self, z):
        """ReLU激活函数"""
        return np.maximum(0, z)

    def _relu_derivative(self, z):
        """ReLU导数"""
        return (z > 0).astype(float)

    def _tanh(self, z):
        """Tanh激活函数"""
        return np.tanh(z)

    def _tanh_derivative(self, a):
        """Tanh导数"""
        return 1 - a**2

    def _softmax(self, z):
        """Softmax激活函数（用于输出层）"""
        # 减去最大值，防止指数溢出
        exp_z = np.exp(z - np.max(z, axis=1, keepdims=True))
        return exp_z / np.sum(exp_z, axis=1, keepdims=True)

    def _activate(self, z, derivative=False):
        """应用激活函数"""
        if self.activation_name == 'sigmoid':
            if derivative:
                a = self._sigmoid(z)
                return self._sigmoid_derivative(a)
            return self._sigmoid(z)

        elif self.activation_name == 'relu':
            if derivative:
                return self._relu_derivative(z)
            return self._relu(z)

        elif self.activation_name == 'tanh':
            if derivative:
                a = self._tanh(z)
                return self._tanh_derivative(a)
            return self._tanh(z)

    def forward(self, X):
        """
        前向传播

        参数：
            X: 输入数据 (batch_size, input_dim)

        返回：
            输出和中间结果
        """
        activations = [X]
        zs = []

        # 隐藏层
        for i in range(self.num_layers - 2):
            z = np.dot(activations[-1], self.weights[i]) + self.biases[i]
            zs.append(z)
            a = self._activate(z)
            activations.append(a)

        # 输出层（使用softmax）
        z = np.dot(activations[-1], self.weights[-1]) + self.biases[-1]
        zs.append(z)
        a = self._softmax(z)
        activations.append(a)

        return activations, zs

    def backward(self, X, y, activations, zs, learning_rate):
        """
        反向传播

        参数：
            X: 输入数据
            y: 真实标签（one-hot编码）
            activations: 前向传播的激活值
            zs: 前向传播的线性组合
            learning_rate: 学习率
        """
        m = X.shape[0]  # 批量大小

        # 计算输出层误差
        # 对于softmax + cross_entropy，导数简化为 (y_pred - y)
        delta = activations[-1] - y

        # 反向传播误差
        for i in range(self.num_layers - 2, -1, -1):
            # 计算梯度
            dW = np.dot(activations[i].T, delta) / m
            db = np.sum(delta, axis=0, keepdims=True) / m

            # 更新权重和偏置
            self.weights[i] -= learning_rate * dW
            self.biases[i] -= learning_rate * db

            # 传播误差到前一层
            if i > 0:
                delta = np.dot(delta, self.weights[i].T) * self._activate(zs[i-1], derivative=True)

    def compute_loss(self, y_pred, y_true):
        """计算损失"""
        m = y_true.shape[0]

        if self.loss_name == 'cross_entropy':
            # 交叉熵损失
            # 添加小的epsilon避免log(0)
            epsilon = 1e-8
            loss = -np.sum(y_true * np.log(y_pred + epsilon)) / m

        elif self.loss_name == 'mse':
            # 均方误差
            loss = np.mean((y_pred - y_true)**2)

        return loss

    def train(self, X_train, y_train, X_val=None, y_val=None,
              epochs=100, batch_size=32, learning_rate=0.01, verbose=True):
        """
        训练神经网络

        参数：
            X_train: 训练数据
            y_train: 训练标签（one-hot编码）
            X_val: 验证数据
            y_val: 验证标签
            epochs: 训练轮数
            batch_size: 批量大小
            learning_rate: 学习率
            verbose: 是否打印训练过程
        """
        n_samples = X_train.shape[0]
        n_batches = (n_samples + batch_size - 1) // batch_size

        print(f"\n开始训练...")
        print(f"训练样本: {n_samples}")
        print(f"批量大小: {batch_size}")
        print(f"学习率: {learning_rate}")
        print(f"训练轮数: {epochs}")
        print("=" * 70)

        for epoch in range(epochs):
            # 打乱数据
            indices = np.random.permutation(n_samples)
            X_shuffled = X_train[indices]
            y_shuffled = y_train[indices]

            # 小批量训练
            epoch_loss = 0
            for i in range(n_batches):
                start = i * batch_size
                end = min(start + batch_size, n_samples)

                X_batch = X_shuffled[start:end]
                y_batch = y_shuffled[start:end]

                # 前向传播
                activations, zs = self.forward(X_batch)

                # 计算损失
                loss = self.compute_loss(activations[-1], y_batch)
                epoch_loss += loss

                # 反向传播
                self.backward(X_batch, y_batch, activations, zs, learning_rate)

            # 平均损失
            epoch_loss /= n_batches
            self.train_loss.append(epoch_loss)

            # 计算训练准确率
            train_acc = self.evaluate(X_train, y_train)
            self.train_acc.append(train_acc)

            # 计算验证准确率
            if X_val is not None and y_val is not None:
                activations_val, _ = self.forward(X_val)
                val_loss = self.compute_loss(activations_val[-1], y_val)
                val_acc = self.evaluate(X_val, y_val)
                self.val_loss.append(val_loss)
                self.val_acc.append(val_acc)

            # 打印进度
            if verbose and (epoch + 1) % max(1, epochs // 10) == 0:
                info = f"Epoch {epoch+1}/{epochs} - Loss: {epoch_loss:.4f} - Acc: {train_acc:.4f}"
                if X_val is not None:
                    info += f" - Val Loss: {val_loss:.4f} - Val Acc: {val_acc:.4f}"
                print(info)

        print("=" * 70)
        print("训练完成！")

    def predict(self, X):
        """
        预测

        参数：
            X: 输入数据

        返回：
            预测的类别
        """
        activations, _ = self.forward(X)
        y_pred = activations[-1]
        return np.argmax(y_pred, axis=1)

    def predict_proba(self, X):
        """
        预测概率

        参数：
            X: 输入数据

        返回：
            各类别的概率
        """
        activations, _ = self.forward(X)
        return activations[-1]

    def evaluate(self, X, y):
        """
        评估准确率

        参数：
            X: 输入数据
            y: 真实标签（one-hot编码）

        返回：
            准确率
        """
        y_pred = self.predict(X)
        y_true = np.argmax(y, axis=1)
        accuracy = np.mean(y_pred == y_true)
        return accuracy

    def plot_training_history(self):
        """绘制训练历史"""
        fig, axes = plt.subplots(1, 2, figsize=(14, 5))

        # 损失曲线
        ax1 = axes[0]
        ax1.plot(self.train_loss, label='Training Loss', linewidth=2)
        if self.val_loss:
            ax1.plot(self.val_loss, label='Validation Loss', linewidth=2)
        ax1.set_xlabel('Epoch')
        ax1.set_ylabel('Loss')
        ax1.set_title('Training and Validation Loss')
        ax1.legend()
        ax1.grid(True, alpha=0.3)

        # 准确率曲线
        ax2 = axes[1]
        ax2.plot(self.train_acc, label='Training Accuracy', linewidth=2)
        if self.val_acc:
            ax2.plot(self.val_acc, label='Validation Accuracy', linewidth=2)
        ax2.set_xlabel('Epoch')
        ax2.set_ylabel('Accuracy')
        ax2.set_title('Training and Validation Accuracy')
        ax2.legend()
        ax2.grid(True, alpha=0.3)

        plt.tight_layout()
        plt.show()

    def plot_predictions(self, X, y, num_samples=10):
        """可视化预测结果（用于图像数据）"""
        # 随机选择样本
        indices = np.random.choice(len(X), num_samples, replace=False)

        # 预测
        y_pred = self.predict(X[indices])
        y_true = np.argmax(y[indices], axis=1)

        # 绘制
        fig, axes = plt.subplots(2, 5, figsize=(12, 5))
        axes = axes.flatten()

        for i, idx in enumerate(indices):
            # 重塑为图像（假设是28x28的图像）
            img_size = int(np.sqrt(X.shape[1]))
            img = X[idx].reshape(img_size, img_size)

            axes[i].imshow(img, cmap='gray')
            axes[i].set_title(f'True: {y_true[i]}, Pred: {y_pred[i]}')
            axes[i].axis('off')

            # 标记正确/错误
            if y_true[i] == y_pred[i]:
```

## 项目4：时间序列分析工具

一个完整的时间序列分析系统，支持趋势分析、预测等功能。

### 需求分析

**项目背景**：

时间序列数据无处不在：股票价格、气温变化、销售额等。时间序列分析帮助我们理解数据的趋势和周期性，并做出预测。NumPy的数组操作使时间序列处理变得高效。

**核心功能**：
- 移动平均（简单移动平均、加权移动平均、指数移动平均）
- 趋势分解（趋势、季节性、随机性）
- 平滑处理
- 简单预测
- 异常检测
- 可视化分析

**技术要点**：
- 滑动窗口操作
- 卷积实现移动平均
- 傅里叶变换分析周期性
- 线性回归拟合趋势

### 完整实现

```python
import numpy as np
import matplotlib.pyplot as plt
from datetime import datetime, timedelta

class TimeSeriesAnalyzer:
    """时间序列分析工具"""

    def __init__(self, data, dates=None):
        """
        初始化分析器

        参数：
            data: 时间序列数据（一维数组）
            dates: 日期数组（可选）
        """
        self.data = np.array(data, dtype=float)

        if dates is not None:
            self.dates = dates
        else:
            # 如果没有提供日期，使用索引
            self.dates = np.arange(len(data))

        print(f"时间序列加载成功")
        print(f"数据点数: {len(self.data)}")
        print(f"数据范围: [{np.min(self.data):.2f}, {np.max(self.data):.2f}]")

    def moving_average(self, window_size, method='simple'):
        """
        移动平均

        参数：
            window_size: 窗口大小
            method: 方法
                - 'simple': 简单移动平均
                - 'weighted': 加权移动平均
                - 'exponential': 指数移动平均

        返回：
            平滑后的数据
        """
        n = len(self.data)

        if method == 'simple':
            # 简单移动平均：等权重
            result = np.convolve(self.data, np.ones(window_size)/window_size, mode='valid')
            # 填充开头
            pad_size = window_size - 1
            result = np.concatenate([np.full(pad_size, np.nan), result])

        elif method == 'weighted':
            # 加权移动平均：线性递增权重
            weights = np.arange(1, window_size + 1)
            weights = weights / np.sum(weights)
            result = np.convolve(self.data, weights, mode='valid')
            pad_size = window_size - 1
            result = np.concatenate([np.full(pad_size, np.nan), result])

        elif method == 'exponential':
            # 指数移动平均：EMA
            alpha = 2 / (window_size + 1)
            result = np.zeros(n)
            result[0] = self.data[0]

            for i in range(1, n):
                result[i] = alpha * self.data[i] + (1 - alpha) * result[i-1]

        else:
            raise ValueError(f"未知的方法: {method}")

        print(f"{method}移动平均完成 (窗口大小={window_size})")
        return result

    def detect_trend(self, method='linear'):
        """
        检测趋势

        参数：
            method: 方法
                - 'linear': 线性回归
                - 'polynomial': 多项式拟合

        返回：
            趋势线
        """
        x = np.arange(len(self.data))

        if method == 'linear':
            # 线性回归: y = ax + b
            # 使用最小二乘法
            A = np.vstack([x, np.ones(len(x))]).T
            coeffs = np.linalg.lstsq(A, self.data, rcond=None)[0]
            trend = coeffs[0] * x + coeffs[1]

            # 判断趋势方向
            if coeffs[0] > 0.01:
                direction = "上升"
            elif coeffs[0] < -0.01:
                direction = "下降"
            else:
                direction = "平稳"

            print(f"线性趋势: y = {coeffs[0]:.4f}x + {coeffs[1]:.4f}")
            print(f"趋势方向: {direction}")

        elif method == 'polynomial':
            # 多项式拟合
            degree = 3
            coeffs = np.polyfit(x, self.data, degree)
            trend = np.polyval(coeffs, x)

            print(f"多项式趋势 (度数={degree})")

        else:
            raise ValueError(f"未知的方法: {method}")

        return trend

    def decompose(self, period):
        """
        时间序列分解

        将时间序列分解为：趋势 + 季节性 + 残差

        参数：
            period: 季节性周期

        返回：
            (trend, seasonal, residual)
        """
        n = len(self.data)

        # 1. 提取趋势（使用移动平均）
        trend = self.moving_average(period, method='simple')

        # 2. 去趋势
        detrended = self.data - trend

        # 3. 计算季节性成分
        # 将数据重塑为(周期数, 周期长度)
        n_periods = n // period
        detrended_reshaped = detrended[:n_periods * period].reshape(n_periods, period)

        # 每个季节位置的平均值
        seasonal_pattern = np.nanmean(detrended_reshaped, axis=0)

        # 扩展到完整长度
        seasonal = np.tile(seasonal_pattern, n_periods + 1)[:n]

        # 4. 计算残差
        residual = self.data - trend - seasonal

        print(f"时间序列分解完成 (周期={period})")
        print(f"趋势强度: {np.nanstd(trend):.4f}")
        print(f"季节性强度: {np.std(seasonal):.4f}")
        print(f"随机性强度: {np.nanstd(residual):.4f}")

        return trend, seasonal, residual

    def detect_seasonality(self):
        """
        检测季节性（使用FFT）

        返回：
            主要周期
        """
        # 去除NaN
        data_clean = self.data[~np.isnan(self.data)]

        # FFT
        fft = np.fft.fft(data_clean)
        frequencies = np.fft.fftfreq(len(data_clean))

        # 只看正频率
        positive_freq_idx = frequencies > 0
        frequencies = frequencies[positive_freq_idx]
        power = np.abs(fft[positive_freq_idx])**2

        # 找出最强的频率
        top_indices = np.argsort(power)[::-1][:3]

        print("\n检测到的主要周期:")
        for i, idx in enumerate(top_indices, 1):
            period = 1 / frequencies[idx] if frequencies[idx] != 0 else np.inf
            print(f"  {i}. 周期: {period:.2f}, 强度: {power[idx]:.2f}")

        # 返回最强的周期
        main_period = 1 / frequencies[top_indices[0]]
        return main_period

    def forecast_simple(self, n_ahead, method='last_value'):
        """
        简单预测

        参数：
            n_ahead: 预测未来多少个点
            method: 预测方法
                - 'last_value': 使用最后一个值
                - 'mean': 使用历史平均值
                - 'linear': 线性外推

        返回：
            预测值
        """
        if method == 'last_value':
            # 最后一个值
            forecast = np.full(n_ahead, self.data[-1])

        elif method == 'mean':
            # 历史平均值
            forecast = np.full(n_ahead, np.mean(self.data))

        elif method == 'linear':
            # 线性外推
            x = np.arange(len(self.data))
            A = np.vstack([x, np.ones(len(x))]).T
            coeffs = np.linalg.lstsq(A, self.data, rcond=None)[0]

            # 预测
            x_future = np.arange(len(self.data), len(self.data) + n_ahead)
            forecast = coeffs[0] * x_future + coeffs[1]

        else:
            raise ValueError(f"未知的方法: {method}")

        print(f"预测完成: {n_ahead}个未来点 (方法={method})")
        return forecast

    def detect_anomalies(self, method='zscore', threshold=3):
        """
        异常检测

        参数：
            method: 检测方法
                - 'zscore': Z分数法
                - 'iqr': 四分位距法
                - 'moving_avg': 移动平均法
            threshold: 阈值

        返回：
            异常点的索引
        """
        if method == 'zscore':
            # Z分数法
            mean = np.mean(self.data)
            std = np.std(self.data)
            z_scores = np.abs((self.data - mean) / std)
            anomalies = np.where(z_scores > threshold)[0]

        elif method == 'iqr':
            # 四分位距法
            q1 = np.percentile(self.data, 25)
            q3 = np.percentile(self.data, 75)
            iqr = q3 - q1

            lower_bound = q1 - threshold * iqr
            upper_bound = q3 + threshold * iqr

            anomalies = np.where((self.data < lower_bound) | (self.data > upper_bound))[0]

        elif method == 'moving_avg':
            # 移动平均法
            window = 10
            ma = self.moving_average(window, 'simple')
            diff = np.abs(self.data - ma)

            # 使用差值的标准差作为阈值
            threshold_value = threshold * np.nanstd(diff)
            anomalies = np.where(diff > threshold_value)[0]

        else:
            raise ValueError(f"未知的方法: {method}")

        print(f"异常检测完成: 发现{len(anomalies)}个异常点 (方法={method})")

        if len(anomalies) > 0:
            print(f"异常点位置: {anomalies[:10]}...")  # 只显示前10个

        return anomalies

    def calculate_statistics(self):
        """计算统计指标"""
        print("\n" + "=" * 60)
        print("时间序列统计".center(60))
        print("=" * 60)

        print(f"均值: {np.mean(self.data):.4f}")
        print(f"中位数: {np.median(self.data):.4f}")
        print(f"标准差: {np.std(self.data):.4f}")
        print(f"最小值: {np.min(self.data):.4f}")
        print(f"最大值: {np.max(self.data):.4f}")
        print(f"范围: {np.ptp(self.data):.4f}")

        # 变异系数
        cv = np.std(self.data) / np.mean(self.data) * 100
        print(f"变异系数: {cv:.2f}%")

        # 自相关（lag-1）
        lag1_corr = np.corrcoef(self.data[:-1], self.data[1:])[0, 1]
        print(f"滞后1自相关: {lag1_corr:.4f}")

        print("=" * 60)

    def plot_analysis(self):

        fig = plt.figure(figsize=(16, 12))

def generate_sample_timeseries(n=200, trend=True, seasonal=True, noise=True):
    """生成样本时间序列"""
    np.random.seed(42)
    x = np.arange(n)

    # 基础值
    y = np.zeros(n)

    # 添加趋势
    if trend:
        y += 0.5 * x + 50

    # 添加季节性
    if seasonal:
        period = 12
        y += 10 * np.sin(2 * np.pi * x / period)

    # 添加噪声
    if noise:
        y += np.random.normal(0, 5, n)

    # 添加几个异常值
    anomaly_indices = [50, 100, 150]
    y[anomaly_indices] += np.random.choice([-30, 30], len(anomaly_indices))

    return y
```

```python
import numpy as np
import matplotlib.pyplot as plt

class ScientificComputing:
    """科学计算工具类"""

    @staticmethod
    def integrate_trapezoid(f, a, b, n=1000):
        """
        梯形法则数值积分

        ∫[a,b] f(x)dx ≈ h/2 * [f(x0) + 2f(x1) + 2f(x2) + ... + 2f(xn-1) + f(xn)]

        参数：
            f: 被积函数
            a, b: 积分区间
            n: 分段数

        返回：
            积分值
        """
        # 生成等间距点
        x = np.linspace(a, b, n+1)
        y = f(x)

        # 步长
        h = (b - a) / n

        # 梯形法则
        integral = h * (y[0]/2 + np.sum(y[1:-1]) + y[-1]/2)

        return integral

    @staticmethod
    def integrate_simpson(f, a, b, n=1000):
        """
        辛普森法则数值积分（更精确）

        参数：
            f: 被积函数
            a, b: 积分区间
            n: 分段数（必须是偶数）

        返回：
            积分值
        """
        if n % 2 != 0:
            n += 1  # 确保是偶数

        # 生成等间距点
        x = np.linspace(a, b, n+1)
        y = f(x)

        # 步长
        h = (b - a) / n

        # 辛普森法则
        # ∫ ≈ h/3 * [f(x0) + 4f(x1) + 2f(x2) + 4f(x3) + ... + f(xn)]
        integral = h/3 * (y[0] + 4*np.sum(y[1:-1:2]) + 2*np.sum(y[2:-1:2]) + y[-1])

        return integral

    @staticmethod
    def solve_bisection(f, a, b, tol=1e-6, max_iter=100):
        """
        二分法求解方程 f(x) = 0

        参数：
            f: 函数
            a, b: 区间 [a, b]，要求 f(a) 和 f(b) 异号
            tol: 容差
            max_iter: 最大迭代次数

        返回：
            (根, 迭代次数)
        """
        if f(a) * f(b) > 0:
            raise ValueError("f(a) 和 f(b) 必须异号")

        for i in range(max_iter):
            # 中点
            c = (a + b) / 2
            fc = f(c)

            # 检查收敛
            if abs(fc) < tol or abs(b - a) < tol:
                return c, i+1

            # 更新区间
            if f(a) * fc < 0:
                b = c
            else:
                a = c

        return (a + b) / 2, max_iter

    @staticmethod
    def solve_newton(f, df, x0, tol=1e-6, max_iter=100):
        """
        牛顿法求解方程 f(x) = 0

        x_{n+1} = x_n - f(x_n) / f'(x_n)

        参数：
            f: 函数
            df: 导数
            x0: 初始猜测
            tol: 容差
            max_iter: 最大迭代次数

        返回：
            (根, 迭代次数)
        """
        x = x0

        for i in range(max_iter):
            fx = f(x)

            # 检查收敛
            if abs(fx) < tol:
                return x, i+1

            # 牛顿迭代
            dfx = df(x)
            if abs(dfx) < 1e-12:
                raise ValueError("导数接近零，牛顿法失败")

            x = x - fx / dfx

        return x, max_iter

    @staticmethod
    def solve_secant(f, x0, x1, tol=1e-6, max_iter=100):
        """
        割线法求解方程 f(x) = 0（不需要导数）

        x_{n+1} = x_n - f(x_n) * (x_n - x_{n-1}) / (f(x_n) - f(x_{n-1}))

        参数：
            f: 函数
            x0, x1: 初始两个猜测
            tol: 容差
            max_iter: 最大迭代次数

        返回：
            (根, 迭代次数)
        """
        for i in range(max_iter):
            fx0 = f(x0)
            fx1 = f(x1)

            # 检查收敛
            if abs(fx1) < tol:
                return x1, i+1

            # 割线迭代
            if abs(fx1 - fx0) < 1e-12:
                raise ValueError("分母接近零，割线法失败")

            x_new = x1 - fx1 * (x1 - x0) / (fx1 - fx0)

            # 更新
            x0, x1 = x1, x_new

        return x1, max_iter

    @staticmethod
    def optimize_gradient_descent(f, grad_f, x0, learning_rate=0.01,
                                  tol=1e-6, max_iter=10000):
        """
        梯度下降优化

        参数：
            f: 目标函数
            grad_f: 梯度函数
            x0: 初始点
            learning_rate: 学习率
            tol: 容差
            max_iter: 最大迭代次数

        返回：
            (最优解, 最优值, 迭代历史)
        """
        x = np.array(x0, dtype=float)
        history = [x.copy()]

        for i in range(max_iter):
            # 计算梯度
            grad = grad_f(x)

            # 检查收敛
            if np.linalg.norm(grad) < tol:
                break

            # 更新
            x = x - learning_rate * grad
            history.append(x.copy())

        return x, f(x), np.array(history)

    @staticmethod
    def optimize_newton_method(f, grad_f, hess_f, x0, tol=1e-6, max_iter=100):
        """
        牛顿法优化

        x_{n+1} = x_n - H^{-1} * grad

        参数：
            f: 目标函数
            grad_f: 梯度函数
            hess_f: Hessian矩阵函数
            x0: 初始点
            tol: 容差
            max_iter: 最大迭代次数

        返回：
            (最优解, 最优值, 迭代历史)
        """
        x = np.array(x0, dtype=float)
        history = [x.copy()]

        for i in range(max_iter):
            grad = grad_f(x)

            # 检查收敛
            if np.linalg.norm(grad) < tol:
                break

            # 计算Hessian矩阵
            hess = hess_f(x)

            # 牛顿步: x = x - H^{-1} * grad
            try:
                delta = np.linalg.solve(hess, grad)
                x = x - delta
                history.append(x.copy())
            except np.linalg.LinAlgError:
                print("Hessian矩阵奇异，优化失败")
                break

        return x, f(x), np.array(history)

    @staticmethod
    def solve_linear_system(A, b, method='lu'):
        """
        求解线性方程组 Ax = b

        参数：
            A: 系数矩阵
            b: 右端向量
            method: 方法
                - 'lu': LU分解
                - 'qr': QR分解
                - 'cholesky': Cholesky分解（A必须正定）
                - 'direct': 直接求解

        返回：
            解 x
        """
        A = np.array(A, dtype=float)
        b = np.array(b, dtype=float)

        if method == 'lu':
            # LU分解
            from scipy.linalg import lu_factor, lu_solve
            lu, piv = lu_factor(A)
            x = lu_solve((lu, piv), b)

        elif method == 'qr':
            # QR分解
            Q, R = np.linalg.qr(A)
            x = np.linalg.solve(R, Q.T @ b)

        elif method == 'cholesky':
            # Cholesky分解（要求A正定）
            L = np.linalg.cholesky(A)
            # 前向替换: Ly = b
            y = np.linalg.solve(L, b)
            # 后向替换: L^T x = y
            x = np.linalg.solve(L.T, y)

        elif method == 'direct':
            # 直接求解
            x = np.linalg.solve(A, b)

        else:
            raise ValueError(f"未知的方法: {method}")

        return x

    @staticmethod
    def interpolate_lagrange(x_data, y_data, x_eval):
        """
        拉格朗日插值

        参数：
            x_data: 数据点x坐标
            y_data: 数据点y坐标
            x_eval: 插值点x坐标

        返回：
            插值结果
        """
        x_data = np.array(x_data)
        y_data = np.array(y_data)
        n = len(x_data)

        if np.isscalar(x_eval):
            x_eval = np.array([x_eval])
        else:
            x_eval = np.array(x_eval)

        # 计算拉格朗日基函数
        result = np.zeros_like(x_eval, dtype=float)

        for i in range(n):
            # L_i(x) = ∏_{j≠i} (x - x_j) / (x_i - x_j)
            Li = np.ones_like(x_eval, dtype=float)
            for j in range(n):
                if i != j:
                    Li *= (x_eval - x_data[j]) / (x_data[i] - x_data[j])

            result += y_data[i] * Li

        return result

    @staticmethod
    def fit_polynomial(x_data, y_data, degree):
        """
        多项式拟合

        参数：
            x_data: x坐标
            y_data: y坐标
            degree: 多项式次数

        返回：
            系数（从高次到低次）
        """
        coeffs = np.polyfit(x_data, y_data, degree)
        return coeffs

    @staticmethod
    def solve_ode_euler(f, y0, t_span, n_steps):
        """
        欧拉法求解常微分方程 dy/dt = f(t, y)

        参数：
            f: 导数函数 f(t, y)
            y0: 初始值
            t_span: 时间区间 (t0, tf)
            n_steps: 步数

        返回：
            (t, y)
        """
        t0, tf = t_span
        h = (tf - t0) / n_steps

        t = np.linspace(t0, tf, n_steps+1)
        y = np.zeros((n_steps+1, len(np.atleast_1d(y0))))
        y[0] = y0

        # 欧拉迭代: y_{n+1} = y_n + h * f(t_n, y_n)
        for i in range(n_steps):
            y[i+1] = y[i] + h * f(t[i], y[i])

        return t, y

    @staticmethod
    def solve_ode_rk4(f, y0, t_span, n_steps):
        """
        四阶龙格-库塔法求解常微分方程（更精确）

        参数：
            f: 导数函数 f(t, y)
            y0: 初始值
            t_span: 时间区间
            n_steps: 步数

        返回：
            (t, y)
        """
        t0, tf = t_span
        h = (tf - t0) / n_steps

        t = np.linspace(t0, tf, n_steps+1)
        y0 = np.atleast_1d(y0)
        y = np.zeros((n_steps+1, len(y0)))
        y[0] = y0

        # RK4迭代
        for i in range(n_steps):
            k1 = f(t[i], y[i])
            k2 = f(t[i] + h/2, y[i] + h/2*k1)
            k3 = f(t[i] + h/2, y[i] + h/2*k2)
            k4 = f(t[i] + h, y[i] + h*k3)

            y[i+1] = y[i] + h/6 * (k1 + 2*k2 + 2*k3 + k4)

        return t, y
```
