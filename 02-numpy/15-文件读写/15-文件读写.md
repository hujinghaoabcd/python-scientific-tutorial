# 第15章：NumPy文件读写

掌握NumPy数组的保存和加载，实现数据的持久化存储和高效传输。

## 为什么需要文件读写？

在数据分析和科学计算中，我们经常需要保存计算结果、分享数据或者从文件中加载数据。NumPy提供了多种文件读写方法，适用于不同的场景。

```python
import numpy as np

# 创建数组
data = np.array([[1, 2, 3], [4, 5, 6]])

# 程序关闭后数据消失
# 需要保存到文件中
np.save('data.npy', data)

# 下次可以加载
loaded_data = np.load('data.npy')
print(loaded_data)
```

**NumPy文件操作的优势**：
- **速度快** - 二进制格式读写速度远超文本格式
- **保持精度** - 不会损失浮点数精度
- **保留结构** - 自动保存数组形状和数据类型
- **节省空间** - 二进制格式更紧凑
- **方便共享** - .npy/.npz是通用的NumPy格式

## 二进制文件：save()和load()

### 基本用法

```python
import numpy as np

# 创建数组
arr = np.array([1, 2, 3, 4, 5])

# 保存为.npy文件
np.save('array.npy', arr)
print("数组已保存")

# 加载.npy文件
loaded = np.load('array.npy')
print(loaded)  # [1 2 3 4 5]
print(loaded.dtype)  # int32
```

**特点**：
- 自动添加`.npy`扩展名（如果没有）
- 保存数组的dtype、shape等元信息
- 读取速度非常快
- 仅支持NumPy格式

## 多数组保存：savez()和savez_compressed()

### savez() - 保存多个数组

```python
import numpy as np

# 创建多个数组
arr1 = np.array([1, 2, 3])
arr2 = np.array([[4, 5], [6, 7]])
arr3 = np.random.rand(2, 3)

# 保存多个数组到一个文件
np.savez('data.npz', arr1, arr2, arr3)

# 加载.npz文件
data = np.load('data.npz')
print(type(data))  # <class 'numpy.lib.npyio.NpzFile'>

# 通过索引访问
print(data['arr_0'])  # [1 2 3]
print(data['arr_1'])  # [[4 5] [6 7]]
print(data['arr_2'])  # 随机数组

# 记得关闭文件
data.close()
```

### 使用命名参数

```python
import numpy as np

# 使用关键字参数命名数组
train_data = np.random.rand(100, 10)
train_labels = np.random.randint(0, 10, size=100)
test_data = np.random.rand(20, 10)
test_labels = np.random.randint(0, 10, size=20)

# 保存时命名
np.savez('dataset.npz',
         train_data=train_data,
         train_labels=train_labels,
         test_data=test_data,
         test_labels=test_labels)

# 加载时使用名称访问
data = np.load('dataset.npz')

print(data['train_labels'].shape)  # (100,)
print(data['test_data'].shape)     # (20, 10)
print(data['test_labels'].shape)   # (20,)

# 查看所有数组名称
print(data.files)  # ['train_data', 'train_labels', 'test_data', 'test_labels']

data.close()
```

### savez_compressed() - 压缩保存

```python
import numpy as np
import os

# 创建大数组
big_array = np.random.rand(1000, 1000)

# 普通保存
np.savez('big_data.npz', data=big_array)
size1 = os.path.getsize('big_data.npz')

# 压缩保存
np.savez_compressed('big_data_compressed.npz', data=big_array)
size2 = os.path.getsize('big_data_compressed.npz')

print(f"普通保存: {size1 / 1024 / 1024:.2f} MB")
print(f"压缩保存: {size2 / 1024 / 1024:.2f} MB")
print(f"压缩率: {(1 - size2/size1) * 100:.1f}%")

# 加载方式相同
data1 = np.load('big_data.npz')
data2 = np.load('big_data_compressed.npz')

# 验证数据一致
print(np.array_equal(data1['data'], data2['data']))  # True
```

### 使用with语句

### 何时使用savez vs savez_compressed

```python
import numpy as np
import time

# 随机数据（不易压缩）
random_data = np.random.rand(1000, 1000)

# 规律数据（易压缩）
regular_data = np.ones((1000, 1000))

# 测试随机数据
start = time.time()
np.savez('random.npz', data=random_data)
t1 = time.time() - start

start = time.time()
np.savez_compressed('random_compressed.npz', data=random_data)
t2 = time.time() - start

print("随机数据:")
print(f"普通保存: {t1:.3f}秒")
print(f"压缩保存: {t2:.3f}秒")

# 测试规律数据
start = time.time()
np.savez('regular.npz', data=regular_data)
t3 = time.time() - start

start = time.time()
np.savez_compressed('regular_compressed.npz', data=regular_data)
t4 = time.time() - start

print("\n规律数据:")
print(f"普通保存: {t3:.3f}秒")
print(f"压缩保存: {t4:.3f}秒")
```

**选择建议**：
- **savez()** - 数据随机性强、需要快速保存
- **savez_compressed()** - 数据有规律、对文件大小敏感、长期存储

### 自定义格式

```python
import numpy as np

arr = np.array([[1.23456, 2.34567],
                [3.45678, 4.56789]])

# fmt参数控制格式
# %.2f - 保留2位小数
np.savetxt('float.txt', arr, fmt='%.2f')
# 文件内容: 1.23 2.35
#          3.46 4.57

# %d - 整数格式
int_arr = np.array([[1, 2], [3, 4]])
np.savetxt('int.txt', int_arr, fmt='%d')

# %.4e - 科学计数法，4位小数
big_arr = np.array([[1234.5678, 9876.5432]])
np.savetxt('scientific.txt', big_arr, fmt='%.4e')
# 文件内容: 1.2346e+03 9.8765e+03

# 混合格式
mixed = np.array([[1, 2.5, 3], [4, 5.5, 6]])
np.savetxt('mixed.txt', mixed, fmt=['%d', '%.2f', '%d'])
```

### 处理缺失值的实战技巧

```python
import numpy as np

# 创建测试数据
data = np.array([[1.0, 2.0, 3.0],
                 [4.0, np.nan, 6.0],
                 [7.0, 8.0, np.nan],
                 [np.nan, 10.0, 11.0]])

# 1. 删除含缺失值的行
clean_data = data[~np.isnan(data).any(axis=1)]
print("删除缺失行:")
print(clean_data)
# [[1. 2. 3.]]

# 2. 删除含缺失值的列
clean_data = data[:, ~np.isnan(data).any(axis=0)]
print("\n删除缺失列:")
print(clean_data)
# []  (所有列都有缺失值)

# 3. 用均值填充
data_filled = data.copy()
for i in range(data.shape[1]):
    col = data[:, i]
    col_mean = np.nanmean(col)  # 忽略nan计算均值
    col[np.isnan(col)] = col_mean
print("\n均值填充:")
print(data_filled)

# 4. 用中位数填充
data_filled = data.copy()
for i in range(data.shape[1]):
    col = data_filled[:, i]
    col_median = np.nanmedian(col)
    col[np.isnan(col)] = col_median
print("\n中位数填充:")
print(data_filled)

# 5. 前向填充
data_filled = data.copy()
for i in range(data.shape[1]):
    col = data_filled[:, i]
    mask = np.isnan(col)
    idx = np.where(~mask)[0]
    if len(idx) > 0:
        col[mask] = np.interp(np.flatnonzero(mask), idx, col[idx])
print("\n插值填充:")
print(data_filled)
```

## 原始二进制：tofile()和fromfile()

### 基本用法

```python
import numpy as np

# 创建数组
arr = np.array([1, 2, 3, 4, 5], dtype=np.int32)

# 保存为原始二进制
arr.tofile('data.bin')
```

### 注意事项

### 保存形状信息

```python
import numpy as np

def save_array_with_shape(filename, arr):
    """保存数组及其形状信息"""
    # 先保存形状
    header = np.array(arr.shape, dtype=np.int32)
    with open(filename, 'wb') as f:
        header.tofile(f)
        arr.tofile(f)

def load_array_with_shape(filename, dtype):
    """加载数组并恢复形状"""
    with open(filename, 'rb') as f:
        # 读取形状信息
        ndim = np.fromfile(f, dtype=np.int32, count=1)[0]
        shape = tuple(np.fromfile(f, dtype=np.int32, count=ndim))
        # 读取数据
        data = np.fromfile(f, dtype=dtype)
        return data.reshape(shape)

# 使用示例
arr = np.random.rand(3, 4, 5)
save_array_with_shape('array.bin', arr)

loaded = load_array_with_shape('array.bin', np.float64)
print(loaded.shape)  # (3, 4, 5)
print(np.array_equal(arr, loaded))  # True
```

```python
        """保存模型相关数据"""
        timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
        filename = os.path.join(self.data_dir, f'model_{timestamp}.npz')
        np.savez_compressed(filename, **arrays)
        print(f"模型数据已保存: {filename}")
        return filename

    def save_results(self, data, name):
        """保存计算结果"""
        filename = os.path.join(self.data_dir, f'{name}.npy')
        np.save(filename, data)
        print(f"结果已保存: {filename}")
        return filename

    def export_csv(self, data, name, header=None):
        """导出CSV供外部使用"""
        filename = os.path.join(self.data_dir, f'{name}.csv')
        if header:
            np.savetxt(filename, data, delimiter=',',
                      header=header, comments='')
        else:
            np.savetxt(filename, data, delimiter=',')
        print(f"CSV已导出: {filename}")
        return filename

        """加载模型数据"""
        filepath = os.path.join(self.data_dir, filename)
        with np.load(filepath) as data:
            return {key: data[key] for key in data.files}

    def load_results(self, name):
        """加载结果"""
        filename = os.path.join(self.data_dir, f'{name}.npy')
        return np.load(filename)
```

```python
def process_large_file(filename, chunk_size=1000):

    n_samples = data.shape[0]

    results = []
    for i in range(0, n_samples, chunk_size):
        chunk = data[i:i+chunk_size]
        # 处理chunk
        result = chunk.mean(axis=1)
        results.append(result)

    return np.concatenate(results)
```

```python
import numpy as np
import pandas as pd

class DataPipeline:
    """数据处理管道"""

    def __init__(self):
        self.data = None

    def load_from_csv(self, filename):
        """从CSV加载数据（用Pandas）"""
        df = pd.read_csv(filename)
        print(f"加载数据: {df.shape}")
        return df

    def preprocess_pandas(self, df):
        """使用Pandas预处理"""
        # 处理缺失值
        df = df.fillna(df.mean())
        # 筛选数据
        df = df[df['Score'] > 60]
        return df

    def compute_numpy(self, df):
        """转换为NumPy进行数值计算"""
        arr = df[['Math', 'English', 'Physics']].values

        # NumPy的快速计算
        mean_scores = arr.mean(axis=1)
        std_scores = arr.std(axis=1)
        normalized = (arr - arr.mean(axis=0)) / arr.std(axis=0)

        return {
            'mean': mean_scores,
            'std': std_scores,
            'normalized': normalized
        }

    def save_results(self, results, prefix='result'):
        """保存结果（混合使用）"""
        # NumPy格式保存（供Python使用）
        np.savez_compressed(f'{prefix}.npz', **results)

        # CSV格式保存（供Excel使用）
        df_result = pd.DataFrame(results['normalized'],
                                 columns=['Math', 'English', 'Physics'])
        df_result['Mean'] = results['mean']
        df_result['Std'] = results['std']
        df_result.to_csv(f'{prefix}.csv', index=False)

        print(f"结果已保存: {prefix}.npz 和 {prefix}.csv")

# 使用示例
pipeline = DataPipeline()

# 创建测试数据
test_df = pd.DataFrame({
    'Name': ['Alice', 'Bob', 'Charlie', 'Diana'],
    'Math': [85, 90, 55, 92],  # Charlie不及格

    'Physics': [78, 95, 50, 88],
    'Score': [85, 91, 55, 92]
})
test_df.to_csv('test_scores.csv', index=False)

# 执行管道
```

```python
import numpy as np
import os

class ImageDataset:
    """图像数据集管理"""

    def __init__(self, data_dir='image_data'):
        self.data_dir = data_dir
        if not os.path.exists(data_dir):
            os.makedirs(data_dir)

    def save_dataset(self, images, labels, name='dataset'):
        """保存图像数据集"""
        filename = os.path.join(self.data_dir, f'{name}.npz')

        # 压缩保存（图像数据通常可以很好地压缩）
        np.savez_compressed(filename,
                          images=images,
                          labels=labels,
                          image_shape=images.shape[1:],
                          num_classes=len(np.unique(labels)))

        print(f"数据集已保存: {filename}")
        print(f"图像数量: {len(images)}")
        print(f"图像形状: {images.shape[1:]}")
        print(f"类别数: {len(np.unique(labels))}")

    def load_dataset(self, name='dataset'):
        """加载图像数据集"""
        filename = os.path.join(self.data_dir, f'{name}.npz')

        with np.load(filename) as data:
            images = data['images']
            labels = data['labels']

            print(f"数据集已加载: {filename}")
            print(f"图像数量: {len(images)}")
            print(f"图像形状: {images.shape[1:]}")

            return images, labels

    def split_dataset(self, images, labels, train_ratio=0.8):
        """划分训练集和测试集"""
        n_samples = len(images)
        n_train = int(n_samples * train_ratio)

        # 随机打乱
        indices = np.random.permutation(n_samples)
        train_idx = indices[:n_train]
        test_idx = indices[n_train:]

        return (images[train_idx], labels[train_idx],
                images[test_idx], labels[test_idx])

    def save_splits(self, images, labels, train_ratio=0.8):
        """划分并保存训练集和测试集"""
        train_X, train_y, test_X, test_y = self.split_dataset(
            images, labels, train_ratio)

        self.save_dataset(train_X, train_y, 'train')
        self.save_dataset(test_X, test_y, 'test')

        return train_X, train_y, test_X, test_y

# 使用示例
dataset = ImageDataset()

# 模拟图像数据 (1000张28x28的灰度图像)
images = np.random.randint(0, 256, size=(1000, 28, 28), dtype=np.uint8)
labels = np.random.randint(0, 10, size=1000)
```

```python
import numpy as np
from datetime import datetime, timedelta

class TimeSeriesManager:
    """时间序列数据管理"""

    def __init__(self, base_dir='timeseries_data'):
        self.base_dir = base_dir
        if not os.path.exists(base_dir):
            os.makedirs(base_dir)

    def generate_sample_data(self, n_days=365, n_features=5):
        """生成示例时间序列数据"""
        timestamps = [datetime.now() - timedelta(days=i)
                     for i in range(n_days)]
        timestamps = np.array(timestamps, dtype='datetime64[s]')

        # 生成趋势 + 季节性 + 噪声
        t = np.arange(n_days)
        data = np.zeros((n_days, n_features))

        for i in range(n_features):
            trend = t * 0.1
            seasonal = 10 * np.sin(2 * np.pi * t / 365)
            noise = np.random.randn(n_days) * 2
            data[:, i] = trend + seasonal + noise + 50

        return timestamps, data

    def save_timeseries(self, timestamps, data, features, name='series'):
        """保存时间序列"""
        filename = os.path.join(self.base_dir, f'{name}.npz')

        # 转换时间戳为可保存的格式
        timestamps_int = timestamps.astype(np.int64)

        np.savez_compressed(filename,
                          timestamps=timestamps_int,
                          data=data,
                          features=features)

        print(f"时间序列已保存: {filename}")
        print(f"时间范围: {timestamps[0]} 到 {timestamps[-1]}")
        print(f"数据点数: {len(data)}")
        print(f"特征数: {len(features)}")

    def load_timeseries(self, name='series'):
        """加载时间序列"""
        filename = os.path.join(self.base_dir, f'{name}.npz')

        with np.load(filename, allow_pickle=True) as f:
            timestamps = f['timestamps'].astype('datetime64[s]')
            data = f['data']
            features = f['features']

            print(f"时间序列已加载: {filename}")
            return timestamps, data, features

    def compute_statistics(self, data, features):
        """计算统计信息"""
        stats = {}
        for i, feature in enumerate(features):
            stats[feature] = {
                'mean': data[:, i].mean(),
                'std': data[:, i].std(),
                'min': data[:, i].min(),
                'max': data[:, i].max(),
                'median': np.median(data[:, i])
            }
        return stats

    def export_statistics(self, stats, name='stats'):
        """导出统计信息"""
        # 转换为数组以便保存
        features = list(stats.keys())
        metrics = ['mean', 'std', 'min', 'max', 'median']

        data = np.array([[stats[f][m] for m in metrics]
                        for f in features])

        # 保存为CSV（供查看）
        csv_file = os.path.join(self.base_dir, f'{name}.csv')
        header = ','.join(metrics)
        np.savetxt(csv_file, data, delimiter=',',
                  header=header, comments='')

        # 保存为NPY（供程序使用）
        npy_file = os.path.join(self.base_dir, f'{name}.npy')
        np.save(npy_file, data)

        print(f"统计信息已导出: {csv_file} 和 {npy_file}")
```

```python
import numpy as np
import os

class DataAugmenter:
    """数据增强与缓存管理"""

    def __init__(self, cache_dir='augmented_cache'):
        self.cache_dir = cache_dir
        if not os.path.exists(cache_dir):
            os.makedirs(cache_dir)

    def augment_image(self, image):
        """图像增强"""
        augmented = []

        # 原始图像
        augmented.append(image)

        # 水平翻转
        augmented.append(np.fliplr(image))

        # 垂直翻转
        augmented.append(np.flipud(image))

        # 旋转90度
        augmented.append(np.rot90(image))

        # 旋转180度
        augmented.append(np.rot90(image, 2))

        # 旋转270度
        augmented.append(np.rot90(image, 3))

        # 添加噪声
        noisy = image + np.random.randn(*image.shape) * 10
        noisy = np.clip(noisy, 0, 255).astype(image.dtype)
        augmented.append(noisy)

        return np.array(augmented)

    def augment_dataset(self, images, labels, cache_name='augmented'):
        """增强整个数据集"""
        print(f"增强数据集 ({len(images)} 张图像)...")

        augmented_images = []
        augmented_labels = []

        for i, (img, label) in enumerate(zip(images, labels)):
            if (i + 1) % 100 == 0:
                print(f"进度: {i+1}/{len(images)}")

            aug_imgs = self.augment_image(img)
            augmented_images.append(aug_imgs)
            augmented_labels.extend([label] * len(aug_imgs))

        augmented_images = np.concatenate(augmented_images, axis=0)
        augmented_labels = np.array(augmented_labels)

        # 缓存增强后的数据
        cache_file = os.path.join(self.cache_dir, f'{cache_name}.npz')
        np.savez_compressed(cache_file,
                          images=augmented_images,
                          labels=augmented_labels,
                          original_count=len(images),
                          augmented_count=len(augmented_images))

        print(f"数据增强完成: {len(images)} -> {len(augmented_images)}")
        print(f"缓存已保存: {cache_file}")

        return augmented_images, augmented_labels

    def load_augmented(self, cache_name='augmented'):
        """加载缓存的增强数据"""
        cache_file = os.path.join(self.cache_dir, f'{cache_name}.npz')

        if not os.path.exists(cache_file):
            raise FileNotFoundError(f"缓存不存在: {cache_file}")

        print(f"从缓存加载: {cache_file}")
        data = np.load(cache_file)

        print(f"原始数量: {data['original_count']}")
        print(f"增强数量: {data['augmented_count']}")

        return data['images'], data['labels']

    def check_cache(self, cache_name='augmented'):
        """检查缓存是否存在"""
        cache_file = os.path.join(self.cache_dir, f'{cache_name}.npz')
        return os.path.exists(cache_file)

# 使用示例
augmenter = DataAugmenter()

# 生成示例图像数据
images = np.random.randint(0, 256, size=(100, 28, 28), dtype=np.uint8)
labels = np.random.randint(0, 10, size=100)

# 检查缓存
if augmenter.check_cache('mnist_aug'):
    print("使用缓存数据")
    aug_images, aug_labels = augmenter.load_augmented('mnist_aug')
else:
    print("生成增强数据")
    aug_images, aug_labels = augmenter.augment_dataset(
        images, labels, 'mnist_aug')

print(f"\n最终数据集大小: {aug_images.shape}")
```

```python
import numpy as np
import json
from datetime import datetime

class DataWithMetadata:
    """带元数据的数据保存"""

    @staticmethod
    def save(data, filename, **metadata):
        """保存数据和元数据"""
        # 添加自动元数据
        auto_metadata = {
            'shape': data.shape,
            'dtype': str(data.dtype),
            'size_mb': data.nbytes / 1024 / 1024,
            'creation_date': datetime.now().isoformat(),
            'numpy_version': np.__version__
        }

        # 合并用户元数据
        full_metadata = {**auto_metadata, **metadata}

        # 保存
        np.savez_compressed(filename,
                          data=data,
                          metadata=json.dumps(full_metadata))

        print(f"数据已保存: {filename}")
        print(f"元数据: {full_metadata}")

    @staticmethod
    def load(filename):
        """加载数据和元数据"""
        with np.load(filename) as f:
            data = f['data']
            metadata = json.loads(str(f['metadata']))

        print(f"数据已加载: {filename}")
        print(f"形状: {metadata['shape']}")
        print(f"类型: {metadata['dtype']}")
        print(f"创建时间: {metadata['creation_date']}")

        return data, metadata

# 使用示例
data = np.random.rand(1000, 100)

                     description='Training data for model v2',
```

### 4. 错误处理

```python
import numpy as np
import os

def safe_save(data, filename, overwrite=False):
    """安全地保存数据"""
    # 检查目录是否存在
    directory = os.path.dirname(filename)
    if directory and not os.path.exists(directory):
        os.makedirs(directory)

    # 检查文件是否已存在
    if os.path.exists(filename) and not overwrite:
        backup = filename + '.backup'
        print(f"文件已存在，创建备份: {backup}")
        os.rename(filename, backup)

    try:
        # 保存数据
        if filename.endswith('.npz'):
            np.savez_compressed(filename, data=data)
        else:
            np.save(filename, data)

        print(f"保存成功: {filename}")
        return True

    except Exception as e:
        print(f"保存失败: {e}")

        # 如果有备份，恢复
        backup = filename + '.backup'
        if os.path.exists(backup):
            os.rename(backup, filename)
            print("已从备份恢复")

        return False

def safe_load(filename, default=None):
    """安全地加载数据"""
    if not os.path.exists(filename):
        print(f"文件不存在: {filename}")
        return default

    try:
        if filename.endswith('.npz'):
            with np.load(filename) as f:
                data = f['data']
        else:
            data = np.load(filename)

        print(f"加载成功: {filename}")
        return data

    except Exception as e:
        print(f"加载失败: {e}")
        return default

# 使用示例
data = np.random.rand(100, 50)
safe_save(data, 'test.npy')
loaded = safe_load('test.npy')
```

### 5. 性能监控

```python
import numpy as np
import time
import os
from functools import wraps

def monitor_io(func):
    """监控文件IO性能"""
    @wraps(func)
    def wrapper(filename, *args, **kwargs):
        start = time.time()
        result = func(filename, *args, **kwargs)
        elapsed = time.time() - start

        if os.path.exists(filename):
            size_mb = os.path.getsize(filename) / 1024 / 1024
            speed = size_mb / elapsed if elapsed > 0 else 0
            print(f"{func.__name__}: {elapsed:.3f}秒, "
                  f"{size_mb:.2f}MB, {speed:.2f}MB/s")
        else:
            print(f"{func.__name__}: {elapsed:.3f}秒")

        return result
    return wrapper

@monitor_io
def save_with_monitor(filename, data):
    """带监控的保存"""
    np.save(filename, data)

@monitor_io
def load_with_monitor(filename):
    """带监控的加载"""
    return np.load(filename)

# 使用示例
data = np.random.rand(10000, 1000)
save_with_monitor('test.npy', data)
loaded = load_with_monitor('test.npy')
```
